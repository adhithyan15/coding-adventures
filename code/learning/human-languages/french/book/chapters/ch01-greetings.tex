\chapter{Greetings}
\label{ch:greetings}

The French greetings, built from their pieces --- a word for ``good,'' words
for ``day,'' ``evening,'' ``night,'' and the gender machine that decides their
endings. Where a French word has a Spanish twin, both appear, because the
\emph{difference} between them is often the lesson. \emph{Practice lessons:
\texttt{lessons/FR-C01-*.md}.}

\section{\emph{salut} --- ``hi,'' a wish for your health}
\label{lesson:salut}

\begin{sounds}
Final \textbf{t} is silent: \emph{salut} = \emph{sa-LU}. The \textbf{u} is the
front-rounded French \emph{u}.
\end{sounds}

\begin{cousinweb}
\textbf{salut} (informal ``hi/bye'') $\leftarrow$ Latin \emph{salus, salutem}
(``health, safety''). So \emph{salut} literally wishes \emph{health} --- like
old English ``hail.'' Cousins: \textbf{salut}e, \textbf{salut}ation,
\textbf{salut}ary, \textbf{salu}brious; and via \emph{salvus} (``safe''),
\emph{save}, \emph{salvation}.
\end{cousinweb}

\begin{culture}
Strictly \textbf{informal}, and doubles as ``bye.'' For respect or a first
meeting you reach for \emph{bonjour} --- exactly as Spanish reaches past
\emph{hola} for \emph{buenos d\'{i}as}.
\end{culture}

\section{\emph{bien} --- ``well,'' and a Spanish twin}
\label{lesson:bien}

\begin{cousinweb}
\textbf{bien} (``well'') $\leftarrow$ Latin \emph{bene} $\to$ \textbf{bene}fit,
\textbf{bene}volent, \textbf{bene}diction. \emph{Spanish twin:} Spanish also
has \emph{bien}, also from \emph{bene} --- the two Romance languages inherited
the same adverb almost unchanged.
\end{cousinweb}

A complete one-word answer: \emph{Comment \c{c}a va?} $\to$ \emph{Bien.};
intensified, \emph{tr\`{e}s bien}. The adverb (``well''); the \emph{adjective}
``good'' is a different word, \emph{bon}, next.

\section{\emph{bon} / \emph{bonne} --- ``good''}
\label{lesson:bon}

\begin{sounds}
\textbf{bon} is nasal \emph{b\~{o}} (silent \emph{n}); but \textbf{bonne}
(feminine) is \emph{bun} --- the doubled \emph{nn} cancels the nasal, so the
two \emph{sound} different despite looking alike.
\end{sounds}

\begin{cousinweb}
\textbf{bon / bonne} $\leftarrow$ Latin \emph{bonus / bona} $\to$
\textbf{bon}us, \textbf{bon}anza, \textbf{boun}ty (via \emph{bont\'{e}}),
\emph{bonbon}. \emph{Spanish twin:} \emph{bueno / buena} --- same root; French
kept it short (\emph{bon}) where Spanish broke the \emph{o} into a diphthong.
\end{cousinweb}

\begin{grammarlens}
\emph{bon} \textbf{agrees} with its noun (the \emph{bueno/buena} machine from
Spanish): \emph{bon} (m.), \emph{bonne} (f.), \emph{bons} (m.pl.),
\emph{bonnes} (f.pl.). You'll watch it: \emph{bon jour} (m.) but \emph{bonne
nuit} (f.).
\end{grammarlens}

\section{\emph{le} / \emph{la} / \emph{les} --- ``the,'' and the gender it carries}
\label{lesson:le-la}

\begin{cousinweb}
\textbf{le} $\leftarrow$ \emph{ille}, \textbf{la} $\leftarrow$ \emph{illa},
\textbf{les} $\leftarrow$ \emph{illos} --- Latin demonstratives (``that
one/those'') worn into ``the.'' \emph{Spanish twin, exactly:} \emph{el / la /
los}, same source; and the same twins gave French \textbf{il} / \textbf{elle}
(``he/she''). (English ``the'' is Germanic --- unrelated.)
\end{cousinweb}

\begin{grammarlens}
Every French noun is masculine or feminine --- Latin's three genders collapsed
to two, as in Spanish. Learn each noun \emph{with} its article. One mercy over
Spanish: plurals share a single article, \emph{les}, for both genders.
\end{grammarlens}

\section{\emph{jour} --- ``day,'' and the road to English \emph{journey}}
\label{lesson:jour}

\begin{sounds}
\textbf{j} = \emph{zh} (as in \emph{measure}); \textbf{ou} = \emph{oo}. So
\emph{jour} = \emph{zhoor}.
\end{sounds}

\begin{cousinweb}
\textbf{le jour} (masc.) $\leftarrow$ Latin \emph{diurnum} (``daytime''), from
\emph{dies} (``day''). That detour through \emph{diurnum} is why \emph{jour}
looks nothing like Spanish \emph{d\'{i}a} (which took \emph{dies} directly).
Cousins, mostly via French: \textbf{jour}nal, \textbf{jour}ney (a \emph{day's}
travel --- \emph{journ\'{e}e}), so\textbf{journ}, ad\textbf{journ}
(\emph{\`{a} jour}); and via \emph{dies}: \textbf{di}urnal, \textbf{di}ary.\\
\emph{So English \emph{journey} and Spanish \emph{d\'{i}a} are cousins},
meeting at Latin \emph{dies} --- one through French, one direct.
\end{cousinweb}

\begin{grammarlens}
\emph{jour} is \textbf{masculine} (\emph{le jour}). Plural adds a silent
\emph{-s} (\emph{les jours}) --- heard only in \emph{les}.
\end{grammarlens}

\section{\emph{bonjour} --- assembled, and \emph{not} plural}
\label{lesson:bonjour}

\emph{bon} (``good'') $+$ \emph{jour} (``day'') $=$ ``good day,'' one word,
\emph{b\~{o}-zhoor}; \emph{bon} is masculine because \emph{jour} is.

\begin{culture}
The contrast to bank: Spanish says \emph{buenos d\'{i}as} --- \textbf{plural},
the fossil of a blessing. French says \emph{bonjour} --- \textbf{singular},
just ``good day.'' Same two blocks, same roots; Spanish pluralized the
blessing, French didn't. And culturally \emph{bonjour} is near-obligatory ---
opening a shop or a conversation \emph{without} one reads as rude.
\end{culture}

\section{\emph{soir} --- ``evening,'' the same ``late'' Spanish used}
\label{lesson:soir}

\begin{cousinweb}
\textbf{le soir} (masc., \emph{swahr}) $\leftarrow$ Latin \emph{s\={e}rum}
(``the \textbf{late} hour''), from \emph{s\={e}rus} (``late''). Cousin:
\emph{soir\'{e}e}. \emph{Spanish parallel:} \emph{tarde} (``afternoon'')
$\leftarrow$ \emph{tardus}, ``\textbf{late}.'' Two languages, two ``late''
words, the same instinct --- name the later day after its lateness.
\end{cousinweb}

\section{\emph{bonsoir} --- good evening}
\label{lesson:bonsoir}

\emph{bon} $+$ \emph{soir} (masc.) $=$ ``good evening,'' \emph{b\~{o}-swahr};
\emph{bon} stays masculine. The evening swap for \emph{bonjour}, and --- like
Spanish \emph{buenas noches} --- usable both arriving and leaving.

\section{\emph{nuit} --- ``night,'' and a sound-change mirroring Spanish}
\label{lesson:nuit}

\begin{cousinweb}
\textbf{la nuit} (fem., \emph{nwee}) $\leftarrow$ Latin \emph{noctem}
(\emph{noct-}) $\to$ \textbf{noct}urnal, equi\textbf{nox}. The change is a
\emph{rule}, side by side with Spanish: Latin \emph{-ct-} $\to$ Spanish
\emph{-ch-}, French \emph{-it-}.
\end{cousinweb}

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Latin & Spanish & French & English \\
\midrule
\emph{noctem} & noche & nuit & night \\
\emph{lactem} & leche & lait & (lact-) \\
\emph{factum} & hecho & fait & fact \\
\bottomrule
\end{tabular}
\end{center}

\begin{grammarlens}
\emph{nuit} is \textbf{feminine} (\emph{la nuit}, from feminine Latin
\emph{nox}) --- which is why ``good night'' is \emph{bon\textbf{ne} nuit}.
\end{grammarlens}

\section{\emph{bonne nuit} --- good night, and the feminine \emph{bon}}
\label{lesson:bonne-nuit}

\emph{bonne} (fem.) $+$ \emph{nuit} (fem.) $=$ ``good night'' (\emph{bun
nwee}). At last \emph{bon} in its feminine coat --- and the whole agreement
rule in one view:

\begin{center}
\emph{bon\textbf{jour}} (\emph{jour} m.\ $\to$ \emph{bon}) \quad
\emph{bon\textbf{soir}} (\emph{soir} m.\ $\to$ \emph{bon}) \quad
\emph{bon\textbf{ne} nuit} (\emph{nuit} f.\ $\to$ \emph{bonne})
\end{center}

\begin{culture}
Unlike \emph{bonsoir} (an evening greeting), \emph{bonne nuit} is a
\textbf{farewell for bed}. French keeps them separate where Spanish overloads
\emph{buenas noches} for both. Next chapter: introducing yourself, and
French's own \emph{tu} vs.\ \emph{vous}.
\end{culture}
