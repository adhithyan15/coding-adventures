\chapter{How Are You?}
\label{ch:responding}

For most Gujaratis this is the real everyday hello:

\begin{center}
\gu{તમે કેમ છો}? \quad(\emph{tame kem chho} --- ``How are you?'')\\
\gu{હું મજામાં છું, આભાર}. \quad(\emph{hũ maj\=am\=a\d{m} chhũ, \=abh\=ar} --- ``I'm well, thank you.'')
\end{center}

\emph{Practice lessons: \texttt{lessons/GU-C03-*.md}.}

\section{\gu{કેમ} \emph{(kem)} --- ``how,'' a proper k- question}
\label{lesson:kem}

\begin{cousinweb}
\gu{કેમ} (\emph{kem}, ``how''; also ``why'') is from the interrogative stem
\emph{ka-}, PIE *k\ensuremath{^w}o- --- the root behind English \textbf{wh-} and
Latin \emph{qu-}. It heads the plain \emph{k-} family Gujarati kept: \emph{kem}
(how), \emph{k\=o\d{n}} (who), \emph{ky\=a\d{m}} (where), \emph{ky\=are} (when).
And \gu{કેમ છો}? is, for most Gujaratis, the real everyday hello.
\end{cousinweb}

\section{\gu{તમે કેમ છો}? --- ``how are you?''}
\label{lesson:tame-kem-chho}

\gu{તમે} (\emph{tame}, respectful ``you'') $+$ \gu{કેમ} (how) $+$ \gu{છો}
(\emph{chho}, ``are'') --- ``you how are?'', verb last. \emph{Chho} is the
respectful/plural form of the copula \emph{chhe} from Chapter 2. To a friend:
\gu{તું કેમ છે}? (\emph{tũ kem chhe?}).

\section{\gu{હું} \emph{(hũ)} --- ``I''}
\label{lesson:hun}

\begin{cousinweb}
\gu{હું} (\emph{hũ}, ``I'') descends from Sanskrit \emph{aham} (through Old
Gujarati \emph{haũ}). That \emph{aham} is PIE *h\textsubscript{1}e\'{g}(H)om ---
the ancestor of Latin \textbf{ego} and English \textbf{I}. It takes the copula
\emph{chhũ} (``am''): \emph{hũ \ldots chhũ}.
\end{cousinweb}

\section{\gu{મજા} \emph{(maj\=a)} --- ``enjoyment,'' and how to answer}
\label{lesson:majaa}

\begin{cousinweb}
\gu{મજા} (\emph{maj\=a}, ``enjoyment, relish, fun'') is \textbf{not} Sanskrit ---
it is \textbf{Perso-Arabic}, from Persian \emph{maza} (``taste, relish''). Gujaratis,
a great trading people, wove thousands of Persian and Arabic words into their
speech across centuries of commerce over the Arabian Sea --- and even ``how are
you'' is answered with one: \gu{હું મજામાં છું} (\emph{hũ maj\=am\=a\d{m} chhũ}),
literally ``I am \emph{in enjoyment}.''
\end{cousinweb}

\section{\gu{વાંધો નહીં} \emph{(v\=andho nah\=\i\d{m})} --- ``no problem''}
\label{lesson:vandho-nahi}

\begin{cousinweb}
\gu{વાંધો નહીં} literally ``\textbf{[there is] no objection}'' --- \emph{v\=andho}
(``objection, difficulty'') $+$ \gu{નહીં} (\emph{nah\=\i\d{m}}, ``not''), the
fuller form of Chapter 1's \emph{n\=a}. Both from the ancient \emph{na-}, PIE
*ne, behind English \textbf{no}, \textbf{not}.
\end{cousinweb}

\section{The whole exchange}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Gujarati & English \\
\midrule
\gu{તમે કેમ છો}? & How are you? \\
\gu{હું મજામાં છું, આભાર}. & I'm well, thank you. \\
\gu{વાંધો નહીં}. & No problem / you're welcome. \\
\bottomrule
\end{tabular}
\end{center}

Roots banked: \emph{kem} (the \emph{k-} questions), \emph{hũ} (\emph{aham} $\to$
Latin \emph{ego}, English \emph{I}), \emph{maj\=a} (\textbf{Persian} \emph{maza}
--- the trade-language layer), \emph{nah\=\i\d{m}} (PIE *ne, English \emph{no}).
