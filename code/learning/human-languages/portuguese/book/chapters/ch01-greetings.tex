\chapter{Greetings}
\label{ch:greetings}

The Portuguese greetings, built from their pieces --- a word for ``good,'' the
day-parts, the article that carries gender, and a ``thank you'' that changes
with \emph{who is speaking}. Each word is taken apart to its root and its
English cousins. \emph{Practice lessons: \texttt{lessons/PT-C01-*.md}.}

\section{\emph{ol\'{a}} --- ``hi''}
\label{lesson:ola}

\begin{sounds}
\emph{oh-LAH} --- stress on the second syllable (the written accent on \emph{\'{a}}
marks it). The \emph{l} is clear, the \emph{a} open.
\end{sounds}

\begin{cousinweb}
\textbf{ol\'{a}} (``hi'') is a friendly interjection of uncertain birth ---
probably a natural call for attention (like English \emph{hello} / \emph{hallo}),
possibly reinforced by Arabic \emph{w-All\=ah} (``by God'') during the centuries
of Moorish Iberia. Unlike most Portuguese words, it has \emph{no} clean Latin
root: some greetings are just sounds. Spanish, its close sibling, has the near-
identical \emph{hola}.
\end{cousinweb}

\begin{culture}
\emph{ol\'{a}} is friendly and fairly neutral --- fine for most people, though for
real formality or a first meeting you add \emph{bom dia} / \emph{boa tarde},
built next. In Brazil you'll also hear \emph{oi} for a casual ``hi.''
\end{culture}

\section{\emph{bom} / \emph{boa} --- ``good''}
\label{lesson:bom}

\begin{sounds}
\textbf{bom} is \textbf{nasal} --- \emph{bõ}, the \emph{m} not fully closed, the
vowel sent through the nose. \textbf{boa} (feminine) is a clear \emph{BOH-ah}.
\end{sounds}

\begin{cousinweb}
\textbf{bom / boa} (``good'') $\leftarrow$ Latin \emph{bonus / bona} $\to$ English
\textbf{bon}us, \textbf{bon}anza. Spanish, another daughter of Latin, made
\emph{bueno / buena}; Portuguese wore \emph{bonus} down further, to the little
nasal \emph{bom}.
\end{cousinweb}

\begin{grammarlens}
\textbf{bom / boa} \textbf{agrees} with its noun: \emph{bom} (m.), \emph{boa}
(f.), \emph{bons} / \emph{boas} (pl.) --- Latin's adjective agreement, alive
across the Romance family. So it's \emph{bom dia} (m.) but \emph{boa tarde} (f.).
\end{grammarlens}

\section{\emph{o} / \emph{a} --- ``the,'' and an article worn to a single letter}
\label{lesson:o-a}

\begin{cousinweb}
\textbf{o} (masculine ``the'') and \textbf{a} (feminine) $\leftarrow$ Latin
\emph{ille} / \emph{illa} (``that one''). Here Portuguese did something striking:
it wore the word down \emph{further than any of its sisters} --- Latin
\emph{ille} $\to$ Italian \emph{il} $\to$ Spanish \emph{el} $\to$ Portuguese
just \textbf{o}. The whole demonstrative has eroded to a single vowel. (English
``the'' is Germanic, unrelated.)
\end{cousinweb}

\begin{grammarlens}
Every Portuguese noun is \textbf{masculine or feminine} --- Latin's genders
collapsed to two. Learn each noun \emph{with} its article: \emph{o} for
masculine, \emph{a} for feminine. Beware the trap coming next: \emph{o dia}
(``the day'') is \emph{masculine} even though it ends in \emph{-a}.
\end{grammarlens}

\section{\emph{dia} --- ``day''}
\label{lesson:dia}

\begin{cousinweb}
\textbf{o dia} (masc.) $\leftarrow$ Latin \emph{dies} (``day'') --- taken almost
straight, like Spanish \emph{d\'{i}a}. Cousins straight from \emph{dies}:
\textbf{di}urnal, \textbf{di}ary; and, through the derived \emph{diurnum},
English \textbf{jour}nal and \textbf{jour}ney (which is why Italian \emph{giorno}
and French \emph{jour} look so different --- they took the longer road).
\end{cousinweb}

\begin{grammarlens}
\textbf{A gender trap.} \emph{dia} ends in \emph{-a} --- the usual feminine
signal --- yet it is \emph{masculine}: \emph{o dia}. This is a fossil of its
Latin origin (\emph{dies} was masculine), and a reminder to learn each noun
\emph{with} its article rather than guessing gender from the ending.
\end{grammarlens}

\section{\emph{bom dia} --- ``good day / good morning,'' assembled}
\label{lesson:bom-dia}

\emph{bom} (``good'') $+$ \emph{dia} (``day'') $=$ \textbf{bom dia}, ``good
day'' --- and, in practice, ``good morning'': it's the greeting until noon.

\begin{culture}
\emph{bom dia} is the polite, all-purpose morning hello --- safe with anyone.
After midday it gives way to \emph{boa tarde}, and after dark to \emph{boa
noite}. (Spanish pluralises the blessing --- \emph{buenos d\'{i}as} --- where
Portuguese keeps it singular, \emph{bom dia}.)
\end{culture}

\section{\emph{tarde} --- ``afternoon,'' and \emph{boa tarde}}
\label{lesson:tarde}

\begin{cousinweb}
\textbf{a tarde} (fem.) $\leftarrow$ Latin \emph{tardus} (``slow, late'') --- the
same \emph{tardus} behind English \textbf{tard}y and re\textbf{tard} (``to
slow''). Portuguese named the afternoon after its \emph{lateness}, exactly as
Spanish did with the identical \emph{tarde}. \textbf{boa tarde} $=$ ``good
afternoon.''
\end{cousinweb}

\section{\emph{noite} --- ``night,'' and \emph{boa noite}}
\label{lesson:noite}

\begin{cousinweb}
\textbf{a noite} (fem.) $\leftarrow$ Latin \emph{noctem} (\emph{noct-}) $\to$
English \textbf{noct}urnal, equi\textbf{nox}. Portuguese turned the Latin
\emph{-ct-} into \textbf{-it-} (\emph{noite}) --- the same change French made
(\emph{nuit}), while Spanish went \emph{-ch-} (\emph{noche}) and Italian
\emph{-tt-} (\emph{notte}).
\end{cousinweb}

\begin{center}
\begin{tabular}{@{}lllll@{}}
\toprule
Latin & Portuguese & Spanish & French & Italian \\
\midrule
\emph{noctem} & no\emph{it}e & no\emph{ch}e & nu\emph{it} & no\emph{tt}e \\
\emph{factum} & fe\emph{it}o & he\emph{ch}o & fa\emph{it} & fa\emph{tt}o \\
\bottomrule
\end{tabular}
\end{center}

\textbf{boa noite} $=$ ``good night'' --- used both as an evening greeting and as
the farewell for bed.

\section{\emph{obrigado} / \emph{obrigada} --- ``thank you,'' i.e. ``[I am] obliged''}
\label{lesson:obrigado}

\begin{cousinweb}
\textbf{obrigado / obrigada} (``thanks'') $\leftarrow$ Latin \emph{oblig\=atus}
(``bound, obliged''), from \emph{ob-} (``to'') $+$ \emph{lig\=are} (``to bind'').
It is the \emph{same} word as English \textbf{obligated}, and shares its root
with \textbf{oblige}, \textbf{oblig}ation, and (via \emph{ligare}) \textbf{lig}ament
and \textbf{li}able. To say \emph{obrigado} is to say ``I am \emph{bound} to you''
--- indebted by your kindness.
\end{cousinweb}

\begin{grammarlens}
\textbf{It agrees with the speaker, not the listener.} Because \emph{obrigado}
literally means ``[I am] obliged,'' it takes the gender of \emph{whoever is
speaking}: a man says \textbf{obrigado}, a woman says \textbf{obrigada} --- no
matter who they're thanking. This is unlike the other tracks' adjectives (which
match the noun); here the adjective matches \emph{you}.
\end{grammarlens}

\section{The day of greetings}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Portuguese & English \\
\midrule
\emph{ol\'{a}} & hi \\
\emph{bom dia} & good morning \\
\emph{boa tarde} & good afternoon \\
\emph{boa noite} & good night \\
\emph{obrigado / obrigada} & thank you \\
\bottomrule
\end{tabular}
\end{center}

Every piece traced: \emph{bom} (\emph{bonus}), \emph{dia} (\emph{dies}),
\emph{tarde} (\emph{tardus}, ``late''; English \emph{tardy}), \emph{noite}
(\emph{noctem}, \emph{-ct-}$\to$\emph{-it-}), \emph{obrigado} (\emph{oblig\=atus},
``obliged''). And two Portuguese habits: the article eroded to a single vowel
(\emph{o} / \emph{a}), and ``thank you'' agrees with the \emph{speaker}
(\emph{obrigado} / \emph{obrigada}). Next chapter: introducing yourself ---
\emph{o meu nome \'{e}\ldots} / \emph{chamo-me\ldots} --- and Portuguese's
\emph{tu} vs.\ \emph{voc\^{e}} (familiar / polite ``you'').
