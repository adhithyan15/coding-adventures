\section*{The five pure vowels}

Spanish has exactly five vowel sounds, one per letter, every time --- unlike
English, which reuses each vowel letter for several sounds.

\begin{itemize}
  \item \textbf{a} --- as in \emph{father}. Always.
  \item \textbf{e} --- as in \emph{bed}. Always.
  \item \textbf{i} --- as the \emph{ee} in \emph{see} (not English \emph{i}).
    The most common vowel slip for English speakers.
  \item \textbf{o} --- as in \emph{note}, short and clean, no \emph{w}-glide.
  \item \textbf{u} --- as the \emph{oo} in \emph{food}. Always.
\end{itemize}

\section*{Consonants that differ from English}

\begin{itemize}
  \item \textbf{h} --- always silent. \emph{hola} is \emph{OH-la}.
  \item \textbf{qu} --- a hard \emph{k}, with a \textbf{silent u}: \emph{quiero}
    is \emph{KYEH-ro}.
  \item \textbf{ll} --- like English \emph{y}: \emph{llamo} is \emph{YA-mo}.
  \item \textbf{\~{n}} --- the \emph{ny} of \emph{canyon}: \emph{espa\~{n}ol}.
  \item \textbf{rr} --- a rolled \emph{r}; single \emph{r} between vowels is a
    light tap (\emph{pero} vs.\ \emph{perro}).
  \item \textbf{j} --- a throat-\emph{h}, like Scottish \emph{loch}: \emph{jueves}.
  \item \textbf{c} before \emph{e/i} --- soft (\emph{s} in Latin America,
    \emph{th} in much of Spain): \emph{gracias}.
  \item \textbf{d} between vowels --- softens toward \emph{th} in \emph{this}.
  \item \textbf{v} and \textbf{b} --- essentially identical, a soft \emph{b}.
\end{itemize}

\section*{Stress and written accents}

\begin{itemize}
  \item Ends in a vowel, \emph{n}, or \emph{s} $\to$ stress the
    second-to-last syllable (\emph{ca-SA}).
  \item Ends in another consonant $\to$ stress the last syllable
    (\emph{ha-BLAR}).
  \item A written accent (\'{}) overrides both and marks the stress exactly
    (\emph{ha-BL\'{O}}, \emph{D\'{I}-as}); it also separates look-alike words
    (\emph{qu\'{e}}/\emph{que}, \emph{t\'{u}}/\emph{tu}).
  \item Questions and exclamations open with an inverted mark
    (\textquestiondown, \textexclamdown) so the intonation is signalled up
    front.
\end{itemize}
