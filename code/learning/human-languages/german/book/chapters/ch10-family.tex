\chapter{Family}
\label{ch:family}

The months were Latin tourists. Family is the opposite story: German's words
for father, mother, brother and sister were never borrowed --- they are
inherited, and they are blood cousins of the English ones. Along the way they
put a famous sound-law on display.
\emph{Practice lessons: \nolinkurl{lessons/GE-C10-*.md}.}

\section{\emph{der Vater}, \emph{die Mutter} --- the pair that proves Grimm's law}
\label{lesson:eltern}

\begin{sounds}
\textbf{der Vater} is said \emph{FAH-ter}: note the spelling \textbf{V}, but it
is pronounced \textbf{f}. \textbf{die Mutter} is said \emph{MOO-ter}, with a
short \emph{u} as in English \emph{put}.
\end{sounds}

Two words, with their articles:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
German & English & from \\
\midrule
\emph{der Vater} (masculine) & father & PIE \emph{*ph$_2$t\'{e}r} \\
\emph{die Mutter} (feminine) & mother & PIE \emph{*m\'{e}h$_2$t\=er} \\
\bottomrule
\end{tabular}
\end{center}

\begin{grammarlens}
\textbf{Grimm's law, on display.} Grimm's law is the great sound-shift that
split the \textbf{Germanic} languages (English, German, Dutch\ldots) off from
the rest of the family. Watch what it did to the \emph{same} ancient words that
Latin kept as \emph{pater} and \emph{m\=ater}:\\[3pt]
\emph{*ph$_2$t\'{e}r} $\to$ Latin \emph{pater} (French \emph{p\`{e}re}) vs.\
Germanic \textbf{f}ather / \textbf{V}ater\\[3pt]
\emph{*m\'{e}h$_2$t\=er} $\to$ Latin \emph{m\=ater} (French \emph{m\`{e}re})
vs.\ Germanic \textbf{m}other / \textbf{M}utter
\end{grammarlens}

\begin{etymology}
\textbf{p} $\to$ \textbf{f}: Latin kept the \emph{p} (\emph{pater}); Germanic
turned it into \emph{f} --- English \textbf{f}ather, German \textbf{V}ater
(where the \emph{V} is simply an old spelling of that \emph{f} sound).\\[3pt]
\textbf{t} $\to$ \textbf{th} $\to$ \textbf{d}: the middle consonant became
English \emph{th} (mo\textbf{th}er), and then, in German's own later shift,
\emph{d}/\emph{t} (Mu\textbf{tt}er).
\end{etymology}

\begin{culture}
So German and English \emph{agree} here --- both start with \emph{f-}, both
with \emph{m-} --- precisely because they are \textbf{both Germanic}, while
French and Latin sit on the other side of Grimm's line. The month names were
Latin tourists; \emph{Vater} and \emph{Mutter} are natives.
\end{culture}

\section[\emph{Bruder}, \emph{Schwester}, \emph{Geschwister}]{\emph{der Bruder}, \emph{die Schwester} --- and the word for ``siblings''}
\label{lesson:geschwister}

\begin{sounds}
\emph{BROO-der}; and \textbf{Schwester} is \emph{SHVES-ter} --- German
\emph{sch} spells an English \emph{sh} sound, and the \emph{w} that follows is
said \emph{v}, so \emph{Schw-} $=$ \emph{shv-}.
\end{sounds}

\begin{cousinweb}
\textbf{der Bruder} (``brother'') $\leftarrow$ PIE \emph{*b\textsuperscript{h}r\'{e}h$_2$t\=er} ---
the same root as Latin \emph{fr\=ater} (French \emph{fr\`{e}re}), but here kept
with a \emph{b-} (Grimm's law changed \emph{b\textsuperscript{h}} $\to$ \emph{b}),
exactly like English \textbf{b}rother.
\end{cousinweb}

\begin{cousinweb}
\textbf{die Schwester} (``sister'') $\leftarrow$ PIE \emph{*sw\'{e}s\=or}. The
German \emph{sch} spells an \emph{sh} sound; strip it back and
\emph{Schwester} is English \textbf{s}ister and Latin \textbf{s}oror --- all
one root.
\end{cousinweb}

\begin{morphologybox}
\textbf{The collective --- \emph{die Geschwister}.} German has a tidy word
English lacks: \textbf{die Geschwister} $=$ ``\textbf{siblings},''
brothers-and-sisters together. It is built with the \textbf{ge-} prefix, an old
\textbf{collective} marker (``a set of\ldots'') attached to \emph{Schwester} ---
the same \emph{ge-} that bundles mountains into a \textbf{das Gebirge} (``a
mountain range''), and that lets German gather a whole family into one word.
\end{morphologybox}

The four family words now stand as a Germanic set beside the Romance ones:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
German & English & Romance cousin \\
\midrule
\emph{Vater} / \emph{Mutter} & father / mother & \emph{p\`{e}re} / \emph{m\`{e}re} \\
\emph{Bruder} / \emph{Schwester} & brother / sister & \emph{fr\`{e}re} / \emph{s\oe{}ur} \\
\bottomrule
\end{tabular}
\end{center}

\begin{culture}
Why do these match English so closely? Because both languages are
\textbf{Germanic}, drawing on the same ancient roots with no Grimm's-law gap
between them. Months were \emph{borrowed} from Latin; family is
\emph{inherited} Germanic.
\end{culture}
