\chapter{The First Verbs}
\label{ch:first-verbs}

Until now you have assembled fixed phrases. This chapter hands you the machine
itself: the regular German verb, whose endings you learn once and then apply to
thousands of words. Three verbs --- \emph{wohnen}, \emph{machen}, \emph{lernen}
--- and then your first sentence built from parts nobody handed you.
\emph{Practice lessons: \texttt{lessons/GE-C05-*.md}.}

\section{\emph{wohnen} --- ``to live,'' and the regular German verb}
\label{lesson:wohnen}

\begin{sounds}
German \emph{w} is an English \emph{v}: \emph{wohnen} $=$ \emph{VOH-nen}. The
\emph{h} here merely \textbf{lengthens} the \emph{o} (\emph{woh-} $=$ a long
\emph{oh}); it is not breathy.
\end{sounds}

\begin{cousinweb}
\textbf{wohnen} (``to live, to dwell'') $\leftarrow$ Old High German
\emph{won\=en}, ``to dwell, to be accustomed, to be content'' --- dwelling and
\emph{being used to} a place were the same idea. That root survives in English,
just barely: \textbf{wont}, ``accustomed'' (\emph{as is his wont}), and
\emph{won't}'s distant cousin; the archaic \textbf{won} / \textbf{wone}, ``to
dwell.'' It is also kin to \textbf{win} and \textbf{wish} --- all from a root
about \emph{striving for} or \emph{being content}.
\end{cousinweb}

\begin{grammarlens}
\textbf{The regular (weak) present tense.} Drop the \emph{-en} to get the stem
\emph{wohn-}, then add the endings. This one table is the pattern behind
thousands of German verbs.
\end{grammarlens}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Person & Ending & \emph{wohnen} $\to$ \\
\midrule
\emph{ich} (I) & \textbf{-e} & \emph{ich wohne} \\
\emph{du} (you) & \textbf{-st} & \emph{du wohnst} \\
\emph{er} / \emph{sie} / \emph{es} (he / she / it) & \textbf{-t} & \emph{er wohnt} \\
\emph{wir} (we) & \textbf{-en} & \emph{wir wohnen} \\
\emph{ihr} (you all) & \textbf{-t} & \emph{ihr wohnt} \\
\emph{sie} / \emph{Sie} (they / you formal) & \textbf{-en} & \emph{sie wohnen} \\
\bottomrule
\end{tabular}
\end{center}

Unlike French --- where \emph{-e}, \emph{-es}, \emph{-ent} are all silent ---
\textbf{German endings are audible}: \emph{wohne} / \emph{wohnst} /
\emph{wohnt} really do sound different. So you could almost tell who is
speaking from the ending alone. German nevertheless \emph{keeps} the pronoun;
why, comes later in this chapter. Now you can answer a \emph{wo} (``where'')
question:

\begin{center}\itshape
Ich wohne in Berlin. $=$ ``I live in Berlin.''
\end{center}

\section{\emph{machen} --- ``to make / to do,'' and it is just English \emph{make}}
\label{lesson:machen}

\begin{sounds}
The \emph{ch} after \emph{a} is the raspy \emph{ach}-sound, made in the throat
like the \emph{ch} of Scottish \emph{loch}: \emph{machen} $=$ \emph{MAKH-en},
with a clean open \emph{ah}.
\end{sounds}

\begin{cousinweb}
\textbf{machen} $\leftarrow$ Germanic \emph{mak\=on}, ``to make, to shape, to
fit together'' --- the \emph{same word as English} \textbf{make}. The German
\emph{k}-sound became the \emph{ch} rasp along the way, by the \textbf{High
German consonant shift}: \emph{make} $\to$ \emph{machen}, just as \emph{ik}
$\to$ \emph{ich} and \emph{book} $\to$ \emph{Buch}. So \emph{machen} is
\emph{make}, wearing German pronunciation.
\end{cousinweb}

German uses it for both ``make'' and ``do'':

\begin{center}\itshape
Ich mache einen Kuchen. $=$ ``I'm making a cake.''\\
Was machst du? $=$ ``What are you doing?''
\end{center}

\begin{grammarlens}
\textbf{Same weak endings.} \emph{ich mache} $\cdot$ \emph{du machst} $\cdot$
\emph{er macht} (\emph{-e} / \emph{-st} / \emph{-t}); \emph{wir machen} $\cdot$
\emph{ihr macht} $\cdot$ \emph{sie machen}. Identical to \emph{wohnen} ---
nothing new to learn. That is the payoff of a regular pattern.
\end{grammarlens}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Person & \emph{machen} $\to$ \\
\midrule
\emph{ich} & \emph{mache} \\
\emph{du} & \emph{machst} \\
\emph{er} / \emph{sie} / \emph{es} & \emph{macht} \\
\emph{wir} & \emph{machen} \\
\emph{ihr} & \emph{macht} \\
\emph{sie} / \emph{Sie} & \emph{machen} \\
\bottomrule
\end{tabular}
\end{center}

\section{\emph{lernen} --- ``to learn,'' and it \emph{is} English \emph{learn}}
\label{lesson:lernen}

\begin{sounds}
The German \textbf{r} is a soft guttural at the back of the throat:
\emph{lernen} $=$ \emph{LER-nen}.
\end{sounds}

\begin{cousinweb}
\textbf{lernen} $\leftarrow$ Germanic \emph{lizn\=ojan}, ``to learn, to get to
know'' --- the \emph{same root as English} \textbf{learn}. Its close relatives
are lovely: \textbf{lore}, the body of what has been \emph{learned} (folklore,
the \emph{lore} of a craft); and \textbf{last} (a shoemaker's \emph{last})
together with Gothic \emph{lais} (``I know''), which share the deep root --- the
idea of \emph{following a track} to knowledge. So \emph{lernen} is not foreign
at all: it is \emph{learn}, and its noun \emph{die Lehre} (``the teaching, the
lesson'') is \emph{lore}.
\end{cousinweb}

\begin{grammarlens}
\textbf{The same weak endings, one last time} --- \emph{-e} / \emph{-st} /
\emph{-t} in the singular, \emph{-en} / \emph{-t} / \emph{-en} in the plural.
Three verbs, \emph{wohne}, \emph{mache}, \emph{lerne}, one pattern: you now
conjugate the regular German present.
\end{grammarlens}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Person & \emph{lernen} $\to$ \\
\midrule
\emph{ich} & \emph{lerne} \\
\emph{du} & \emph{lernst} \\
\emph{er} / \emph{sie} / \emph{es} & \emph{lernt} \\
\emph{wir} & \emph{lernen} \\
\emph{ihr} & \emph{lernt} \\
\emph{sie} / \emph{Sie} & \emph{lernen} \\
\bottomrule
\end{tabular}
\end{center}

\section{\emph{Ich lerne Deutsch} --- your first real sentence}
\label{lesson:ich-lerne-deutsch}

\begin{sounds}
\emph{Deutsch} $=$ \emph{DOYTCH}: \emph{eu} is the \emph{oy} vowel, and
\emph{tsch} is the \emph{tch} of English \emph{match}. The whole sentence:
\emph{ikh LER-neh DOYTCH}.
\end{sounds}

Everything pays off: you take a \textbf{verb} you have conjugated and a
\textbf{noun}, and build a sentence nobody handed you --- \emph{Ich lerne
Deutsch}, ``I'm learning German.'' (There is no separate ``am \ldots-ing'' in
German: \emph{ich lerne} covers both ``I learn'' and ``I am learning.'')

\begin{cousinweb}
\textbf{Deutsch} $=$ ``German'' (the language), and note it is
\textbf{capitalized}, because it is a noun here. It comes from Old High German
\textbf{diutisc}, ``of the people, of the folk'' --- from \emph{*\th{}eud\=o},
``the people.'' The Germanic tribes simply called their own tongue ``the
people's [language],'' as opposed to Latin.\\[3pt]
That ``folk'' root spread far: English \textbf{Dutch} is the \emph{same} word
(it once meant all continental West Germanic; English later narrowed it to the
Netherlands); \textbf{Teutonic} comes via Latin \emph{Teutonicus} from the same
\emph{*\th{}eud\=o}; Italian calls Germany \emph{Tedesco} and French had
\emph{thiois} --- all echoes of \emph{diutisc}.
\end{cousinweb}

\begin{morphologybox}
\textbf{Ich} (I) $+$ \textbf{lerne} (learn, from \emph{lernen}) $=$
\textbf{Ich lerne Deutsch.}
\end{morphologybox}

\begin{grammarlens}
\textbf{Why German keeps the pronoun.} German keeps \emph{ich}, \emph{du},
\emph{er}\ldots{} --- like French, and unlike Spanish or Italian, which drop
it. But for a \emph{different reason}. Spanish drops \emph{yo} because
\emph{hablo} already says ``I.'' French keeps \emph{je} because its endings are
\emph{silent}. German's endings are \emph{not} silent (\emph{wohne} /
\emph{wohnst} / \emph{wohnt} differ audibly) --- yet German still needs the
pronoun because its \textbf{grammar demands an overt subject}: a German
sentence is built around the verb sitting in second position, with something
before it. So: Spanish \textbf{drops}, French keeps for \textbf{sound}, German
keeps for \textbf{structure}.
\end{grammarlens}

Swap the pieces and the pattern keeps working: \emph{Ich wohne in Berlin.}
$\cdot$ \emph{Was machst du?}

\section{Making sentences with weak verbs}
\label{lesson:ch5-practice}

For the first time you are \emph{building} German sentences from a pattern, not
reciting phrases. Run each verb through \emph{ich} / \emph{du} / \emph{er}:

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Verb & \emph{ich} & \emph{du} & \emph{er} \\
\midrule
\emph{wohnen} (to live) & \emph{wohne} & \emph{wohnst} & \emph{wohnt} \\
\emph{machen} (to make/do) & \emph{mache} & \emph{machst} & \emph{macht} \\
\emph{lernen} (to learn) & \emph{lerne} & \emph{lernst} & \emph{lernt} \\
\bottomrule
\end{tabular}
\end{center}

The same endings every time (\textbf{-e} / \textbf{-st} / \textbf{-t}), and ---
unlike French --- you can \emph{hear} them all. You still keep the pronoun,
because German needs an overt subject.

\medskip
\noindent Now say something real:

\begin{center}\itshape
--- Wo wohnst du? --- Ich wohne in Berlin. Und du?\\
--- Ich wohne in Wien. Was machst du?\\
--- Ich lerne Deutsch! --- Sehr gut!\\
--- Danke! --- Bitte.
\end{center}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Built this chapter & \\
\midrule
the regular (weak) present & drop \emph{-en}, add
\emph{-e}/\emph{-st}/\emph{-t}/\emph{-en}/\emph{-t}/\emph{-en}; the endings are
\textbf{audible} \\
\emph{wohnen} & $\leftarrow$ \emph{won\=en} $\to$ English \emph{wont} \\
\emph{machen} & $\leftarrow$ \emph{mak\=on} $\to$ English \emph{make};
\emph{Was machst du?} \\
\emph{lernen} & $\leftarrow$ \emph{lizn\=ojan} $\to$ English \emph{learn},
\emph{lore} \\
\emph{Ich lerne Deutsch} & first self-assembled sentence (\emph{Deutsch}
$\leftarrow$ \emph{diutisc}, ``of the people'' $\to$ \emph{Dutch}) \\
the pronoun rule & Spanish \textbf{drops} \emph{yo}; French \textbf{keeps}
\emph{je} (silent endings); German \textbf{keeps} \emph{ich} (structure) \\
\bottomrule
\end{tabular}
\end{center}

What are the three singular weak endings, and are they audible? \emph{-e} /
\emph{-st} / \emph{-t}; yes --- unlike French. Give the ``I'' forms of the
three verbs: \emph{ich wohne}, \emph{ich mache}, \emph{ich lerne}. German now
moves in real sentences.
