\chapter{Greetings}
\label{ch:greetings}

The Marathi greetings --- and, taught through them, how to read the Devanagari
script and what makes Marathi its own language, not a dialect of Hindi. Each
lesson introduces the letters its word needs. \emph{Practice lessons:
\texttt{lessons/MR-C01-*.md}.}

\section{\mr{नमस्कार} \emph{(namask\=ar)} --- ``greetings''}
\label{lesson:namaskar}

\begin{sounds}
\emph{(If you already read Devanagari, skim this.)} Devanagari runs
\textbf{left to right}, and every consonant carries a built-in ``a.'' \mr{न}
na, \mr{म} ma, \mr{का} k\=a, \mr{र} ra. The stroke under \mr{स्} (a
\emph{halant}) removes its vowel, so the bare \emph{s} leans onto the next as a
conjunct: \mr{स्} $+$ \mr{का} $\to$ \mr{स्का}. Read \mr{न·म·स्·का·र} $\to$
\emph{namask\=ar}.
\end{sounds}

\begin{cousinweb}
\mr{नमस्कार} (\emph{namask\=ar}) $=$ \mr{नमः} (\emph{nama\d{h}}, ``a bow'') $+$
\mr{कार} (\emph{k\=ara}, ``the making of''), from Sanskrit --- literally ``the
making of a bow.'' It is the \emph{same} word Kannada and Telugu borrowed, and
close kin to Hindi's \emph{namaskār}. But note: where Hindi leans on
\emph{namaste}, Marathi's everyday, all-purpose greeting is
\textbf{namaskār} --- one of many small ways the two Devanagari languages differ.
\end{cousinweb}

\begin{culture}
Said with the palms pressed together, respectful and neutral --- it serves as
both hello and goodbye, morning to night, to almost anyone. Warmer and more
casual is a simple \emph{k\=ay mh\d{n}t\=a?} (``what do you say?,'' i.e.\ ``how's
it going?''), for friends.
\end{culture}

\section{\mr{धन्यवाद} \emph{(dhanyav\=ad)} --- ``thank you''}
\label{lesson:dhanyavad}

\begin{sounds}
New: \mr{ध} \emph{dha} (a breathy \emph{d}), \mr{य} \emph{ya}, \mr{वा}
\emph{v\=a}, \mr{द} \emph{da}, and the conjunct \mr{न्य} (\emph{na} $+$
\emph{ya}). Read \mr{ध·न्य·वा·द} $\to$ \emph{dhanyav\=ad}.
\end{sounds}

\begin{cousinweb}
\mr{धन्यवाद} (\emph{dhanyav\=ad}) $=$ Sanskrit \emph{dhanya} (``worthy'') $+$
\emph{v\=ada} (``a saying'') --- ``an utterance of `[you are] worthy.''' The
same Sanskrit word behind Hindi \emph{dhanyav\=ad}, Kannada \emph{dhanyav\=ada},
Telugu \emph{dhanyav\=adamulu}. Warmer, more spoken Marathi often says
\mr{आभारी आहे} (\emph{\=abh\=ar\=\i\ \=ahe}, ``[I] am grateful''), on the root
\emph{\=abh\=ar}, ``gratitude.''
\end{cousinweb}

\section{\mr{हो} \emph{(ho)} --- ``yes''}
\label{lesson:ho}

\begin{sounds}
\mr{ह} (\emph{ha}) $+$ the \emph{o}-m\=atr\=a \mr{ो} $\to$ \mr{हो} (\emph{ho}).
\end{sounds}

\begin{cousinweb}
\mr{हो} (\emph{ho}, ``yes'') is the plain affirmative --- distinct from Hindi's
\emph{h\=aṁ}. Marathi builds its own everyday words on shared Indo-Aryan roots.
The polite, fuller ``yes'' is \emph{ho, bar\d{n} \=ahe} (``yes, it's fine'').
\end{cousinweb}

\section{\mr{नाही} \emph{(n\=ah\=\i)} --- ``no''}
\label{lesson:nahi}

\begin{sounds}
\mr{ना} (\emph{n\=a}) $+$ \mr{ही} (\emph{h\=\i}) $\to$ \emph{n\=ah\=\i}.
\end{sounds}

\begin{cousinweb}
\mr{नाही} (\emph{n\=ah\=\i}, ``no / is not'') sits on the ancient negative
\emph{na-} --- the \emph{same} Proto-Indo-European \emph{*ne} behind English
\textbf{no}, \textbf{not}, \textbf{none}, Latin \emph{non}, and Hindi
\emph{nah\=\i\d{n}}. It doubles as ``there isn't'': Marathi, like Hindi, denies
with this one word.
\end{cousinweb}

\section{\mr{बरं} \emph{(bar\d{n})} --- ``okay, fine''}
\label{lesson:baram}

\begin{sounds}
\mr{ब} (\emph{ba}) $+$ \mr{रं} (\emph{ra} with the \emph{anusv\=ara} dot \mr{ं},
a light nasal) $\to$ \emph{bar\d{n}}. That dot on top nasalises the vowel.
\end{sounds}

\begin{cousinweb}
\mr{बरं} (\emph{bar\d{n}}, ``okay, fine, good, all right'') is the verbal shrug
of Marathi --- a soft ``sure, fine,'' the oil of everyday talk (its neutral form
is \emph{bar\d{e}}). It is a genuinely Marathi word, not a Sanskrit loan.
\end{cousinweb}

\section{\mr{येतो} / \mr{येते} \emph{(yeto / yete)} --- ``I'll take my leave''}
\label{lesson:yeto}

\begin{sounds}
\mr{ये} (\emph{ye}) $+$ \mr{तो} (\emph{to}) $\to$ \emph{yeto}; \mr{ये} $+$
\mr{ते} (\emph{te}) $\to$ \emph{yete}.
\end{sounds}

\begin{cousinweb}
Marathi, like the Dravidian languages to its south, will not say a bare ``I'm
leaving.'' The everyday goodbye is \mr{येतो} (\emph{yeto}) --- literally ``I
\textbf{come} [again],'' a promise of return, never a final departure. (Fuller:
\emph{yeto, namask\=ar}.)
\end{cousinweb}

\begin{grammarlens}
\textbf{Marathi verbs carry gender.} \emph{yeto} is what a \emph{man} says;
a \emph{woman} says \mr{येते} (\emph{yete}). The verb itself changes with the
speaker's gender --- something Hindi does only in the past tense, but Marathi
does in the present too. (And Marathi keeps \emph{three} genders --- masculine,
feminine, and neuter --- where Hindi kept only two.) You'll meet this again at
every verb.
\end{grammarlens}

\section{Recap}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Marathi & & Meaning \\
\midrule
\mr{नमस्कार} & \emph{namask\=ar} & hello / goodbye \\
\mr{धन्यवाद} & \emph{dhanyav\=ad} & thank you \\
\mr{हो} & \emph{ho} & yes \\
\mr{नाही} & \emph{n\=ah\=\i} & no \\
\mr{बरं} & \emph{bar\d{n}} & okay / fine \\
\mr{येतो} / \mr{येते} & \emph{yeto / yete} & ``I'll be going'' (m./f.) \\
\bottomrule
\end{tabular}
\end{center}

Marathi is written in Devanagari like Hindi, but it is its own Indo-Aryan
language: it prefers \emph{namask\=ar}, keeps \textbf{three} genders, marks
\textbf{gender on the verb} (\emph{yeto/yete}), and --- as the reference page
shows --- has an extra letter, \mr{ळ} (\d{l}a), that Hindi lacks. Next chapter:
introducing yourself --- \emph{m\=ajh\d{e} n\=av\ldots} (``my name\ldots'') ---
and Marathi's \emph{t\=u} / \emph{tumh\=\i} (familiar / respectful ``you'').
