\chapter{Introducing Yourself}
\label{ch:introductions}

Now a first real exchange --- giving your name, asking for someone else's, and
saying you're glad to meet them:

\begin{center}
\gu{મારું નામ મીરા છે}. \quad(\emph{m\=arũ n\=am Mira chhe} --- ``My name is Mira.'')\\
\gu{તમારું નામ શું છે}? \quad(\emph{tam\=arũ n\=am shũ chhe} --- ``What is your name?'')
\end{center}

\emph{Practice lessons: \texttt{lessons/GU-C02-*.md}.}

\section{\gu{નામ} \emph{(n\=am)} --- ``name,'' a word older than Europe}
\label{lesson:naam}

\begin{cousinweb}
\gu{નામ} (\emph{n\=am}) $\leftarrow$ Sanskrit \emph{n\=aman}, from PIE
*h\textsubscript{3}n\'{o}mn\d{} --- the same root as English \textbf{name}, Latin
\emph{n\=omen} ($\to$ \textbf{noun}), Greek \emph{onoma}, Hindi \emph{n\=am}.
Cousins from before either language existed.
\end{cousinweb}

\section{\gu{મારું} \emph{(m\=arũ)} --- ``my,'' and gender again}
\label{lesson:maarun}

\begin{cousinweb}
\gu{મારું} (\emph{m\=arũ}, ``my'') is on the first-person root \emph{ma-} --- the
same behind English \textbf{me}, \textbf{my}, \textbf{mine}, Latin \emph{meus},
Hindi \emph{mer\=a}.
\end{cousinweb}

\begin{grammarlens}
\textbf{``My'' agrees in three genders.} \emph{n\=am} is neuter, so ``my name'' is
\emph{m\=arũ n\=am}. The set: \emph{m\=aro} (m.) / \emph{m\=ar\=\i} (f.) /
\emph{m\=arũ} (n.). Every possessive picks its form from the noun's gender.
\end{grammarlens}

\section{\gu{છે} \emph{(chhe)} --- ``is,'' a copula all Gujarati's own}
\label{lesson:chhe}

\begin{cousinweb}
\gu{છે} (\emph{chhe}, ``is'') is Gujarati's \textbf{copula} --- and, unusually,
\emph{not} Hindi's \emph{hai} or Sanskrit's \emph{asti}. It grew from an Old
Gujarati form (\emph{achhaï}) along its own path, so it quickly marks a sentence
as Gujarati. Its forms: \emph{hũ chhũ} (I am), \emph{tũ chhe} (you are), \emph{te
chhe} (is), \emph{tame chho} (you are, resp.).
\end{cousinweb}

\begin{grammarlens}
\textbf{The verb goes last.} Gujarati is subject--object--verb: ``my name Arun
\emph{is}.'' \emph{Chhe} sits at the end.
\end{grammarlens}

\section{\gu{મારું નામ \ldots છે} --- ``my name is\ldots''}
\label{lesson:maarun-naam-chhe}

\gu{મારું} (my) $+$ \gu{નામ} (name) $+$ name $+$ \gu{છે} (is) $=$ ``my name
is\ldots'' \emph{M\=arũ n\=am Arun chhe.} The verb comes last; \emph{m\=arũ} is
neuter to agree with \emph{n\=am}.

\section{\gu{તું} / \gu{તમે} \emph{(tũ / tame)} --- ``you,'' familiar and respectful}
\label{lesson:tu-tame}

\begin{cousinweb}
\gu{તું} (\emph{tũ}, familiar ``you'') $\leftarrow$ Sanskrit \emph{tvam}, PIE
*t\=u --- the root of English archaic \textbf{thou}, Latin \emph{t\=u}, French
\emph{tu}, Hindi \emph{t\=u}. \gu{તમે} (\emph{tame}, respectful) is grammatically
plural --- courtesy-by-plural, like French \emph{vous}; it takes the plural copula
\emph{chho}.
\end{cousinweb}

\section{\gu{શું} \emph{(shũ)} --- ``what''}
\label{lesson:shun}

\begin{cousinweb}
\gu{શું} (\emph{shũ}, ``what'') descends, by a winding road, from the interrogative
stem \emph{ka-} (PIE *k\ensuremath{^w}o-) --- the root behind English \textbf{what},
Latin \emph{quis}, Hindi \emph{ky\=a}. Most Gujarati questions still show the
\emph{k-} plainly (\emph{kem} how, \emph{k\=o\d{n}} who, \emph{ky\=a\d{m}} where,
\emph{ky\=are} when); ``what'' wore down to this \emph{sh-} form --- the odd cousin
in a \emph{k-} family.
\end{cousinweb}

\section{\gu{તમારું નામ શું છે}? --- ``what's your name?''}
\label{lesson:tamarun-naam-shun-chhe}

\gu{તમારું} (\emph{tam\=arũ}, ``your,'' neuter to match \emph{n\=am}) $+$ \gu{નામ}
$+$ \gu{શું} (what) $+$ \gu{છે} (is) $=$ ``your name what is?'' --- verb last. The
familiar ``your'' is \emph{t\=arũ} (from \emph{tũ}). Note \emph{m\=arũ / t\=arũ /
tam\=arũ} --- ``my / your / your (resp.),'' all neuter before \emph{n\=am}.

\section{\gu{આનંદ} \emph{(\=anand)} --- ``joy,'' and ``a joy to meet you''}
\label{lesson:anand}

\begin{cousinweb}
The full phrase is \gu{તમને મળીને આનંદ થયો} (\emph{tamne ma\d{l}\=\i ne \=anand
thayo}) --- ``having met you, joy happened.'' \gu{આનંદ} (\emph{\=anand}, ``joy,
bliss'') is \textbf{Sanskrit} --- \emph{\=a-} (``fully'') $+$ \emph{nand} (``to
rejoice'') --- the \emph{\=ananda} of \emph{sat-cit-\=ananda} and a common Gujarati
name.
\end{cousinweb}

\section{The whole exchange}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Gujarati & English \\
\midrule
\gu{મારું નામ મીરા છે}. & My name is Mira. \\
\gu{તમારું નામ શું છે}? & What's your name? \\
\gu{મારું નામ અરુણ છે}. & My name is Arun. \\
\gu{તમને મળીને આનંદ થયો}. & Pleased to meet you. \\
\bottomrule
\end{tabular}
\end{center}

Every atom traced: \emph{n\=am} ($\to$ \emph{name}), \emph{m\=arũ} ($\to$
\emph{my}), \emph{shũ} ($\to$ \emph{what}), \emph{tũ} ($\to$ \emph{thou}). And
\emph{chhe} --- Gujarati's own ``is,'' not Hindi's \emph{hai}. Next chapter:
\emph{tame kem chho?} (``how are you?'').
