\chapter{Numbers One to Ten}
\label{ch:numbers-1-10}

Numbers are the most-reused words in any language, and French --- like its
sibling Spanish --- inherited them almost unchanged from Latin. So you already
half-know them from English, and a \emph{Spanish twin} stands beside each one.
\emph{Practice lessons: \texttt{lessons/FR-C06-*.md}.}

\section{\emph{un, deux, trois, quatre, cinq} --- counting to five}
\label{lesson:nombres-1-5}

\begin{sounds}
\emph{un} is a nasal vowel, \emph{\oe{}} said through the nose --- close to
``uh'' hummed nasally; no \emph{n} is actually pronounced.\\[3pt]
\emph{deux} $=$ \emph{deuh}, with the rounded \emph{eu}; the \emph{x} is silent.
\end{sounds}

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
French & $\leftarrow$ Latin & Spanish twin & English cousins \\
\midrule
\emph{un} (1) & \emph{\={u}nus} & \emph{uno} & \textbf{unit}, \textbf{union}, \textbf{unique}, \textbf{universe} \\
\emph{deux} (2) & \emph{duo} & \emph{dos} & \textbf{duo}, \textbf{dual}, \textbf{duet}, \textbf{double} \\
\emph{trois} (3) & \emph{tr\={e}s} & \emph{tres} & \textbf{trio}, \textbf{triple}, \textbf{trident}, \textbf{triangle} \\
\emph{quatre} (4) & \emph{quattuor} & \emph{cuatro} & \textbf{quart}, \textbf{quarter}, \textbf{quatrain}, \textbf{quadrant} \\
\emph{cinq} (5) & \emph{qu\={\i}nque} & \emph{cinco} & \textbf{quintet}, \textbf{quintessence} \\
\bottomrule
\end{tabular}
\end{center}

\begin{cousinweb}
\textbf{un} also lives on in English \emph{a}/\emph{an} and \emph{one}; French
\emph{cinq} was borrowed straight into English as \textbf{cinque}, the 5 on
dice.\\[3pt]
Latin \emph{\={u}nus, duo, tr\={e}s} barely changed on the way into French and
Spanish alike --- \emph{trois} and \emph{tres} are the same Roman
\emph{tr\={e}s}, just worn two different ways.
\end{cousinweb}

\begin{grammarlens}
\textbf{\emph{un} $\to$ \emph{un} / \emph{une} (one $=$ a/an).} \emph{un}
doubles as the article ``a/an,'' and it takes gender: \emph{un} (masculine) /
\emph{une} (feminine) --- \emph{un caf\'{e}} ``a/one coffee,'' \emph{une
bi\`{e}re} ``a/one beer.'' French makes no difference between ``one'' and ``a,''
exactly like Spanish \emph{un}/\emph{una}. The others (\emph{deux}, \emph{trois}
\ldots) don't change.
\end{grammarlens}

\section{\emph{six, sept, huit, neuf, dix} --- and the calendar's secret}
\label{lesson:nombres-6-10}

The rest of the way to ten. These four hide the same fact the English and
Spanish numbers do: \emph{septembre}--\emph{d\'{e}cembre} are just Latin for 7,
8, 9, 10. And \emph{huit} shows how violently French can wear a Latin word down.

\begin{sounds}
\emph{six} and \emph{dix} end in an \emph{-s} sound when counting (\emph{sees},
\emph{dees}) but go silent before a consonant (\emph{six livres} $=$ \emph{see
livres}) --- a liaison quirk.\\[3pt]
\emph{neuf} $=$ \emph{nuhf}; its \emph{f} turns to a \emph{v} sound before a
vowel (\emph{neuf ans} $=$ \emph{neu-v-ans}).
\end{sounds}

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
French & $\leftarrow$ Latin & Spanish twin & English cousins \\
\midrule
\emph{six} (6) & \emph{sex} & \emph{seis} & \textbf{sextet}, \textbf{sextant}, \textbf{semester} \\
\emph{sept} (7) & \emph{septem} & \emph{siete} & \textbf{septembre} (the 7th month!), \textbf{septet} \\
\emph{huit} (8) & \emph{oct\={o}} & \emph{ocho} & \textbf{octobre} (8th), \textbf{octave}, \textbf{octopus} \\
\emph{neuf} (9) & \emph{novem} & \emph{nueve} & \textbf{novembre} (9th), \textbf{novena} \\
\emph{dix} (10) & \emph{decem} & \emph{diez} & \textbf{d\'{e}cembre} (10th), \textbf{decimal}, \textbf{dime} \\
\bottomrule
\end{tabular}
\end{center}

\begin{cousinweb}
Look at \textbf{huit}: Latin \emph{oct\={o}} lost its \emph{c}, softened, and
grew an \emph{h-} prop --- \emph{oct\={o}} $\to$ \emph{oit} $\to$ \emph{huit}.
The number is unrecognizable, but \emph{octobre} keeps the old \emph{oct-}
intact, so the 8 is still visible in the \emph{month}.\\[3pt]
And \emph{dix} gave English \textbf{dime} (a tenth of a dollar), through Old
French \emph{disme} $\leftarrow$ \emph{decima} ``a tenth.''
\end{cousinweb}

\begin{grammarlens}
\textbf{Why the months are ``off by two.''} \emph{Septembre} means ``7th,'' yet
it is the 9th month. Why? The old Roman year began in March, so
\emph{September} really was the seventh. Julius and Augustus later inserted
\emph{juillet} and \emph{ao\^{u}t} (named for themselves), shoving the counting
months out of place --- but the Latin numbers stuck. Every \emph{d\'{e}cembre},
you are saying Roman ``ten.'' (Identical to the Spanish story ---
\emph{diciembre} $\leftarrow$ \emph{decem} too.)
\end{grammarlens}
