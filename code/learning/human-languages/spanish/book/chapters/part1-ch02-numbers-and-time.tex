\section{Numbers 11--100}

\begin{vocabtable}
once--quince & --- & 11--15 \\
diecis\'{e}is--diecinueve & --- & 16--19 \\
veinte, veintiuno\ldots & --- & 20, 21\ldots \\
treinta--noventa & --- & 30--90 \\
cien / ciento & --- & 100 \\
\end{vocabtable}

\begin{etymology}
11--15 $\leftarrow$ Latin \emph{undecim, duodecim, tredecim, quattuordecim,
quindecim} --- literally ``one-ten,'' ``two-ten,'' etc. 16--19 fuse
\emph{diez y seis} (``ten and six'') into one word; 21--29 do the same with
\emph{veinte}. \emph{veinte} $\leftarrow$ \emph{viginti}; \emph{treinta}
$\leftarrow$ \emph{triginta}; the 40s--90s all end in the same \emph{-ginta}
(``tens''). \emph{cien/ciento} $\leftarrow$ \emph{centum} (cognate with
English \emph{cent, century, percent, centipede}).
\end{etymology}

\begin{grammarlens}
Spanish visibly fuses ``ten-and-six'' into \emph{diecis\'{e}is}. English
did the identical thing, just longer ago and less visibly: ``sixteen'' is
\emph{six} + \emph{teen}, and \emph{teen} is a worn-down form of ``ten.''
Separately: \emph{cien} stands alone or before a noun (\emph{cien
d\'{o}lares}); \emph{ciento} is used when another number follows
(\emph{ciento diez}) --- a distinction English ``hundred'' doesn't make.
\end{grammarlens}

\section{Telling Time}

\begin{vocabtable}
la hora & fem. & hour \\
es la una & --- & it's one o'clock \\
son las dos & --- & it's two o'clock \\
y media / y cuarto / menos cuarto & --- & half past / quarter past / quarter to \\
\end{vocabtable}

\begin{grammarlens}
Spanish switches between singular and plural ``to be'' depending on the
hour: \emph{es la una} (singular, ``the one hour'') but \emph{son las
dos} (plural, ``the two hours''). English ``it is two o'clock'' always
uses singular ``is,'' regardless of the hour --- the same \emph{ser}-agrees-
with-its-subject logic from Chapter~1, applied somewhere unexpected.
\end{grammarlens}

\section{Months \& Seasons}

\begin{vocabtable}
enero, marzo, octubre & masc. & January, March, October \\
la primavera & fem. & spring \\
el verano & masc. & summer \\
el oto\~{n}o & masc. & autumn \\
el invierno & masc. & winter \\
\end{vocabtable}

\begin{etymology}
Months follow the same Roman-calendar logic as the days of the week
(Chapter~1): \emph{enero} $\leftarrow$ \emph{Ianuarius} (Janus); \emph{marzo}
$\leftarrow$ \emph{Martius}, same root as \emph{martes}; \emph{octubre}
$\leftarrow$ \emph{October}, ``the eighth month,'' same \emph{octo} root as
the number \emph{ocho}. \emph{oto\~{n}o} $\leftarrow$ Latin \emph{autumnus},
borrowed into English unchanged as \emph{autumn}. \emph{invierno}
$\leftarrow$ \emph{hibernus} (English \emph{hibernate}).
\end{etymology}

\begin{grammarlens}
Spanish months and seasons are not capitalized (\emph{enero}, not
\emph{Enero}) --- unlike English, which always capitalizes month names.
Same rule as the days of the week.
\end{grammarlens}

\section{Question Words}

\begin{vocabtable}
qu\'{e}, d\'{o}nde, c\'{o}mo & --- & what, where, how \\
cu\'{a}ndo, por qu\'{e} & --- & when, why \\
qui\'{e}n, cu\'{a}nto & --- & who, how much/many \\
\end{vocabtable}

\begin{grammarlens}
Every one of these carries a written accent when used as a question word
--- \emph{qu\'{e}} vs.\ plain \emph{que} (``that''), \emph{c\'{o}mo} vs.\
\emph{como} (``like/as''). English has no equivalent marker: ``that'' and
``what'' look the same whether they're part of a question or not; only
context disambiguates. Spanish put a visible flag on it.
\end{grammarlens}

\section{Survival Phrases}

\begin{quote}
\itshape
No entiendo. \textquestiondown Puede repetir, m\'{a}s despacio por favor?
Perd\'{o}n --- \textquestiondown d\'{o}nde est\'{a} la estaci\'{o}n?
\end{quote}

\begin{etymology}
\emph{entender} $\leftarrow$ Latin \emph{intendere} (English \emph{intend,
intent}). \emph{despacio} $\leftarrow$ \emph{de spatio} (``with room'' ---
English \emph{space}). \emph{perd\'{o}n} $\leftarrow$ \emph{perdonare}
(\emph{per-} + \emph{donare}, ``to give'' --- English \emph{pardon,
donate}). \emph{disculpe} $\leftarrow$ \emph{dis-} + \emph{culpa}
(``blame'' --- English \emph{culpable, culprit}), literally ``to remove
blame.''
\end{etymology}
