\chapter{Greetings}
\label{ch:greetings}

The Hindi greetings --- and, taught through them, how to read Devanagari and
Hindi's double inheritance (Sanskrit and Perso-Arabic). Each lesson introduces
the letters its word needs, so you learn to read the word and understand it at
once. \emph{Practice lessons: \texttt{lessons/HI-C01-*.md}.}

\section{\hi{नमस्ते} \emph{(namaste)} --- ``I bow to you''}
\label{lesson:namaste}

\begin{sounds}
Hindi reads \textbf{left to right}, and every consonant carries a built-in
``a.'' \hi{न} na, \hi{म} ma, \hi{स} sa, \hi{त} ta. A vowel sign changes the
vowel: \hi{े} makes ``e'' (\hi{ते} = te). A \emph{halant} (the stroke under
\hi{स्}) \emph{removes} the vowel, and the bare consonant leans onto the next
as a \textbf{conjunct}: \hi{स्} + \hi{ते} $\to$ \hi{स्ते}. Read left to right:
\hi{न·म·स्·ते} $\to$ \emph{namaste}.
\end{sounds}

\begin{cousinweb}
\hi{नमस्ते} = \hi{नमः} (\emph{nama\d{h}}, ``a bow, reverence,'' from Sanskrit
root \textbf{\hi{नम्}} \emph{nam}, ``to bow'') + \hi{ते} (\emph{te}, ``to
you'') = literally ``\textbf{I bow to you}.'' The root \emph{nam} also builds
\emph{namask\=ar} (next), and \emph{namaste} is borrowed whole into English.
\end{cousinweb}

\begin{culture}
A real gesture, not a bland ``hi'': said with palms pressed together, in
respect. Works as both hello and goodbye, any time, polite without being
stiff.
\end{culture}

\section{\hi{नमस्कार} \emph{(namask\=ar)} --- the more formal bow}
\label{lesson:namaskar}

\begin{sounds}
New: \hi{क} ka, \hi{र} ra, and the sign \hi{ा} for long ``\=a'' (\hi{का} =
k\=a). You already read \hi{नमस्} (\emph{namas}). \hi{न·म·स्·का·र} $\to$
\emph{namask\=ar}.
\end{sounds}

\begin{cousinweb}
\hi{नमस्कार} = \hi{नमः} (\emph{nama\d{h}}, ``bow,'' root \emph{nam}) + \hi{कार}
(\emph{k\=ara}, ``the making/doing of,'' from Sanskrit \textbf{\hi{कृ}}
\emph{k\d{r}}, ``to do''). Literally ``the making of a bow.'' That piece
\emph{-k\=ara} (``the act of \_\_\_'') recurs across Sanskrit vocabulary.
\end{cousinweb}

A notch more \textbf{formal} than \emph{namaste} --- ceremonial, or for elders
and writing.

\section{\hi{धन्यवाद} \emph{(dhanyav\=ad)} --- ``thank you,'' the Sanskrit way}
\label{lesson:dhanyavad}

\begin{sounds}
New: \hi{ध} dha, \hi{य} ya, \hi{व} va, \hi{द} da, and a new conjunct \hi{न्} +
\hi{य} $\to$ \hi{न्य} (``nya''). \hi{ध·न्·य·वा·द} $\to$ \emph{dhanyav\=ad}.
\end{sounds}

\begin{cousinweb}
\hi{धन्यवाद} = \hi{धन्य} (\emph{dhanya}, ``worthy, blessed,'' from \hi{धन}
\emph{dhana}, ``wealth'') + \hi{वाद} (\emph{v\=ada}, ``a saying,'' from
Sanskrit \textbf{\hi{वद्}} \emph{vad}, ``to speak''). Literally ``an utterance
of [you are] worthy'' --- a formal, literary ``thanks.''
\end{cousinweb}

\begin{culture}
The \textbf{formal, Sanskritic} ``thank you'' --- written, official,
respectful. For everyday thanks Hindi often reaches for a lighter Persian
word, \hi{शुक्रिया} --- next --- and holding the two together is the best
window into Hindi.
\end{culture}

\section{\hi{शुक्रिया} \emph{(shukriy\=a)} --- Hindi's other heritage}
\label{lesson:shukriya}

\begin{sounds}
New: \hi{श} sha, and the sign \hi{ि} for ``i'' (written \emph{before} the
consonant, read \emph{after}). With a conjunct \hi{क्} + \hi{र} $\to$ \hi{क्र}
(``kra''): \hi{शु·क्·रि·या} $\to$ \emph{shukriy\=a}.
\end{sounds}

\begin{cousinweb}
\hi{शुक्रिया} is \emph{not} Sanskrit. It came through Persian from \textbf{Arabic
\emph{shukr}} (``gratitude''), root \textbf{sh-k-r} --- the \emph{same word} as
Arabic \emph{shukran}. Two unrelated languages, one Semitic root for thanks.
\end{cousinweb}

\begin{culture}
Centuries of Persian as the court language poured Persian and Arabic words into
everyday Hindi/Urdu, so Hindi carries \textbf{two vocabularies}: Sanskrit
(\emph{dhanyav\=ad}, \emph{namaste}) and Perso-Arabic (\emph{shukriy\=a},
\emph{alvid\=a}). Often the Sanskrit word is formal and the Persian one
everyday. \emph{shukriy\=a} is the casual, everyday ``thanks.''
\end{culture}

\section{\hi{अलविदा} \emph{(alvid\=a)} --- ``the farewell''}
\label{lesson:alvida}

\begin{sounds}
New: the independent vowel \hi{अ} (``a,'' for a word that \emph{starts} with a
vowel) and \hi{ल} la. \hi{अ·ल·वि·दा} $\to$ \emph{alvid\=a}.
\end{sounds}

\begin{cousinweb}
\hi{अलविदा} came through Persian from \textbf{Arabic} \emph{al-wid\=a`}
(``the farewell''), root \textbf{w-d-`}. At the front is \hi{अल} --- the Arabic
article \textbf{\emph{al-}}, ``the,'' the same one riding inside English
\emph{algebra} and \emph{alcohol}. So \emph{alvid\=a} is literally ``the
farewell.''
\end{cousinweb}

\begin{culture}
Carries \textbf{finality} --- the goodbye for a long or final parting. For
casual ``bye,'' Hindi uses \emph{namaste} again, or \emph{phir milenge}
(``we'll meet again''). Next chapter: \emph{\=ap} vs.\ \emph{tum} (formal and
informal ``you''), and introducing yourself.
\end{culture}
