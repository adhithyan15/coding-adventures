\chapter{Greetings}
\label{ch:greetings}

The Italian greetings, built from their pieces --- a word for ``good,'' the
day-parts, the article that carries gender, and the most famous ``hello'' in
the world. Each word is taken apart to its root and its English cousins.
\emph{Practice lessons: \texttt{lessons/IT-C01-*.md}.}

\section{\emph{ciao} --- ``hi,'' and the word that means ``I am your slave''}
\label{lesson:ciao}

\begin{sounds}
\emph{CHOW} --- \emph{ci} before a vowel is the \emph{ch} of \emph{church}, and
\emph{-ao} is one gliding syllable, like the English \emph{ciao} you may already
have heard.
\end{sounds}

\begin{cousinweb}
\textbf{ciao} began as the Venetian greeting \emph{s-ci\`{a}o (vostro)} --- ``[I
am your] slave / servant,'' a humble ``at your service.'' That \emph{s-ci\`{a}o}
is Venetian for \emph{schiavo} (``slave'') $\leftarrow$ Medieval Latin
\emph{sclavus}. The very same word is English \textbf{slave} --- and, astonishingly,
\textbf{Slav}: medieval slaves were so often Slavic captives that the ethnic name
\emph{became} the word for ``slave'' across Europe. So every casual \emph{ciao}
is a fossil of ``I am your servant,'' carrying the buried history of a whole
people.
\end{cousinweb}

\begin{culture}
\emph{ciao} is \textbf{informal}, and works both coming and going (``hi'' and
``bye'') --- for friends, family, peers. To someone you'd show respect, or on a
first meeting, you reach instead for \emph{buongiorno}, built over the next
lessons. (Doubled, \emph{ciao ciao}, it's a breezy ``bye-bye.'')
\end{culture}

\section{\emph{buono} / \emph{buon} --- ``good''}
\label{lesson:buono}

\begin{sounds}
\textbf{buono} $=$ \emph{BWOH-no}; the \emph{uo} is a glide. Before a masculine
noun it clips to \textbf{buon} (\emph{bwon}) --- \emph{buon giorno}.
\end{sounds}

\begin{cousinweb}
\textbf{buono} (``good'') $\leftarrow$ Latin \emph{bonus} $\to$ English
\textbf{bon}us, \textbf{bon}anza, \textbf{bon}bon, and (through French)
\textbf{boun}ty. Spanish, another daughter of Latin, made the same \emph{bonus}
into \emph{bueno}; Italian kept it closer, \emph{buono}.
\end{cousinweb}

\begin{grammarlens}
\emph{buono} \textbf{agrees} with its noun in gender and number ---
\emph{buono} (m.), \emph{buona} (f.), \emph{buoni} / \emph{buone} (pl.). This
gender-agreement of adjectives is Latin's legacy across the Romance family.
Before a masculine noun \emph{buono} drops its \emph{-o} to \emph{buon}: you'll
see it in \emph{buon giorno}.
\end{grammarlens}

\section{\emph{il} / \emph{la} / \emph{lo} --- ``the,'' and the gender it carries}
\label{lesson:il-la-lo}

\begin{cousinweb}
\textbf{il} and \textbf{lo} (masculine ``the'') and \textbf{la} (feminine)
$\leftarrow$ Latin \emph{ille} / \emph{illa} (``that one''), the pointing word
worn down to ``the.'' The same \emph{ille/illa} gave Spanish \emph{el/la} and
French \emph{le/la} --- and also Italian \textbf{lui}/\textbf{lei} (``he/she'').
(English ``the'' is Germanic, unrelated.)
\end{cousinweb}

\begin{grammarlens}
\textbf{Every Italian noun is masculine or feminine} --- Latin's three genders
(the neuter folded mostly into the masculine) collapsed to two. Learn each noun
\emph{with} its article. Italian has \emph{two} masculine ``the''s:
\textbf{lo} before tricky clusters (\emph{lo studente}, \emph{lo zio}) and
\textbf{il} everywhere else (\emph{il giorno}); feminine is always \emph{la}.
Don't drill the rule --- just meet \emph{il} at the first noun, next.
\end{grammarlens}

\section{\emph{giorno} --- ``day,'' and the road to English \emph{journey}}
\label{lesson:giorno}

\begin{sounds}
\textbf{gi} $=$ the \emph{j} of \emph{jam}; \emph{gior-no} $=$ \emph{JOR-no}.
\end{sounds}

\begin{cousinweb}
\textbf{il giorno} (masc.) $\leftarrow$ Latin \emph{diurnum} (``daytime''), from
\emph{dies} (``day''). That detour through \emph{diurnum} is why \emph{giorno}
looks nothing like Latin \emph{dies}: cousins are \textbf{jour}nal, ad\textbf{journ},
so\textbf{journ}, and \textbf{journ}ey (a \emph{day's} travel) --- all from the
same \emph{diurnum} --- plus \textbf{di}urnal and \textbf{di}ary straight from
\emph{dies}. Italian \emph{giorno} and French \emph{jour} are near-twins here;
Spanish, taking \emph{dies} directly, got the very different \emph{d\'{i}a}.
\end{cousinweb}

\begin{grammarlens}
\emph{giorno} is \textbf{masculine} (\emph{il giorno}). Plural \emph{i giorni}
--- Italian marks the plural by changing the final vowel (\emph{-o} $\to$
\emph{-i}), not by adding \emph{-s}.
\end{grammarlens}

\section{\emph{buongiorno} --- ``good day,'' assembled}
\label{lesson:buongiorno}

\emph{buon} (``good'') $+$ \emph{giorno} (``day'') $=$ \textbf{buongiorno},
``good day,'' one word --- the polite, all-purpose daytime hello.

\begin{culture}
Where the casual \emph{ciao} says ``I'm your servant,'' \emph{buongiorno} simply
wishes you a good day --- and it is the \textbf{safe, respectful} choice with
anyone: shopkeepers, strangers, elders. Used through the day until late
afternoon, when it gives way to \emph{buonasera}.
\end{culture}

\section{\emph{sera} --- ``evening,'' and \emph{buonasera}}
\label{lesson:sera}

\begin{cousinweb}
\textbf{la sera} (fem.) $\leftarrow$ Latin \emph{sera} (``late [hour]''), from
\emph{serus}, ``late.'' English cousin: the rare \emph{serotine} (a bat of the
\emph{late} evening). \textbf{buonasera} $=$ \emph{buona} $+$ \emph{sera},
``good evening'' --- \emph{buona} in its feminine form, because \emph{sera} is
feminine.
\end{cousinweb}

\section[notte --- night, and the -ct- to -tt- rule]{\emph{notte} --- ``night,'' and the \emph{-ct-} $\to$ \emph{-tt-} rule}
\label{lesson:notte}

\begin{cousinweb}
\textbf{la notte} (fem.) $\leftarrow$ Latin \emph{noctem} (\emph{noct-}) $\to$
\textbf{noct}urnal, equi\textbf{nox}. The change is a \emph{rule}: Latin
\emph{-ct-} became Italian \textbf{-tt-} (\emph{no\underline{tt}e}), where
Spanish softened it to \emph{-ch-} (\emph{noche}) and French to \emph{-it-}
(\emph{nuit}).
\end{cousinweb}

\begin{center}
\begin{tabular}{@{}lllll@{}}
\toprule
Latin & Italian & Spanish & French & English \\
\midrule
\emph{noctem} & no\emph{tt}e & no\emph{ch}e & nu\emph{it} & night \\
\emph{lactem} & la\emph{tt}e & le\emph{ch}e & la\emph{it} & (lact-) \\
\emph{factum} & fa\emph{tt}o & he\emph{ch}o & fa\emph{it} & fact \\
\bottomrule
\end{tabular}
\end{center}

\emph{buonanotte} $=$ \emph{buona} $+$ \emph{notte}, ``good night'' --- the
farewell for bed (not an evening hello; that's \emph{buonasera}).

\section{\emph{grazie} --- ``thank you,'' from \emph{grace}}
\label{lesson:grazie}

\begin{sounds}
\emph{GRAH-tsyeh} --- \textbf{z} in Italian is \emph{ts}; the final \emph{-e} is
sounded (two syllables in \emph{-zie}).
\end{sounds}

\begin{cousinweb}
\textbf{grazie} (``thanks'') $\leftarrow$ Latin \emph{grati\ae}, plural of
\emph{gratia} (``favour, thanks, grace''). It is the \emph{same} \emph{gratia}
behind English \textbf{grace}, \textbf{grat}itude, \textbf{grat}eful,
\textbf{grat}is (``free''), con\textbf{grat}ulate, and in\textbf{grat}iate. So
\emph{grazie} literally offers someone \emph{grace}. (Reply: \emph{prego},
``you're welcome'' --- literally ``I pray [it's nothing].'')
\end{cousinweb}

\section{The day of greetings}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Italian & English \\
\midrule
\emph{ciao} & hi / bye (informal) \\
\emph{buongiorno} & good day \\
\emph{buonasera} & good evening \\
\emph{buonanotte} & good night \\
\emph{grazie} & thank you \\
\bottomrule
\end{tabular}
\end{center}

Every piece from a Latin root: \emph{ciao} (\emph{sclavus}, ``slave''),
\emph{buono} (\emph{bonus}), \emph{giorno} (\emph{diurnum}), \emph{notte}
(\emph{noctem}, via the \emph{-ct-}$\to$\emph{-tt-} rule), \emph{grazie}
(\emph{gratia}, ``grace''). Next chapter: introducing yourself --- \emph{mi
chiamo\ldots} (``my name is\ldots'') --- and Italian's \emph{tu} vs.\ \emph{Lei}
(familiar / formal ``you'').
