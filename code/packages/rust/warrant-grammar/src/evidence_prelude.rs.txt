/// Dynamic value of the Warrant pure profile (VER04e section 2.1 subset).
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum V {
    Bool(bool),
//@if v0
    Nat(u64),
    U64(u64),
//@endif
//@if v1
    /// `Int` and `Nat` (the numeric tower below the fixed-width types).
    Int(i128),
    /// A fixed-width integer: its type and canonical value.
    Fixed(F, i128),
    /// `Option<T>` (`None` carries no type).
    Opt(Option<Box<V>>),
//@endif
    Bytes(Vec<u8>),
    Ok(Box<V>),
    Err(String),
    Name(String),
}

//@if v0
impl V {
    pub fn truthy(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn length(&self) -> V {
        match self {
            V::Bytes(b) => V::Nat(b.len() as u64),
            _ => panic!("len of a non-bytes value"),
        }
    }
    pub fn err_of(v: V) -> V {
        match v {
            V::Name(n) => V::Err(n),
            _ => panic!("Err needs an error constructor"),
        }
    }
    fn num(&self) -> u64 {
        match self {
            V::Nat(n) | V::U64(n) => *n,
            _ => panic!("numeric operand expected"),
        }
    }
    pub fn compare(l: &V, r: &V, op: &str) -> V {
        let (a, b) = (l.num(), r.num());
        V::Bool(match op {
            "<" => a < b,
            ">" => a > b,
            "<=" => a <= b,
            _ => a >= b,
        })
    }
    pub fn arith(l: &V, r: &V, op: char) -> V {
        let (a, b) = (l.num(), r.num());
        let n = match op {
            '+' => a.checked_add(b),
            '-' => a.checked_sub(b),
            _ => a.checked_mul(b),
        };
        V::Nat(n.expect("model arithmetic overflow"))
    }
    fn bytes(&self) -> &[u8] {
        match self {
            V::Bytes(b) => b,
            _ => panic!("bytes expected"),
        }
    }
    fn u64v(&self) -> u64 {
        self.num()
    }
    fn boolv(&self) -> bool {
        matches!(self, V::Bool(true))
    }
}

//@if v1
/// The eight fixed-width integer types of Warrant v1 (VER04j section 2.2).
///
/// A value is stored as the mathematical integer it denotes (an `i128`, so
/// `U64` and `I64` both fit) and is always canonical, i.e. inside the range of
/// its type.  Bitwise functions work on the width-`w` two's-complement
/// *pattern*, held in a `u128` and masked to `w` bits.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum F {
    U8,
    U16,
    U32,
    U64,
    I8,
    I16,
    I32,
    I64,
}

impl F {
    pub fn bits(self) -> u32 {
        match self {
            F::U8 | F::I8 => 8,
            F::U16 | F::I16 => 16,
            F::U32 | F::I32 => 32,
            F::U64 | F::I64 => 64,
        }
    }
    pub fn signed(self) -> bool {
        matches!(self, F::I8 | F::I16 | F::I32 | F::I64)
    }
    pub fn min(self) -> i128 {
        if self.signed() {
            -(1i128 << (self.bits() - 1))
        } else {
            0
        }
    }
    pub fn max(self) -> i128 {
        if self.signed() {
            (1i128 << (self.bits() - 1)) - 1
        } else {
            (1i128 << self.bits()) - 1
        }
    }
    pub fn mask(self) -> u128 {
        (1u128 << self.bits()) - 1
    }
    /// The two's-complement pattern of `v` masked to `w` bits (any `v`).
    pub fn pattern(self, v: i128) -> u128 {
        (v as u128) & self.mask()
    }
    /// Read a pattern back as a canonical value of the type.
    pub fn canon(self, p: u128) -> i128 {
        let p = p & self.mask();
        if self.signed() && p >> (self.bits() - 1) == 1 {
            (p as i128) - (1i128 << self.bits())
        } else {
            p as i128
        }
    }
    fn contains(self, v: i128) -> bool {
        self.min() <= v && v <= self.max()
    }
}

/// Edge values of a fixed-width type (VER04j section 2.6), in a fixed order.
/// Unsigned width `w`: `0..=255`, `1<<i` for `8 <= i < w`, `MAX`, then
/// `MAX ^ (1<<i)` for `0 <= i < w`.  Signed: the same patterns read signed,
/// plus `-1 ..= -255`, deduplicated, ascending.
pub fn fx_edge(f: F) -> Vec<i128> {
    let w = f.bits();
    let umax = f.mask();
    let mut u: Vec<u128> = (0..=255u128).filter(|v| *v <= umax).collect();
    u.extend((8..w).map(|i| 1u128 << i));
    u.push(umax);
    u.extend((0..w).map(|i| umax ^ (1u128 << i)));
    if !f.signed() {
        return u.into_iter().map(|p| p as i128).collect();
    }
    let mut s: Vec<i128> = u.into_iter().map(|p| f.canon(p)).collect();
    s.extend((1..=255i128).map(|k| -k).filter(|v| f.contains(*v)));
    s.sort();
    s.dedup();
    s
}

/// Every value of a narrow type (`U8 I8 U16 I16`), ascending.
pub fn fx_full(f: F) -> Vec<i128> {
    (f.min()..=f.max()).collect()
}

impl V {
    pub fn truthy(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn length(&self) -> V {
        match self {
            V::Bytes(b) => V::Int(b.len() as i128),
            _ => panic!("len of a non-bytes value"),
        }
    }
    pub fn err_of(v: V) -> V {
        match v {
            V::Name(n) => V::Err(n),
            _ => panic!("Err needs an error constructor"),
        }
    }
    /// The mathematical value of an `Int`/`Nat` or fixed-width operand.
    pub fn int(&self) -> i128 {
        match self {
            V::Int(n) | V::Fixed(_, n) => *n,
            _ => panic!("numeric operand expected"),
        }
    }
    pub fn fx(&self) -> (F, i128) {
        match self {
            V::Fixed(f, n) => (*f, *n),
            _ => panic!("fixed-width operand expected"),
        }
    }
    /// A shift count or bit index: a mathematical natural number.
    pub fn count(&self) -> u128 {
        u128::try_from(self.int()).expect("count must be a natural number")
    }
    fn bytes(&self) -> &[u8] {
        match self {
            V::Bytes(b) => b,
            _ => panic!("bytes expected"),
        }
    }
    fn boolv(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn compare(l: &V, r: &V, op: &str) -> V {
        let (a, b) = (l.int(), r.int());
        V::Bool(match op {
            "<" => a < b,
            ">" => a > b,
            "<=" => a <= b,
            _ => a >= b,
        })
    }
    /// `+ - *`: wrapping on equal fixed-width operands (the pattern product
    /// keeps `U64 * U64` exact), checked `i128` on `Int`/`Nat`.
    pub fn arith(l: &V, r: &V, op: char) -> V {
        if let (V::Fixed(f, a), V::Fixed(g, b)) = (l, r) {
            assert_eq!(f, g, "mixed fixed-width operands");
            let (pa, pb) = (f.pattern(*a), f.pattern(*b));
            let p = match op {
                '+' => pa.wrapping_add(pb),
                '-' => pa.wrapping_sub(pb),
                _ => pa.wrapping_mul(pb),
            };
            return V::Fixed(*f, f.canon(p));
        }
        let (a, b) = (l.int(), r.int());
        let n = match op {
            '+' => a.checked_add(b),
            '-' => a.checked_sub(b),
            _ => a.checked_mul(b),
        };
        V::Int(n.expect("model arithmetic overflow"))
    }
    /// Unary minus: wrapping on a signed fixed-width value, exact on `Int`.
    pub fn neg(v: &V) -> V {
        match v {
            V::Fixed(f, a) => V::Fixed(*f, f.canon(f.pattern(*a).wrapping_neg())),
            _ => V::Int(v.int().checked_neg().expect("model arithmetic overflow")),
        }
    }
    fn opt(v: Option<i128>, f: F) -> V {
        V::Opt(v.map(|x| Box::new(V::Fixed(f, x))))
    }
    pub fn widen(t: F, x: &V) -> V {
        let v = x.int();
        assert!(t.contains(v), "widen operand outside the target range");
        V::Fixed(t, v)
    }
    pub fn wrap(t: F, x: &V) -> V {
        V::Fixed(t, t.canon(t.pattern(x.int())))
    }
    pub fn try_(t: F, x: &V) -> V {
        let v = x.int();
        V::Opt(t.contains(v).then(|| Box::new(V::Fixed(t, v))))
    }
    pub fn to_int(x: &V) -> V {
        V::Int(x.fx().1)
    }
    pub fn to_nat(x: &V) -> V {
        let (f, v) = x.fx();
        assert!(!f.signed(), "to_nat of a signed value");
        V::Int(v)
    }
    pub fn max_value(t: F) -> V {
        V::Fixed(t, t.max())
    }
    pub fn min_value(t: F) -> V {
        V::Fixed(t, t.min())
    }
    pub fn checked(op: char, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let exact = match op {
            '+' => x.checked_add(y),
            '-' => x.checked_sub(y),
            _ => x.checked_mul(y),
        };
        V::opt(exact.filter(|v| f.contains(*v)), f)
    }
    /// `div`/`rem`: unsigned operands, or `Int`/`Nat` with floor semantics.
    pub fn divrem(div: bool, a: &V, b: &V) -> V {
        if let V::Fixed(f, x) = a {
            let (_, y) = b.fx();
            assert!(!f.signed(), "div/rem of signed operands");
            return V::opt((y != 0).then(|| if div { x / y } else { x % y }), *f);
        }
        let (x, y) = (a.int(), b.int());
        if y == 0 || (x == i128::MIN && y == -1) {
            return V::Opt(None);
        }
        let r = if div {
            let q = x / y;
            if x % y != 0 && ((x < 0) != (y < 0)) {
                q - 1
            } else {
                q
            }
        } else if y == -1 {
            0
        } else {
            let m = x % y;
            if m != 0 && ((m < 0) != (y < 0)) {
                m + y
            } else {
                m
            }
        };
        V::Opt(Some(Box::new(V::Int(r))))
    }
    /// `div_trunc`/`rem_trunc` on signed operands: `None` for `b == 0` and for
    /// `min / -1`.
    pub fn divrem_trunc(div: bool, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let ok = y != 0 && !(y == -1 && x == f.min());
        V::opt(ok.then(|| if div { x / y } else { x % y }), f)
    }
    pub fn unwrap_or(o: &V, d: &V) -> V {
        match o {
            V::Opt(Some(v)) => (**v).clone(),
            V::Opt(None) => d.clone(),
            _ => panic!("Option operand expected"),
        }
    }
    /// `bit_and` (`'&'`), `bit_or` (`'|'`), `bit_xor` (`'^'`).
    pub fn bitop(op: char, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let (px, py) = (f.pattern(x), f.pattern(y));
        V::Fixed(
            f,
            f.canon(match op {
                '&' => px & py,
                '|' => px | py,
                _ => px ^ py,
            }),
        )
    }
    pub fn bit_not(a: &V) -> V {
        let (f, x) = a.fx();
        V::Fixed(f, f.canon(!f.pattern(x)))
    }
    /// `shl shr sar rotl rotr`.  A count `>= w` is handled *before* any host
    /// shift (VER04j section 3.5), so the host never sees an over-wide shift.
    pub fn shift(op: &str, a: &V, n: &V) -> V {
        let (f, x) = a.fx();
        let (n, w, p) = (n.count(), u128::from(f.bits()), f.pattern(x));
        let v = match op {
            "shl" => {
                if n >= w {
                    0
                } else {
                    f.canon(p << n)
                }
            }
            "shr" => {
                if n >= w {
                    0
                } else {
                    f.canon(p >> n)
                }
            }
            "sar" => {
                assert!(f.signed(), "sar of an unsigned value");
                if n >= w {
                    if x < 0 {
                        -1
                    } else {
                        0
                    }
                } else {
                    x >> n
                }
            }
            _ => {
                let k = if op == "rotl" { n % w } else { (w - n % w) % w };
                if k == 0 {
                    x
                } else {
                    f.canon((p << k) | (p >> (w - k)))
                }
            }
        };
        V::Fixed(f, v)
    }
    /// `popcount clz ctz` of the width-`w` pattern.
    pub fn bitcount(op: &str, a: &V) -> V {
        let (f, x) = a.fx();
        let p = f.pattern(x);
        let w = f.bits();
        V::Int(match op {
            "popcount" => i128::from(p.count_ones()),
            "clz" => {
                if p == 0 {
                    i128::from(w)
                } else {
                    i128::from(p.leading_zeros() - (128 - w))
                }
            }
            _ => {
                if p == 0 {
                    i128::from(w)
                } else {
                    i128::from(p.trailing_zeros())
                }
            }
        })
    }
    pub fn bit(a: &V, i: &V) -> V {
        let (f, x) = a.fx();
        let i = i.count();
        V::Bool(i < u128::from(f.bits()) && (f.pattern(x) >> i) & 1 == 1)
    }
    /// `be_bytes` / `le_bytes`.
    pub fn endian_bytes(big: bool, a: &V) -> V {
        let (f, x) = a.fx();
        let p = f.pattern(x);
        let n = (f.bits() / 8) as usize;
        let mut v: Vec<u8> = (0..n).rev().map(|i| (p >> (8 * i)) as u8).collect();
        if !big {
            v.reverse();
        }
        V::Bytes(v)
    }
    /// `from_be<T>` / `from_le<T>`.
    pub fn from_endian(big: bool, t: F, b: &V) -> V {
        let mut bytes = b.bytes().to_vec();
        assert_eq!(bytes.len() as u32 * 8, t.bits(), "wrong byte width");
        if !big {
            bytes.reverse();
        }
        let p = bytes.iter().fold(0u128, |acc, x| (acc << 8) | u128::from(*x));
        V::Fixed(t, t.canon(p))
    }
}

fn mask_last_byte(bytes: &[u8]) -> Vec<u8> {
    let mut v = bytes.to_vec();
    if let Some(last) = v.last_mut() {
        *last = 0;
    }
    v
}

/// Run `body`, converting a panic into `None` (used for declared panic errors).
fn contained<T>(body: impl FnOnce() -> T) -> Option<T> {
    catch_unwind(AssertUnwindSafe(body)).ok()
}

//@if scoped
thread_local! {
    /// The one operation a negative control is currently attacking, if any.
    static ATTACK_OP: std::cell::RefCell<Option<&'static str>> = const { std::cell::RefCell::new(None) };
}

/// The fault `op`'s adapter should obey: the requested one unless a negative
/// control is attacking a different operation.
fn fault_for(op: &str, fault: Fault) -> Fault {
    ATTACK_OP.with(|a| match *a.borrow() {
        Some(target) if target != op => Fault::None,
        _ => fault,
    })
}

/// Run `body` with only `op` obeying injected faults.
fn attacking<T>(op: &'static str, body: impl FnOnce() -> T) -> T {
    struct Reset;
    impl Drop for Reset {
        fn drop(&mut self) {
            ATTACK_OP.with(|a| *a.borrow_mut() = None);
        }
    }
    ATTACK_OP.with(|a| *a.borrow_mut() = Some(op));
    let _reset = Reset;
    body()
}

/// Binder types the generator understands.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Ty {
    Bool,
    U8,
    U64,
    Nat,
    Bytes,
    BytesN(usize),
//@if v1
    /// A fixed-width binder at its default domain: every value of a narrow
    /// type (`U8 I8 U16 I16`), the edge set of a wide one.
    Fx(F),
    /// A narrow binder demoted to its edge set by the budget rule (section 2.6).
    FxEdge(F),
    /// A `Nat` that flows directly into a shift count: `0..=64` plus the
    /// boundary counts `65 127 128 129 255 256 65535 65536` (section 3.6).
    NatCount,
//@endif
}

const ALPHABET: [u8; 4] = [0x00, 0x01, 0x80, 0xff];

fn alpha_strings(len: usize) -> Vec<Vec<u8>> {
    let total = 4usize.pow(len as u32);
    (0..total)
        .map(|mut n| {
            let mut s = vec![0u8; len];
            for slot in s.iter_mut() {
                *slot = ALPHABET[n % 4];
                n /= 4;
            }
            s
        })
        .collect()
}

/// The finite domain of one binder (VER04h section 3).
pub fn domain(ty: Ty) -> Vec<V> {
    match ty {
        Ty::Bool => vec![V::Bool(false), V::Bool(true)],
//@if v0
        Ty::U8 => (0..=255u64).map(V::Nat).collect(),
        Ty::Nat => (0..=64u64).map(V::Nat).collect(),
        Ty::U64 => {
            let mut v: Vec<u64> = (0..=255u64).collect();
            v.extend((8..64).map(|i| 1u64 << i));
            v.push(u64::MAX);
            v.extend((0..64).map(|i| u64::MAX ^ (1u64 << i)));
            v.into_iter().map(V::U64).collect()
        }
//@endif
//@if v1
        Ty::U8 | Ty::U64 => unreachable!("a v1 document binds U8/U64 as Ty::Fx"),
        Ty::Nat => (0..=64i128).map(V::Int).collect(),
        Ty::NatCount => (0..=64i128)
            .chain([65, 127, 128, 129, 255, 256, 65535, 65536])
            .map(V::Int)
            .collect(),
        Ty::Fx(f) if f.bits() <= 16 => fx_full(f).into_iter().map(|v| V::Fixed(f, v)).collect(),
        Ty::Fx(f) | Ty::FxEdge(f) => fx_edge(f).into_iter().map(|v| V::Fixed(f, v)).collect(),
//@endif
        Ty::Bytes => {
            let mut v: Vec<Vec<u8>> = vec![Vec::new()];
            v.extend((0..=255u8).map(|b| vec![b]));
            v.extend(alpha_strings(2));
            v.extend(alpha_strings(3));
            for len in 4..=8 {
                v.push(vec![0x00; len]);
                v.push(vec![0xa5; len]);
            }
            v.into_iter().map(V::Bytes).collect()
        }
        Ty::BytesN(0) => vec![V::Bytes(Vec::new())],
        Ty::BytesN(1) => (0..=255u8).map(|b| V::Bytes(vec![b])).collect(),
        Ty::BytesN(n) if n <= 3 => alpha_strings(n).into_iter().map(V::Bytes).collect(),
        Ty::BytesN(n) => {
            let mut v: Vec<Vec<u8>> = ALPHABET.iter().map(|&s| vec![s; n]).collect();
            for pos in 0..n {
                for &s in &ALPHABET[1..] {
                    let mut x = vec![0u8; n];
                    x[pos] = s;
                    v.push(x);
                }
            }
            v.into_iter().map(V::Bytes).collect()
        }
    }
}

/// Closed-form size of [`domain`], `None` on `u64` overflow or an over-cap
/// width.  Must agree with the generator's `Ty::domain_size`; the generator
/// test suite checks `domain_size(t) == domain(t).len()` for every type.
pub fn domain_size(ty: Ty) -> Option<u64> {
    match ty {
        Ty::Bool => Some(2),
//@if v0
        Ty::U8 => Some(256),
        Ty::Nat => Some(65),
        Ty::U64 => Some(256 + 56 + 1 + 64),
//@endif
//@if v1
        Ty::U8 | Ty::U64 => None,
        Ty::Nat => Some(65),
        Ty::NatCount => Some(65 + 8),
        Ty::Fx(f) if f.bits() <= 16 => Some(1u64 << f.bits()),
        Ty::Fx(f) | Ty::FxEdge(f) => Some(fx_edge(f).len() as u64),
//@endif
        Ty::Bytes => Some(1 + 256 + 16 + 64 + 10),
        Ty::BytesN(n) if n as u64 > MAX_FIXED_WIDTH => None,
        Ty::BytesN(0) => Some(1),
        Ty::BytesN(1) => Some(256),
        Ty::BytesN(n) if n <= 3 => 4u64.checked_pow(n as u32),
        Ty::BytesN(n) => 3u64.checked_mul(n as u64)?.checked_add(4),
    }
}

/// Upper bound on the memory [`domain`] materialises, in bytes.
pub fn domain_bytes(ty: Ty) -> Option<u64> {
    let payload = match ty {
        Ty::Bytes => 8,
        Ty::BytesN(n) => n as u64,
        _ => 0,
    };
    domain_size(ty)?.checked_mul(payload.checked_add(VALUE_OVERHEAD)?)
}

/// Whether the Cartesian product of the binder domains fits both budgets.
/// Every step is checked: a wrapped product (eight `U8` binders is `2^64`,
/// which is `0` in `u64`) must count as over budget, never as zero.
pub fn within_budget(tys: &[Ty]) -> bool {
    let mut product = 1u64;
    let mut bytes = 0u64;
    for t in tys {
        let (Some(size), Some(b)) = (domain_size(*t), domain_bytes(*t)) else {
            return false;
        };
        let (Some(p), Some(total)) = (product.checked_mul(size), bytes.checked_add(b)) else {
            return false;
        };
        product = p;
        bytes = total;
    }
    product <= EXHAUSTIVE_BUDGET && bytes <= DOMAIN_BYTES_BUDGET
}

        .collect::<Vec<_>>()
        .join(" x ")
}

//@if v1
/// The manifest `domain` text of a claim (VER04j section 2.6): per binder its
/// type, whether the domain is `full` (every value), `edge` (the biased
/// boundary set), `count` (the shift-count domain) or `set` (a fixed finite
/// set), and its size, for example `a: U16 full(65536); b: U16 edge(281)`.
fn describe(tys: &[Ty], names: &[&str]) -> String {
    tys.iter()
        .zip(names)
        .map(|(t, n)| {
            let size = domain_size(*t).map_or_else(|| "over-budget".to_string(), |n| n.to_string());
            let (ty, kind) = match t {
                Ty::Fx(f) if f.bits() <= 16 => (format!("{f:?}"), "full"),
                Ty::Fx(f) | Ty::FxEdge(f) => (format!("{f:?}"), "edge"),
                Ty::NatCount => ("Nat".to_string(), "count"),
                Ty::Nat => ("Nat".to_string(), "set"),
                other => (format!("{other:?}"), "set"),
            };
            format!("{n}: {ty} {kind}({size})")
        })
        .collect::<Vec<_>>()
        .join("; ")
}
//@endif

/// Per-test counters that end up in the manifest.
#[derive(Clone, Debug, Default)]
pub struct Stats {
    pub cases: u64,
    pub discarded: u64,
    pub iterations: u64,
    pub admitted: u64,
    /// Property cases whose first two binders were equal (the equal branch).
    pub equal: u64,
}

fn counterexample_message(what: &str, env: &[V]) -> String {
    let repr = format!("{env:?}");
    if std::env::var("CI").is_ok() {
        format!("{what}; repro-fnv: {:016x}", fnv1a64(repr.as_bytes()))
    } else {
        format!("{what}; counterexample: {repr}")
    }
}

type Admit = fn(&[V]) -> bool;
type Holds = fn(&[V], Fault) -> bool;

/// Bounded-exhaustive enumeration of the Cartesian product of binder domains.
pub fn exhaustive(tys: &[Ty], admit: Admit, holds: Holds, fault: Fault) -> (Verdict, Stats) {
    let mut stats = Stats::default();
    // Check the budget BEFORE materialising any domain: a hostile width would
    // otherwise allocate gigabytes first and be refused afterwards.
    if !within_budget(tys) {
        return (
            Verdict::fail("corpus-malformed", "exhaustive domain exceeds the budget"),
            stats,
        );
    }
    let domains: Vec<Vec<V>> = tys.iter().map(|t| domain(*t)).collect();
    let mut idx = vec![0usize; domains.len()];
    loop {
        let env: Vec<V> = idx.iter().zip(&domains).map(|(i, d)| d[*i].clone()).collect();
        if admit(&env) {
            stats.cases += 1;
            if !holds(&env, fault) {
                return (
                    Verdict::fail(
                        "property-counterexample",
                        counterexample_message("exhaustive suite falsified the claim", &env),
                    ),
                    stats,
                );
            }
        } else {
            stats.discarded += 1;
        }
        let mut pos = idx.len();
        loop {
            if pos == 0 {
                if stats.cases == 0 {
                    return (
                        Verdict::fail("property-exhausted", "refinement admitted no case"),
                        stats,
                    );
                }
                return (Verdict::Pass, stats);
            }
            pos -= 1;
            idx[pos] += 1;
            if idx[pos] < domains[pos].len() {
                break;
            }
            idx[pos] = 0;
        }
    }
}

// ----- relation-biased generation (shared by property and fuzz) -----------

trait Src {
    fn byte(&mut self) -> u8;
    fn bounded(&mut self, n: u64) -> u64;
    fn word(&mut self) -> u64;
}

struct RngSrc<'a>(&'a mut SplitMix64);
impl Src for RngSrc<'_> {
    fn byte(&mut self) -> u8 {
        self.0.bounded(256) as u8
    }
    fn bounded(&mut self, n: u64) -> u64 {
        self.0.bounded(n)
    }
    fn word(&mut self) -> u64 {
        self.0.next_u64()
    }
}

/// Fuzz input as a randomness source; exhausted input reads as zeros.
struct StreamSrc<'a> {
    data: &'a [u8],
    pos: usize,
}
impl Src for StreamSrc<'_> {
    fn byte(&mut self) -> u8 {
        let b = self.data.get(self.pos).copied().unwrap_or(0);
        self.pos += 1;
        b
    }
    fn bounded(&mut self, n: u64) -> u64 {
        u64::from(self.byte()) % n
    }
    fn word(&mut self) -> u64 {
        let mut w = 0u64;
        for i in 0..8 {
            w |= u64::from(self.byte()) << (8 * i);
        }
        w
    }
}

//@if v1
/// Draw a fixed-width value (VER04j section 2.6).  `U8` is uniform and `U64`
/// keeps the v0 generator (zero, MAX, powers of two, small values, uniform);
/// every other type draws 3/4 uniformly over its full range and 1/4 from its
/// edge set, so boundaries are reached without exhaustive enumeration.
fn fresh_fx(f: F, src: &mut dyn Src) -> V {
    let v = match f {
        F::U8 => i128::from(src.byte()),
        F::U64 => i128::from(match src.bounded(8) {
            0 => 0,
            1 => u64::MAX,
            2 => 1u64 << src.bounded(64),
            3..=5 => src.bounded(300),
            _ => src.word(),
        }),
        _ => {
            if src.bounded(4) == 0 {
                let edge = fx_edge(f);
                edge[(src.word() % edge.len() as u64) as usize]
            } else {
                f.canon(u128::from(src.word()))
            }
        }
    };
    V::Fixed(f, v)
}

//@endif
fn fresh(ty: Ty, src: &mut dyn Src) -> V {
    match ty {
        Ty::Bool => V::Bool(src.bounded(2) == 1),
//@if v0
        Ty::U8 => V::Nat(u64::from(src.byte())),
        Ty::Nat => V::Nat(src.bounded(65)),
        Ty::U64 => V::U64(match src.bounded(8) {
            0 => 0,
            1 => u64::MAX,
            2 => 1u64 << src.bounded(64),
//@if u8
            // Documents that mix U8 and U64 binders (an exponent next to a
            // byte) need small U64 values often enough for refinements such as
            // `e < 255` to be admitted.
            3..=5 => src.bounded(300),
//@endif
            _ => src.word(),
        }),
//@endif
//@if v1
        Ty::U8 | Ty::U64 => unreachable!("a v1 document binds U8/U64 as Ty::Fx"),
        Ty::Nat => V::Int(i128::from(src.bounded(65))),
        // Half of all draws come from 0..=130 so that n = w and n = w + 1 are
        // reached for every width including 64 (VER04j section 3.6).
        Ty::NatCount => V::Int(i128::from(if src.bounded(2) == 0 {
            src.bounded(131)
        } else {
            src.bounded(65)
        })),
        Ty::Fx(f) | Ty::FxEdge(f) => fresh_fx(f, src),
//@endif
        Ty::Bytes => {
            let len = src.bounded(65) as usize;
            V::Bytes((0..len).map(|_| src.byte()).collect())
        }
        Ty::BytesN(n) => V::Bytes((0..n).map(|_| src.byte()).collect()),
    }
}

/// Derive a second binder from the first (mode 0..8, see VER04h section 4).
fn relate(prev: &V, ty: Ty, mode: u8, src: &mut dyn Src) -> V {
    match (prev, mode) {
        (_, 0) | (_, 1) => fresh(ty, src),
        (p, 2) | (p, 3) => p.clone(),
        (V::Bytes(b), 4) | (V::Bytes(b), 5) if !b.is_empty() => {
            let mut o = b.clone();
            let bit = src.bounded(8 * o.len() as u64) as usize;
            o[bit / 8] ^= 1 << (bit % 8);
            V::Bytes(o)
        }
        (V::Bytes(b), 6) if !b.is_empty() => {
            let mut o = b.clone();
            let at = src.bounded(o.len() as u64) as usize;
            o[at] = o[at].wrapping_add(1 + src.bounded(255) as u8);
            V::Bytes(o)
        }
        (V::Bytes(b), 7) if !matches!(ty, Ty::BytesN(_)) => {
            let mut o = b.clone();
            if o.is_empty() || src.bounded(2) == 0 {
                o.push(src.byte());
            } else {
                o.pop();
            }
            V::Bytes(o)
        }
//@if v0
//@if u8
        // A small U64 pairs with another small U64, so refinements that relate
        // two exponents (`e < 255 and f < 255 and e != f`) are admitted.
        (V::U64(n), 6..=7) if *n < 1000 => V::U64(src.bounded(300)),
//@endif
        (V::U64(n), 4) | (V::U64(n), 5) => V::U64(n ^ (1u64 << src.bounded(64))),
        (V::U64(n), 6) => V::U64(n | (1u64 << src.bounded(64))),
//@if u8
        // A U8 binder stays inside 0..=255: flip one bit, or step by one mod 256.
        (V::Nat(n), 4..=5) if ty == Ty::U8 => V::Nat(n ^ (1u64 << src.bounded(8))),
        (V::Nat(n), 6..=7) if ty == Ty::U8 => V::Nat((n + 1) % 256),
//@endif
        (V::Nat(n), 4..=7) => V::Nat(n.wrapping_add(1)),
//@endif
//@if v1
        // A small U64 pairs with another small U64 (refinements such as
        // `e < 255 and f < 255 and e != f` must be admitted); every other
        // fixed-width value flips one bit (modes 4-5) or steps by one with
        // wrap-around (modes 6-7), so related pairs straddle every boundary.
        (V::Fixed(F::U64, n), 6..=7) if *n < 1000 => V::Fixed(F::U64, i128::from(src.bounded(300))),
        (V::Fixed(f, n), 4..=5) => V::Fixed(*f, f.canon(f.pattern(*n) ^ (1u128 << src.bounded(u64::from(f.bits()))))),
        (V::Fixed(f, n), 6..=7) => V::Fixed(*f, f.canon(f.pattern(*n).wrapping_add(1))),
        (V::Int(n), 4..=7) => V::Int(n + 1),
//@endif
        (p, _) => p.clone(),
    }
}

fn build(tys: &[Ty], src: &mut dyn Src, mode: u8) -> Vec<V> {
    let mut env: Vec<V> = Vec::with_capacity(tys.len());
    for (i, ty) in tys.iter().enumerate() {
        let partner = (0..i).find(|j| tys[*j] == *ty);
        let v = match partner {
            Some(j) => relate(&env[j].clone(), *ty, mode, src),
            None => fresh(*ty, src),
        };
        env.push(v);
    }
    env
}

fn decode(input: &[u8], tys: &[Ty]) -> Vec<V> {
    let mode = input.first().copied().unwrap_or(0) % 8;
    let mut src = StreamSrc { data: input, pos: 1 };
    build(tys, &mut src, mode)
}

fn shrink_vals(cand: Vec<V>, prop_raw: &dyn Fn(&Vec<V>) -> bool) -> (Vec<V>, usize) {
    // A shrink candidate may leave the operation's domain (a `Bytes<N>` binder
    // made shorter panics inside the adapter).  Such a candidate is not a
    // counterexample, so a panic counts as "the claim holds".
    let prop = |t: &Vec<V>| catch_unwind(AssertUnwindSafe(|| prop_raw(t))).unwrap_or(true);
    let mut cur = cand;
    let mut steps = 0;
    for i in 0..cur.len() {
        let base = cur.clone();
        match &base[i] {
            V::Bytes(bytes) => {
                let (s, n) = shrink_bytes(
                    bytes,
                    |b| {
                        let mut t = base.clone();
                        t[i] = V::Bytes(b.to_vec());
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::Bytes(s);
                steps += n;
            }
//@if v0
            V::U64(v) => {
                let (s, n) = shrink_int(
                    *v as i128,
                    0,
                    0,
                    u64::MAX as i128,
                    |x| {
                        let mut t = base.clone();
                        t[i] = V::U64(x as u64);
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::U64(s as u64);
                steps += n;
            }
//@endif
//@if v1
            // Unsigned values shrink toward 0 (bisection, then single steps);
            // a signed value first tries its negation, then shrinks toward 0
            // from its own side.
            V::Fixed(f, v) => {
                let f = *f;
                let start = if *v < 0 && f.contains(-*v) {
                    let mut t = base.clone();
                    t[i] = V::Fixed(f, -*v);
                    if prop(&t) { *v } else { -*v }
                } else {
                    *v
                };
                let (s, n) = shrink_int(
                    start,
                    0,
                    f.min(),
                    f.max(),
                    |x| {
                        let mut t = base.clone();
                        t[i] = V::Fixed(f, x);
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::Fixed(f, s);
                steps += n;
            }
//@endif
            _ => {}
        }
    }
    (cur, steps)
}

/// L2 property suite over relation-biased generators.
pub fn property(tag: &str, tys: &[Ty], admit: Admit, holds: Holds, fault: Fault) -> (Verdict, Stats) {
    let seed = SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64();
    let options = PbtOptions {
        num_tests: PROPERTY_CASES as usize,
        max_retries: 100,
        max_shrink_steps: 1000,
        is_ci: std::env::var("CI").is_ok(),
    };
    let cases = Cell::new(0u64);
    let discarded = Cell::new(0u64);
    let equal = Cell::new(0u64);
    let gen = |rng: &mut SplitMix64| -> Result<Vec<V>, String> {
        for _ in 0..100 {
            let mode = rng.bounded(8) as u8;
            let env = build(tys, &mut RngSrc(rng), mode);
            if admit(&env) {
                cases.set(cases.get() + 1);
                if env.len() >= 2 && env[0] == env[1] {
                    equal.set(equal.get() + 1);
                }
                return Ok(env);
            }
            discarded.set(discarded.get() + 1);
        }
        Err(FID_EXHAUSTED.to_string())
    };
    let verdict = check_property(
        seed,
        &options,
        gen,
        move |env: &Vec<V>| holds(env, fault),
        shrink_vals,
    );
    let stats = Stats {
        cases: cases.get(),
        discarded: discarded.get(),
        equal: equal.get(),
        ..Stats::default()
    };
    (verdict, stats)
}

fn cover(cov: &mut CoverageBitmap, env: &[V]) {
    for (i, v) in env.iter().enumerate() {
        match v {
            V::Bytes(b) => cov.record_edge(300 + b.len().min(64)),
//@if v0
            V::U64(n) => cov.record_edge(500 + n.count_ones() as usize),
//@endif
            V::Bool(x) => cov.record_edge(600 + usize::from(*x)),
//@if v0
//@if u8
            V::Nat(n) => cov.record_edge(2000 + (*n as usize).min(512)),
//@endif
//@endif
//@if v1
            V::Fixed(f, n) => cov.record_edge(500 + f.pattern(*n).count_ones() as usize),
            V::Int(n) => cov.record_edge(2000 + (*n).clamp(0, 512) as usize),
//@endif
            _ => {}
        }
        for w in &env[i + 1..] {
            match (v, w) {
                (V::Bytes(a), V::Bytes(b)) => {
                    cov.record_edge(if a == b { 1 } else { 2 });
                    cov.record_edge(if a.len() == b.len() { 3 } else { 4 + (a.len() ^ b.len()) % 64 });
                    let pre = a.iter().zip(b.iter()).take_while(|(x, y)| x == y).count();
                    cov.record_edge(1000 + pre.min(64));
                }
//@if v0
                (V::U64(a), V::U64(b)) => {
                    cov.record_edge(if a == b { 700 } else { 701 + (a ^ b).count_ones() as usize });
                }
//@if u8
                (V::Nat(a), V::Nat(b)) => {
                    cov.record_edge(if a == b { 3000 } else { 3001 + (a ^ b).count_ones() as usize });
                }
//@endif
//@endif
//@if v1
                (V::Fixed(f, a), V::Fixed(_, b)) => {
                    cov.record_edge(if a == b { 700 } else { 701 + (f.pattern(*a) ^ f.pattern(*b)).count_ones() as usize });
                }
//@endif
                _ => {}
            }
        }
    }
}

/// Coverage-guided fuzz campaign; the claim body is the oracle.
pub fn fuzz_claim(
    tag: &str,
    tys: &[Ty],
    admit: Admit,
    holds: Holds,
    fault: Fault,
    iterations: u64,
) -> (Verdict, Stats) {
    let opts = FuzzOptions {
        max_iterations: iterations as usize,
        seed: SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64(),
        max_input_len: 256,
        base_energy: 10,
    };
    let corpus: Vec<Vec<u8>> = (0..8u8)
        .map(|m| {
            let mut s = vec![m, 4];
            s.extend((0..40u8).map(|i| i.wrapping_mul(37).wrapping_add(m)));
            s
        })
        .collect();
    let admitted = Cell::new(0u64);
    let report = fuzz::fuzz_target(&corpus, &opts, |input, cov| {
        let env = decode(input, tys);
        cover(cov, &env);
        if !admit(&env) {
            cov.record_edge(4000);
            return FuzzOutcome::Pass;
        }
        admitted.set(admitted.get() + 1);
        match catch_unwind(AssertUnwindSafe(|| holds(&env, fault))) {
            Ok(true) => FuzzOutcome::Pass,
            Ok(false) => FuzzOutcome::Crash {
                kind: "claim-violated".into(),
                frames: vec![tag.to_string()],
            },
            Err(_) => FuzzOutcome::Crash {
                kind: "panic".into(),
                frames: vec![tag.to_string()],
            },
        }
    });
    let stats = Stats {
        iterations: report.iterations as u64,
        admitted: admitted.get(),
        ..Stats::default()
    };
    if let Some(c) = report.crashes.first() {
        return (
            Verdict::fail(FID_FUZZ_CRASH, format!("fuzz campaign found {} ({} hits)", c.kind, c.hits)),
            stats,
        );
    }
    if stats.admitted == 0 {
        return (Verdict::fail("corpus-mismatch", "fuzz campaign admitted no input"), stats);
    }
    (Verdict::Pass, stats)
}

// ----- manifest recording ---------------------------------------------------

#[derive(Clone, Debug)]
struct Meta {
    id: String,
    kind: &'static str,
    op: String,
    claim: String,
    fault: String,
    target: String,
    domain: String,
    seed: String,
//@if v1
    /// Binders the budget rule demoted from `full` to `edge` (section 2.6).
    demoted: Vec<String>,
//@endif
}

static META: Mutex<Vec<Meta>> = Mutex::new(Vec::new());
static STATS: Mutex<BTreeMap<String, (Stats, String)>> = Mutex::new(BTreeMap::new());
/// SHA-256 of the crate source the target was built against (`run_main`'s
/// argument), recorded so a checker can detect a manifest older than the source.
static SOURCE_SHA256: Mutex<String> = Mutex::new(String::new());

fn reg(
    suite: &mut Suite,
    meta: Meta,
    body: impl Fn() -> (Verdict, Stats, String) + 'static,
) {
    // WARRANT_EVIDENCE_FILTER=<substring> keeps only matching test ids; it
    // exists so BUILD can run a sabotage mode against a cheap subset.  A
    // filtered run never writes a manifest (see `write_manifest`).
    if let Ok(filter) = std::env::var("WARRANT_EVIDENCE_FILTER") {
        if !meta.id.contains(&filter) {
            return;
        }
    }
    let id = meta.id.clone();
    META.lock().unwrap().push(meta);
    suite.add(id.clone(), move |_ctx| {
        let (verdict, stats, note) = body();
        STATS.lock().unwrap().insert(id.clone(), (stats, note));
        verdict
    });
}

fn is_detected(v: &Verdict) -> bool {
    match v {
        Verdict::Fail { failure_id, .. } => {
            coding_adventures_test_bootstrap::verdict::is_registered_failure_id(failure_id)
        }
        Verdict::Pass => false,
    }
}

fn seed_of(tag: &str) -> u64 {
    SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64()
}

fn write_manifest(outcomes: &[Outcome]) {
    let Some(path) = manifest::output_path("WARRANT_EVIDENCE_OUT") else {
        return;
    };
    if std::env::var("WARRANT_EVIDENCE_FAULT").is_ok()
        || std::env::var("WARRANT_EVIDENCE_FILTER").is_ok()
    {
        eprintln!("refusing to write an evidence manifest from a sabotaged or filtered run");
        return;
    }
    let metas = META.lock().unwrap().clone();
    let stats = STATS.lock().unwrap().clone();
    let mut tests: Vec<Json> = Vec::new();
    for m in metas {
        let verdict = outcomes
            .iter()
            .find(|o| o.id == m.id)
            .map(|o| if o.passed() { "pass" } else { "fail" })
            .unwrap_or("not-run");
        let (s, note) = stats.get(&m.id).cloned().unwrap_or_default();
        let mut e: BTreeMap<String, Json> = BTreeMap::new();
        e.insert("id".into(), Json::Str(m.id));
        e.insert("kind".into(), Json::str(m.kind));
        e.insert("op".into(), Json::Str(m.op));
        e.insert("claim".into(), Json::Str(m.claim));
        e.insert("verdict".into(), Json::str(verdict));
        match m.kind {
            "example" => {
                e.insert("cases".into(), Json::Num(1));
            }
            "exhaustive" => {
                e.insert("cases".into(), Json::Num(s.cases));
                e.insert("discarded".into(), Json::Num(s.discarded));
                e.insert("domain".into(), Json::Str(m.domain));
//@if v1
                e.insert(
                    "demoted".into(),
                    Json::Arr(m.demoted.into_iter().map(Json::Str).collect()),
                );
//@endif
            }
            "property" => {
                e.insert("cases".into(), Json::Num(s.cases));
                e.insert("discarded".into(), Json::Num(s.discarded));
                e.insert("equal_cases".into(), Json::Num(s.equal));
                e.insert("seed".into(), Json::Str(m.seed));
            }
            "fuzz" => {
                e.insert("iterations".into(), Json::Num(s.iterations));
                e.insert("admitted".into(), Json::Num(s.admitted));
                e.insert("seed".into(), Json::Str(m.seed));
            }
            _ => {
                e.insert("fault".into(), Json::Str(m.fault));
                e.insert("target".into(), Json::Str(m.target));
                e.insert("caught_by".into(), Json::Str(note));
            }
        }
        tests.push(Json::Obj(e));
    }
    let doc = Json::obj([
        ("schema", Json::str("warrant-test-evidence-v1")),
//@if v1
        ("warrant_version", Json::str("v1")),
//@endif
        ("module", Json::str(MODULE)),
        ("spec", Json::str(SPEC_PATH)),
        ("spec_sha256", Json::str(SPEC_SHA256)),
        ("source_sha256", Json::Str(SOURCE_SHA256.lock().unwrap().clone())),
        ("evidence_seed", Json::Str(EVIDENCE_SEED.to_string())),
        ("tests", Json::Arr(tests)),
    ]);
    if let Err(e) = manifest::write(&path, &doc) {
        eprintln!("cannot write evidence manifest: {e}");
    }
}

//@if v1
/// Write the manifest of an in-process run (`Suite::run_recorded`) to
/// `WARRANT_EVIDENCE_OUT`; the generated test target does this through
/// `run_main`, a test of the generator does it through this wrapper.
pub fn write_manifest_for(outcomes: &[Outcome]) {
    write_manifest(outcomes);
}

//@endif
/// Decide the injected fault from the two sabotage variables.
///
/// `WARRANT_EVIDENCE_FILTER` exists only so BUILD can aim a *sabotage* run at
/// a cheap subset.  Without a real fault it would turn the target into a suite
/// that silently runs fewer tests and still exits 0, so a filter is an error
/// unless the fault is a real (non-`none`) one (audit A3; `FAULT=none` plus a
/// filter was the residual hole).  Pure function so it can be tested without
/// touching the process environment.
pub fn resolve_sabotage(fault: Option<&str>, filter: Option<&str>) -> Result<Fault, String> {
    let resolved = match fault {
        None => Fault::None,
        Some(text) => Fault::parse(text).ok_or_else(|| format!("unknown WARRANT_EVIDENCE_FAULT {text:?}"))?,
    };
    match (resolved, filter) {
        (Fault::None, Some(f)) => Err(format!(
            "WARRANT_EVIDENCE_FILTER={f:?} requires WARRANT_EVIDENCE_FAULT to name a real fault \
             (not unset or \"none\"); a filtered run is only meaningful as a sabotage run"
        )),
        (fault, _) => Ok(fault),
    }
}

/// A filtered suite that selects nothing proves nothing and must not pass.
pub fn require_tests(count: usize, filter: Option<&str>) -> Result<(), String> {
    match filter {
        Some(f) if count == 0 => Err(format!("WARRANT_EVIDENCE_FILTER={f:?} matched no test")),
        _ => Ok(()),
    }
}

/// Entry point of the generated test target. Returns the process exit code.
///
/// `src_dir` is the crate's `src/` directory (the hand-written shim passes
/// `CARGO_MANIFEST_DIR/src`); the digest of **every** `src/**/*.rs` file, framed
/// by `manifest::frame_sources`, goes into the manifest as `source_sha256`, the
/// same definition the proof and bench manifests and the checker use.
pub fn run_main(src_dir: &std::path::Path) -> i32 {
    match manifest::source_tree_framing(src_dir) {
        Ok(framed) => *SOURCE_SHA256.lock().unwrap() = sha256_hex(&framed),
        Err(e) => {
            eprintln!("cannot bind evidence to {}: {e}", src_dir.display());
            return 2;
        }
    }
    std::panic::set_hook(Box::new(|_| ()));
    let fault_var = std::env::var("WARRANT_EVIDENCE_FAULT").ok();
    let filter_var = std::env::var("WARRANT_EVIDENCE_FILTER").ok();
    let fault = match resolve_sabotage(fault_var.as_deref(), filter_var.as_deref()) {
        Ok(f) => f,
        Err(msg) => {
            eprintln!("{msg}");
            return 2;
        }
    };
    let suite = build_suite(fault);
    let registered = META.lock().unwrap().len();
    if let Err(msg) = require_tests(registered, filter_var.as_deref()) {
        eprintln!("{msg}");
        return 2;
    }
    run::main_recorded(&suite, write_manifest)
}

impl Meta {
    fn new(id: &str, kind: &'static str, op: &str, claim: &str) -> Meta {
        Meta {
            id: id.to_string(),
            kind,
            op: op.to_string(),
            claim: claim.to_string(),
            fault: String::new(),
            target: String::new(),
            domain: String::new(),
            seed: String::new(),
//@if v1
            demoted: Vec::new(),
//@endif
        }
    }
}

fn example_verdict(ok: bool) -> Verdict {
    if ok {
        Verdict::Pass
    } else {
        Verdict::fail("not-deep-equal", "example expression did not evaluate to true")
    }
}

/// A negative control passes iff at least one suite caught the injected fault.
fn negctl_result(caught: Vec<&str>, fault: &str) -> (Verdict, Stats, String) {
    if caught.is_empty() {
        (
            Verdict::fail(
                "corpus-mismatch",
                format!("fault {fault} was accepted by every suite of the operation"),
            ),
            Stats::default(),
            String::new(),
        )
    } else {
        (Verdict::Pass, Stats::default(), caught.join(","))
    }
}
