\chapter{Greetings}
\label{ch:greetings}

The Kannada greetings and the words that oil a first conversation --- and,
taught through them, how to read the Kannada script and how Kannada sits inside
the Dravidian family. Each lesson introduces the letters its word needs, so you
learn to read the word and understand it at once. \emph{Practice lessons:
\texttt{lessons/KA-C01-*.md}.}

\section{\kn{ನಮಸ್ಕಾರ} \emph{(namaskāra)} --- ``greetings,'' a making of a bow}
\label{lesson:namaskara}

\begin{sounds}
Kannada reads \textbf{left to right}, and every consonant carries a built-in
``a.'' \kn{ನ} na, \kn{ಮ} ma, \kn{ರ} ra. A \textbf{vowel sign} changes that
vowel: \kn{ಕ} ka $+$ the long-``ā'' sign $\to$ \kn{ಕಾ} kā. And a vowel-less
consonant \textbf{stacks under} the next as a conjunct (\emph{ottakṣara}):
\kn{ಸ} loses its vowel and tucks under \emph{ka} $\to$ \kn{ಸ್ಕ} ``ska.'' Read
\kn{ನ·ಮ·ಸ್·ಕಾ·ರ} $\to$ \emph{namaskāra}.
\end{sounds}

\begin{cousinweb}
\kn{ನಮಸ್ಕಾರ} (\emph{namaskāra}) is \textbf{not} native Kannada --- it is
\textbf{Sanskrit}: \emph{namas} (``a bow, homage'') $+$ \emph{kāra} (``a
making,'' from the root \emph{kṛ}, ``to do'') $=$ ``the making of a bow.'' The
very same word as Hindi \emph{namaskār}. Where its sister Tamil kept a
home-grown greeting (\emph{vaṇakkam}), Kannada borrowed freely from Sanskrit
--- the theme of this whole book.
\end{cousinweb}

\begin{cognates}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Language & ``Hello'' & Source \\
\midrule
\textbf{Kannada} & \kn{ನಮಸ್ಕಾರ} \emph{namaskāra} & Sanskrit \\
Telugu & \emph{namaskāram} & Sanskrit \\
Malayalam & \emph{namaskāram} & Sanskrit \\
Hindi & \emph{namaste} & Sanskrit \\
Tamil & \emph{vaṇakkam} & \textbf{native Dravidian} \\
English & \emph{hello} & Germanic \\
\bottomrule
\end{tabular}
\end{center}
Tamil alone kept its own word; the rest lean on the Sanskrit \emph{namas-}.
\end{cognates}

\begin{culture}
Said with the palms pressed together and a small bow --- the gesture \emph{is}
the word. It serves as both \textbf{hello and goodbye}, any time, to almost
anyone, respectful without being stiff.
\end{culture}

\section{\kn{ಧನ್ಯವಾದ} \emph{(dhanyavāda)} --- ``thank you,'' an utterance of ``worthy''}
\label{lesson:dhanyavada}

\begin{sounds}
New: \kn{ಧ} \emph{dha} (an \textbf{aspirated} \emph{d} --- a puff of breath),
and the conjunct \kn{ನ್ಯ} (\emph{na} stacked with \emph{ya} $\to$ ``nya'').
\kn{ವಾ} is \emph{vā}. Read \kn{ಧ·ನ್ಯ·ವಾ·ದ} $\to$ \emph{dhanyavāda}.
\end{sounds}

\begin{cousinweb}
\kn{ಧನ್ಯವಾದ} (\emph{dhanyavāda}) is \textbf{Sanskrit}: \emph{dhanya} (``worthy,
blessed,'' from \emph{dhana}, ``wealth'') $+$ \emph{vāda} (``a saying,'' root
\emph{vad}, ``to speak'') $=$ ``an utterance of `[you are] worthy.''' The same
word as Hindi \emph{dhanyavād} and Telugu \emph{dhanyavādamulu}.
\end{cousinweb}

\begin{cognates}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Language & ``Thanks'' & Note \\
\midrule
\textbf{Kannada} & \kn{ಧನ್ಯವಾದ} \emph{dhanyavāda} & Sanskrit \\
Telugu & \emph{dhanyavādamulu} & Sanskrit --- same word \\
Hindi & \emph{dhanyavād} & Sanskrit --- same word \\
Tamil & \emph{naṉṟi} & \textbf{native} (``goodness'') \\
Malayalam & \emph{nandi} & \textbf{native} --- Tamil's cousin \\
\bottomrule
\end{tabular}
\end{center}
A stark split: Kannada, Telugu, Hindi take the Sanskrit word; Tamil and
Malayalam keep native \emph{naṉṟi / nandi}.
\end{cognates}

\section{\kn{ಹೌದು} \emph{(haudu)} --- ``yes''}
\label{lesson:haudu}

\begin{sounds}
\kn{ಹೌ} is \emph{hau} (\kn{ಹ} \emph{ha} with the ``au'' vowel sign hugging it);
\kn{ದು} is \emph{du} (\kn{ದ} \emph{da} with the ``u'' sign beneath). Read
\kn{ಹೌ·ದು} $\to$ \emph{haudu}.
\end{sounds}

\begin{cousinweb}
\kn{ಹೌದು} (\emph{haudu}) is the everyday ``yes / that's so'' --- a genuinely
\textbf{native} Kannada word, after two Sanskrit loans.
\end{cousinweb}

\begin{grammarlens}
Like its sister Tamil, Kannada often says ``yes'' by \textbf{echoing the verb}:
asked \emph{bandiddīyā?} (``have you come?''), a natural yes is
\emph{bandiddēne} (``I have come''). Bank \emph{haudu} as the simple word, but
expect a ``yes'' to come back as a whole verb --- a habit shared across the
Dravidian family.
\end{grammarlens}

\section{\kn{ಇಲ್ಲ} \emph{(illa)} --- ``no,'' a word Tamil shares almost exactly}
\label{lesson:illa}

\begin{sounds}
\kn{ಇ} is the \textbf{independent vowel} ``i'' (word-initial, so a full
letter). \kn{ಲ್ಲ} is ``lla'' --- \kn{ಲ} \emph{la} vowel-stripped and stacked
under another \emph{la}, a held ``ll.'' Read \kn{ಇ·ಲ್ಲ} $\to$ \emph{illa}.
\end{sounds}

\begin{cousinweb}
\kn{ಇಲ್ಲ} (\emph{illa}) is built on the old Dravidian root \emph{il},
``not-being'' --- so, like Tamil \emph{illai}, it really means \textbf{``[it]
is not, [there] is not,''} serving as the everyday ``no.'' Its positive twin is
\kn{ಇದೆ} (\emph{ide}), ``there is.''
\end{cousinweb}

\begin{grammarlens}
\textbf{Dravidian negates by \emph{being}.} Where English inserts \emph{not},
Kannada --- exactly like Tamil --- swaps in the negative verb of existence:
\emph{pustaka \textbf{ide}} (``there \emph{is} a book'') $\to$ \emph{pustaka
\textbf{illa}} (``there \emph{is not} a book''). A shared inheritance: Tamil
\emph{illai}, Kannada \emph{illa}, Malayalam \emph{illa} --- one family, one way
to say a thing is not.
\end{grammarlens}

\begin{cognates}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Language & ``No / isn't'' & Note \\
\midrule
\textbf{Kannada} & \kn{ಇಲ್ಲ} \emph{illa} & native (root \emph{il}) \\
Tamil & \emph{illai} & the \emph{same} root \\
Malayalam & \emph{illa} & the \emph{same} root \\
Telugu & \emph{lēdu} & native, a different root \\
Hindi & \emph{nahīṁ} & Indo-European (unrelated) \\
\bottomrule
\end{tabular}
\end{center}
\end{cognates}

\section{\kn{ಸರಿ} \emph{(sari)} --- ``okay,'' shared whole with Tamil}
\label{lesson:sari}

\begin{sounds}
\kn{ಸ} is \emph{sa}; \kn{ರಿ} is \emph{ri} (\kn{ರ} \emph{ra} with the ``i''
sign). No conjunct here --- two clean, round syllables. Read \kn{ಸ·ರಿ} $\to$
\emph{sari}.
\end{sounds}

\begin{cousinweb}
\kn{ಸರಿ} (\emph{sari}) is native Dravidian for ``correct, even, right,'' and so
``okay, agreed'' --- the verbal nod of Kannada. It is the \emph{same word} as
Tamil \emph{sari} --- identical in sound and meaning, only the script differs;
doubled, \emph{sari sari} means ``alright, alright.''
\end{cousinweb}

\begin{grammarlens}
\textbf{One shared word, two scripts.} \emph{sari} is identical in Kannada and
Tamil in sound and meaning --- only the letters differ (Kannada writes it
\kn{ಸರಿ}, Tamil in its own script). Watching a shared word change its clothes
but not its sound is the
fastest way to feel how close these languages are --- the pattern-matching that
carries you across Kannada, Telugu, and Malayalam.
\end{grammarlens}

\section{Recap, and how Kannada says goodbye}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Read & & Meaning & Origin \\
\midrule
\kn{ನಮಸ್ಕಾರ} & \emph{namaskāra} & hello & Sanskrit \\
\kn{ಧನ್ಯವಾದ} & \emph{dhanyavāda} & thank you & Sanskrit \\
\kn{ಹೌದು} & \emph{haudu} & yes & native \\
\kn{ಇಲ್ಲ} & \emph{illa} & no / isn't & native (= Tamil \emph{illai}) \\
\kn{ಸರಿ} & \emph{sari} & okay & native (= Tamil \emph{sari}) \\
\bottomrule
\end{tabular}
\end{center}

The pattern is already visible: the two greeting/politeness words are Sanskrit
loans; the three everyday grammar words are native Dravidian, two of them
shared almost unchanged with Tamil.

\begin{culture}
\textbf{The same ``go and come back'' as Tamil.} Kannada, too, avoids a bare
``I'm leaving.'' The everyday goodbye is \kn{ಹೋಗಿ ಬರುತ್ತೇನೆ} (\emph{hōgi
baruttēne}), ``having gone, I [will] come [back],'' and the reply is
\kn{ಹೋಗಿ ಬನ್ನಿ} (\emph{hōgi banni}), ``go, and come back.'' The very worldview
you met in Tamil's \emph{pōy varugiṟēṉ}: never simply depart --- promise a
return.
\end{culture}
