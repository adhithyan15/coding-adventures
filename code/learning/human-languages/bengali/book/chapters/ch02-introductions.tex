\chapter{Introducing Yourself}
\label{ch:introductions}

Now a first real exchange --- giving your name, asking for someone else's, and
saying you're glad to meet them:

\begin{center}
\bn{আমার নাম মিরা।} \quad(\emph{\=am\=ar n\=am Mira} --- ``My name is Mira.'')\\
\bn{তোমার নাম কি?} \quad(\emph{tom\=ar n\=am ki} --- ``What is your name?'')
\end{center}

Built the same way as the greetings --- each piece taken apart, its letters and
root, then assembled. \emph{Practice lessons: \texttt{lessons/BN-C02-*.md}.}

\section{\bn{নাম} \emph{(n\=am)} --- ``name,'' a word older than Europe}
\label{lesson:naam}

\begin{sounds}
\bn{ন} (\emph{n\^{o}}) $+$ \bn{া} (long \=a, overriding the inherent \^{o}) $+$
\bn{ম} (\emph{m\^{o}}) $\to$ \bn{নাম} (\emph{n\=am}).
\end{sounds}

\begin{cousinweb}
\bn{নাম} (\emph{n\=am}) $\leftarrow$ Sanskrit \emph{n\=aman}, from
Proto-Indo-European *h\textsubscript{3}n\'{o}mn\d{} --- the \emph{same} root as
English \textbf{name}, Latin \emph{n\=omen} (English \textbf{noun}), and Greek
\emph{onoma}. Bengali \emph{n\=am} and English \emph{name} are cousins from before
either language existed.
\end{cousinweb}

\section{\bn{আমার} \emph{(\=am\=ar)} --- ``my''}
\label{lesson:amar}

\begin{cousinweb}
\bn{আমার} (\emph{\=am\=ar}, ``my'') is the possessive of \bn{আমি} (\emph{\=ami},
``I''), on the first-person stem \emph{\=am-} $\leftarrow$ Sanskrit \emph{asmad}.
Buried in it is the same ancient \textbf{m-} of English \textbf{me}, \textbf{my},
\textbf{mine} and Latin \emph{m\=e}.
\end{cousinweb}

\begin{grammarlens}
\textbf{No gender here.} \emph{\=am\=ar} never changes for gender, where Hindi and
Punjabi have \emph{mer\=a} (m.) / \emph{mer\=\i} (f.). Bengali threw grammatical
gender away entirely --- one of its great simplifications, which you will meet
again and again.
\end{grammarlens}

\section{\bn{আমার নাম \ldots} \emph{(\=am\=ar n\=am \ldots)} --- ``my name is\ldots,'' with no ``is''}
\label{lesson:amar-naam}

\bn{আমার} (my) $+$ \bn{নাম} (name) $+$ name $=$ ``my name \ldots'' $\to$ ``my name
is\ldots'' \emph{\=Am\=ar n\=am Arun.}

\begin{grammarlens}
\textbf{Bengali drops ``is'' in the present.} There is \emph{no verb}: Bengali
sets ``my name'' beside ``Arun'' and the ``is'' is understood. This is the
\textbf{zero copula} --- for present-tense equations (X is Y), Bengali needs no
``to be'' (like Tamil, and unlike Hindi's \emph{mer\=a n\=am \ldots hai}). A verb
returns only for a \emph{state} or \emph{place} (Chapter 3).
\end{grammarlens}

\section{\bn{তুমি} / \bn{আপনি} \emph{(tumi / \=apni)} --- ``you,'' three ways}
\label{lesson:tumi-apni}

\begin{cousinweb}
\bn{তুমি} (\emph{tumi}, familiar ``you'') $\leftarrow$ Sanskrit \emph{tvam}, from
PIE *t\=u --- the same root as English archaic \textbf{thou}, Latin \emph{t\=u}.
Bengali grades respect in \textbf{three}: \bn{তুই} (\emph{tui}, intimate),
\bn{তুমি} (\emph{tumi}, familiar), \bn{আপনি} (\emph{\=apni}, respectful) --- like
Hindi's \emph{t\=u/tum/\=ap}. For a first meeting, \emph{\=apni}.
\end{cousinweb}

\section{\bn{কি} \emph{(ki)} --- ``what''}
\label{lesson:ki}

\begin{cousinweb}
\bn{কি} (\emph{ki}, ``what'') $\leftarrow$ the interrogative stem \emph{ka-}, PIE
*k\ensuremath{^w}o- --- the root behind English \textbf{wh-} and Latin \emph{qu-}.
Bengali's questions nearly all begin with \emph{k-}: \emph{ki} (what), \emph{ke}
(who), \emph{k\^{o}be} (when), \emph{k\=oth\=ay} (where), \emph{k\=emon} (how).
\end{cousinweb}

\section{\bn{তোমার নাম কি?} \emph{(tom\=ar n\=am ki)} --- ``what's your name?''}
\label{lesson:tomar-naam-ki}

\bn{তোমার} (\emph{tom\=ar}, ``your,'' from \emph{tumi}) $+$ \bn{নাম} (name) $+$
\bn{কি} (what) $=$ ``\textbf{your name what?}'' --- still \emph{no ``is.''} The
respectful ``your'' is \bn{আপনার} (\emph{\=apn\=ar}, from \emph{\=apni}):
\emph{\=apn\=ar n\=am ki?} --- the one for a stranger.

\section{\bn{আলাপ করে ভালো লাগলো} \emph{(\=al\=ap kore bh\=alo l\=aglo)} --- ``pleased to meet you''}
\label{lesson:alaap}

\begin{cousinweb}
\bn{আলাপ করে ভালো লাগলো} --- ``having made [your] acquaintance, it felt good.'' The
key word \bn{আলাপ} (\emph{\=al\=ap}, ``acquaintance, conversation'') is
\textbf{Sanskrit} (\emph{\=al\=apa}, \emph{\=a-} ``toward'' $+$ \emph{lap} ``to
speak'') --- the same \emph{\=al\=ap} that names the slow, exploratory opening of
a r\=aga, where a musician ``converses'' with the notes.
\end{cousinweb}

\section{The whole exchange}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Bengali & English \\
\midrule
\bn{আমার নাম মিরা।} & My name is Mira. \\
\bn{আপনার নাম কি?} & What's your name? \\
\bn{আমার নাম অরুণ।} & My name is Arun. \\
\bn{আলাপ করে ভালো লাগলো।} & Pleased to meet you. \\
\bottomrule
\end{tabular}
\end{center}

Every atom traced: \emph{n\=am} (Sanskrit \emph{n\=aman}, English \emph{name}),
\emph{\=am\=ar} (first-person \emph{\=am-}, no gender), \emph{ki} (root
\emph{ka-}, English \emph{what}), \emph{\=al\=ap} (Sanskrit ``conversing''). Two
Bengali habits already: the \textbf{zero copula} (no ``is''), and \textbf{no
grammatical gender}. Next chapter: \emph{tumi k\=emon \=achho?} (``how are
you?'').
