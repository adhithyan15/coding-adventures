\section{\emph{Hay} --- ``There Is / There Are''}

\begin{etymology}
\textbf{hay} $\leftarrow$ Latin \emph{habet} (``[it] has''), the same root
behind \emph{haber}, an auxiliary ``to have'' met later, related to English
\emph{have} through French.
\end{etymology}

\begin{grammarlens}
\emph{Hay} answers a different question than \emph{ser} or \emph{estar}:
not ``what is this'' or ``what state is it in,'' but simply ``does this
exist, here?'' --- \emph{Hay un libro} / \emph{Hay dos libros}. \emph{Hay}
itself never changes between those two sentences, even though English
``there IS a book'' / ``there ARE two books'' does change --- Spanish puts
all the number information on the noun and leaves the existence-word
alone. A third ``to be''-shaped concept, after \emph{ser} and \emph{estar},
that English collapses into one (number-agreeing) verb.
\end{grammarlens}

\section{\emph{Tener} --- ``To Have''}

\begin{vocabtable}
tengo / tienes / tiene & --- & I have / you have / he-she-it has \\
tenemos / tienen & --- & we have / they have \\
\end{vocabtable}

\begin{etymology}
$\leftarrow$ Latin \emph{tenere} (``to hold'' --- English \emph{tenant,
tenacious, contain, maintain, retain}).
\end{etymology}

\begin{grammarlens}
\emph{Tener} is a \textbf{stem-changing verb}: \emph{tengo} keeps a plain
stem, but \emph{tienes/tiene/tienen} insert an \emph{-ie-} the infinitive
doesn't have --- an internal vowel shift with no English parallel (English
verbs conjugate by changing endings, never by altering the middle of the
word). Spanish also uses \emph{tener} for states English expresses with
``to be'': \emph{tengo hambre} (``I'm hungry,'' literally ``I have
hunger''), \emph{tengo fr\'{i}o} (``I'm cold,'' literally ``I have cold'')
--- a real, non-obvious mapping worth learning directly rather than
translating word for word.
\end{grammarlens}

\section{Possessive Adjectives}

\begin{vocabtable}
mi / tu / su / nuestro & --- & my / your / his-her-your-their / our \\
\end{vocabtable}

\begin{grammarlens}
Possessives agree with the noun they describe, \textbf{not} with the
owner: \emph{nuestro libro} (masc.) but \emph{nuestra casa} (fem.),
regardless of who ``we'' are --- English possessives never change form
based on the thing owned. \emph{su} is worth sitting with: one word covers
English ``his,'' ``her,'' ``your'' (formal), and ``their'' all at once,
disambiguated only by context, the same trust-the-context logic as
\emph{c\'{o}mo}'s accent mark.
\end{grammarlens}

\section{Negation}

\begin{vocabtable}
no / nunca / nada / nadie & --- & no/not / never / nothing / nobody \\
\end{vocabtable}

\begin{etymology}
\emph{no} $\leftarrow$ Latin \emph{non}. \emph{nunca} $\leftarrow$
\emph{numquam}. \emph{nada} $\leftarrow$ \emph{(res) nata}, ``(a thing)
born,'' worn down into ``not a single thing.'' \emph{nadie} $\leftarrow$
\emph{nati} (``born ones''), the same root as \emph{nada}.
\end{etymology}

\begin{grammarlens}
Spanish \textbf{requires} double negatives, which English grammar rules
explicitly forbid: \emph{No tengo nada} (literally ``I don't have
nothing'') is the correct, mandatory way to say ``I don't have anything.''
Dropping the \emph{no} doesn't fix it --- it breaks the sentence the other
direction. One of the most-corrected ``mistakes'' English speakers make in
Spanish, precisely because it's the opposite of the English rule.
\end{grammarlens}
