\chapter{Greetings and responses}
\label{ch:urdu-greetings}

This chapter builds one miniature exchange in dependency order: greet, thank,
answer yes, answer no. Four short expressions establish right-to-left reading,
joining, omitted short vowels, long vowels, respectful \emph{j\={i}}, and one
Urdu nasal sign. Companion lessons: \texttt{UR-C01-*.md}.

\section{\texorpdfstring{\ur{سلام}}{salam} \emph{(sal\={a}m)} --- hello through peace}

\begin{scriptstep}
Urdu runs \textbf{right to left}. Read \ur{سلام} from its right edge:
\ur{س} $s$, \ur{ل} $l$, \ur{ا} long \emph{\={a}}, \ur{م} $m$. Letters usually
join inside a word. The tall \ur{ا} does not connect to the following letter,
so the final \ur{م} begins a new stroke group. No other letters are due yet.
\end{scriptstep}

\begin{rootweb}
\emph{Sal\={a}m} is a greeting and the word ``peace.'' Urdu received it through
the Persian-Arabic literary world. Its consonants \textbf{s--l--m} belong to a
Semitic family concerned with peace, safety, and wholeness. Persian
\emph{sal\^am}, Arabic \emph{sal\={a}m}, and Hebrew \emph{shalom} make the
historical family especially visible.
\end{rootweb}

\begin{usage}
Say \ur{سلام} to begin a friendly exchange. For now the safest reply is the
same word: A says \ur{سلام}; B answers \ur{سلام}. One useful move is enough.
\end{usage}

\begin{checkpoint}
Read \ur{سلام} from right to left. Which letter carries long \emph{\={a}}?
Point to \ur{ا}, cover the romanization, and say \emph{sal\={a}m} once.
\end{checkpoint}

\section{\texorpdfstring{\ur{شکریہ}}{shukriya} \emph{(shukriy\={a})} --- thank you}

\begin{scriptstep}
Urdu normally leaves short vowels unmarked. Read the visible guide:
\ur{ش} $sh$, \ur{ک} $k$, \ur{ر} $r$, \ur{ی} $y/i$, \ur{ہ} final $h$-shape.
The learned whole supplies the short $u$ and $i$, while the ending is pronounced
long \emph{\={a}}. First remember one joined word, not every possible letter form.
\end{scriptstep}

\begin{rootweb}
The central consonants \textbf{sh--k--r} belong to the Arabic family for
gratitude; Arabic \emph{shukran} is a close comparison. Urdu
\emph{shukriy\={a}} reflects its long Persian-Arabic lexical history and is an
ordinary modern Urdu expression.
\end{rootweb}

\begin{usage}
Build a two-step chain: \ur{سلام} \emph{sal\={a}m} --- hello;
\ur{شکریہ} \emph{shukriy\={a}} --- thank you. Keep the second word in three
light beats: \emph{shuk-ri-y\={a}}.
\end{usage}

\begin{checkpoint}
Which new letter begins \ur{شکریہ}? Point to \ur{ش}, say $sh$, then produce
\emph{sal\={a}m --- shukriy\={a}} without looking.
\end{checkpoint}

\section{\texorpdfstring{\ur{جی ہاں}}{ji han} \emph{(jī hā̃)} --- yes, respectfully}

\begin{scriptstep}
This answer has two small words. In \ur{جی}, \ur{ج} gives $j$ and \ur{ی}
stretches the vowel to \emph{\={i}}. In \ur{ہاں}, the final \ur{ں} is
\textbf{n\={u}n ghunna}: it nasalizes long \emph{\={a}} rather than adding a
strong English $n$. Let a little air pass through the nose.
\end{scriptstep}

\begin{usage}
\ur{ہاں} means ``yes.'' Adding \ur{جی} makes the response attentive and
respectful; \ur{جی} can also answer by itself. Learn the safely courteous
whole first: \ur{جی ہاں} \emph{jī hā̃}.
\end{usage}

\begin{checkpoint}
Which final sign nasalizes the vowel? Point to \ur{ں}. Then answer politely:
\ur{جی ہاں}.
\end{checkpoint}

\section{\texorpdfstring{\ur{نہیں}}{nahin} \emph{(nahī̃)} --- no, and later not}

\begin{scriptstep}
Reuse \ur{ہ}, \ur{ی}, and nasal \ur{ں}; add \ur{ن} $n$ at the right edge.
The result is \ur{نہیں} \emph{nahī̃}. Stretch \emph{\={i}} slightly and
nasalize it; do not add a separate forceful $n$ at the end.
\end{scriptstep}

\begin{rootweb}
Unlike the Persian-Arabic vocabulary that opened the chapter,
\emph{nahī̃} belongs to Urdu's inherited Indo-Aryan core. Hindi
\emph{nahī̃} is the same spoken word written in a
different script. The shared form previews close grammatical kinship without
making Hindi script a prerequisite.
\end{rootweb}

\begin{usage}
As a complete answer, \ur{نہیں} means ``no.'' Inside a later sentence it will
also mean ``not,'' but that second job can wait. Contrast only:
\ur{جی ہاں} --- yes; \ur{نہیں} --- no.
\end{usage}

\begin{checkpoint}
Without looking back, produce the chain hello, thank you, yes, no. Then read
\ur{سلام}, \ur{شکریہ}, \ur{جی ہاں}, \ur{نہیں} in the same order.
\end{checkpoint}
