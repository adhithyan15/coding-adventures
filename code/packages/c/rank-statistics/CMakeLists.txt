cmake_minimum_required(VERSION 3.20)
project(coding_adventures_rank_statistics VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(rank_statistics STATIC src/rank_statistics.c)
add_library(coding_adventures::rank_statistics ALIAS rank_statistics)
set_target_properties(rank_statistics PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(rank_statistics PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_options(rank_statistics PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(rank_statistics PRIVATE -Wall -Wextra -Werror -pedantic-errors)
endif()
install(TARGETS rank_statistics EXPORT rank_statistics_targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/rank_statistics.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md CHANGELOG.md LICENSE required_capabilities.json DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/coding_adventures_rank_statistics)
install(EXPORT rank_statistics_targets NAMESPACE coding_adventures:: FILE rank_statistics-targets.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rank_statistics)
configure_package_config_file(cmake/rank_statistics-config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/rank_statistics-config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rank_statistics)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/rank_statistics-config-version.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/rank_statistics-config.cmake ${CMAKE_CURRENT_BINARY_DIR}/rank_statistics-config-version.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/rank_statistics)
