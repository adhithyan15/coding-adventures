cmake_minimum_required(VERSION 3.16)
project(ca_heap_installed_consumer LANGUAGES CXX)

if(NOT DEFINED CA_HEAP_RELOCATED_PREFIX OR CA_HEAP_RELOCATED_PREFIX STREQUAL "")
  message(FATAL_ERROR "CA_HEAP_RELOCATED_PREFIX is required")
endif()
if(NOT DEFINED CA_HEAP_SOURCE_ROOT OR CA_HEAP_SOURCE_ROOT STREQUAL "")
  message(FATAL_ERROR "CA_HEAP_SOURCE_ROOT is required")
endif()
if(NOT DEFINED CA_HEAP_ORIGINAL_PREFIX OR CA_HEAP_ORIGINAL_PREFIX STREQUAL "")
  message(FATAL_ERROR "CA_HEAP_ORIGINAL_PREFIX is required")
endif()

find_package(ca_heap CONFIG REQUIRED
  PATHS "${CA_HEAP_RELOCATED_PREFIX}"
  NO_DEFAULT_PATH)

function(ca_heap_assert_inside_prefix value label)
  get_filename_component(_value "${value}" REALPATH)
  get_filename_component(_prefix "${CA_HEAP_RELOCATED_PREFIX}" REALPATH)
  file(TO_CMAKE_PATH "${_value}" _value)
  file(TO_CMAKE_PATH "${_prefix}" _prefix)
  if(WIN32)
    string(TOLOWER "${_value}" _value_cmp)
    string(TOLOWER "${_prefix}" _prefix_cmp)
  else()
    set(_value_cmp "${_value}")
    set(_prefix_cmp "${_prefix}")
  endif()
  string(LENGTH "${_prefix_cmp}" _prefix_len)
  string(LENGTH "${_value_cmp}" _value_len)
  if(_value_len LESS_EQUAL _prefix_len)
    message(FATAL_ERROR "${label} is not below relocated prefix: ${_value}")
  endif()
  string(SUBSTRING "${_value_cmp}" 0 ${_prefix_len} _value_prefix)
  if(NOT _value_prefix STREQUAL _prefix_cmp)
    message(FATAL_ERROR "${label} is outside relocated prefix: ${_value}")
  endif()
  string(SUBSTRING "${_value_cmp}" ${_prefix_len} 1 _separator)
  if(NOT _separator STREQUAL "/")
    message(FATAL_ERROR "${label} is not contained by relocated prefix: ${_value}")
  endif()
endfunction()

ca_heap_assert_inside_prefix("${ca_heap_DIR}" "package config directory")
get_target_property(_heap_includes ca::heap INTERFACE_INCLUDE_DIRECTORIES)
if(NOT _heap_includes)
  message(FATAL_ERROR "ca::heap exports no include directory")
endif()
foreach(_include IN LISTS _heap_includes)
  if(_include MATCHES "\\$<")
    message(FATAL_ERROR "Unresolved generator expression in installed include: ${_include}")
  endif()
  ca_heap_assert_inside_prefix("${_include}" "ca::heap include directory")
endforeach()

get_filename_component(_source "${CA_HEAP_SOURCE_ROOT}" REALPATH)
get_filename_component(_original "${CA_HEAP_ORIGINAL_PREFIX}" REALPATH)
get_filename_component(_config "${ca_heap_DIR}" REALPATH)
foreach(_forbidden IN ITEMS "${_source}" "${_original}")
  file(TO_CMAKE_PATH "${_forbidden}" _forbidden)
  file(TO_CMAKE_PATH "${_config}" _config_normalized)
  if(WIN32)
    string(TOLOWER "${_forbidden}" _forbidden_cmp)
    string(TOLOWER "${_config_normalized}" _config_cmp)
  else()
    set(_forbidden_cmp "${_forbidden}")
    set(_config_cmp "${_config_normalized}")
  endif()
  string(FIND "${_config_cmp}" "${_forbidden_cmp}" _found)
  if(NOT _found EQUAL -1)
    message(FATAL_ERROR "Package config points into forbidden path: ${_config}")
  endif()
endforeach()

add_executable(ca_heap_consumer main.cpp)
target_link_libraries(ca_heap_consumer PRIVATE ca::heap)
enable_testing()
add_test(NAME ca_heap_installed_consumer COMMAND ca_heap_consumer)
