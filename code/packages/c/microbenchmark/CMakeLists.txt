cmake_minimum_required(VERSION 3.20)
project(coding_adventures_microbenchmark VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(rank_statistics 0.1 CONFIG REQUIRED)
find_package(bounded_json 0.1 CONFIG REQUIRED)
add_library(microbenchmark STATIC src/microbenchmark.c src/native_clock.c src/report_json.c)
add_library(coding_adventures::microbenchmark ALIAS microbenchmark)
set_target_properties(microbenchmark PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
target_include_directories(microbenchmark PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(microbenchmark PUBLIC coding_adventures::rank_statistics coding_adventures::bounded_json)
if(MSVC)
  target_compile_options(microbenchmark PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(microbenchmark PRIVATE -Wall -Wextra -Werror -pedantic-errors)
endif()
add_executable(array-microbenchmark src/array-microbenchmark.c)
target_link_libraries(array-microbenchmark PRIVATE coding_adventures::microbenchmark)
install(TARGETS microbenchmark array-microbenchmark EXPORT microbenchmark_targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/microbenchmark.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md CHANGELOG.md LICENSE required_capabilities.json testdata/microbenchmark-pairs-v1.json DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/coding_adventures_microbenchmark)
install(EXPORT microbenchmark_targets NAMESPACE coding_adventures:: FILE microbenchmark-targets.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/microbenchmark)
configure_package_config_file(cmake/microbenchmark-config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/microbenchmark-config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/microbenchmark)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/microbenchmark-config-version.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/microbenchmark-config.cmake ${CMAKE_CURRENT_BINARY_DIR}/microbenchmark-config-version.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/microbenchmark)
