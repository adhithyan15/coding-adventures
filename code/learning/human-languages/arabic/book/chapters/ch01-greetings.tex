\chapter{Greetings}
\label{ch:greetings}

The Arabic greetings, built from three-consonant roots and the attached
article \emph{al-}. Because the reader knows Spanish, Arabic's shadow over
Spanish (the \emph{al-} words, the sun-letter assimilation in \emph{az\'{u}car})
runs throughout. \emph{Practice lessons: \texttt{lessons/AR-C01-*.md}.}

\section{The root system --- this whole book, but literal}
\label{lesson:root-system}

Most Arabic words are built on a \textbf{root} of three consonants carrying a
core meaning, poured into fixed vowel \textbf{patterns}. The classic:
\textbf{k--t--b} (\ar{ك ت ب}), \emph{writing} --- \emph{kataba} (he wrote),
\emph{kit\=ab} (\ar{كتاب}, book), \emph{k\=atib} (\ar{كاتب}, writer),
\emph{maktab} (\ar{مكتب}, office = ``a place of writing''), \emph{maktaba}
(library). Six words, one root; the pattern \emph{ma--\,-a-} makes ``a place
of ---.''

\begin{cousinweb}
This is the Latin/Greek root idea from the earlier books, but \emph{more
regular and more visible}: where English hides \emph{scribe/describe/script}
inside \emph{scrib-}, Arabic wears the root on the surface of every derived
word. Every greeting below is a root plus a pattern. (And: Arabic reads
\textbf{right to left}, letters connecting --- reintroduced inside real words,
not drilled up front.)
\end{cousinweb}

\section{\ar{مرحبا} \emph{(mar\d{h}aban)} --- ``hello,'' i.e. ``there's room for you''}
\label{lesson:marhaba}

\begin{cousinweb}
\ar{مرحبا} (\emph{mar\d{h}aban}), the casual ``hi.'' Root \textbf{r--\d{h}--b}
(\ar{ر ح ب}), \emph{spaciousness}: \emph{ra\d{h}uba} (to be wide), \emph{ra\d{h}b}
(roomy) $\to$ \emph{mar\d{h}aban} = ``[a place of] welcome, there is room for
you.'' Not an empty sound --- a little sentence, like \emph{buenos d\'{i}as}
was a blessing. Reply: \emph{mar\d{h}abtayn}, ``two welcomes.''
\end{cousinweb}

\begin{sounds}
\ar{ح} (\d{h}) is a heavy, throaty \emph{h} --- \emph{mar-\d{H}A-ban}.
\end{sounds}

\section{\ar{سلام} \emph{(sal\=am)} --- ``peace,'' the most famous root}
\label{lesson:salam}

\begin{cousinweb}
\ar{سلام} (\emph{sal\=am}), ``peace.'' Root \textbf{s--l--m} (\ar{س ل م}),
\emph{wholeness, safety, peace} --- an enormous family: \emph{sal\=am} (peace),
\emph{isl\=am} (submission $\to$ peace with God), \emph{muslim} (``one who
submits,'' the \emph{mu--\,-i-} pattern), \emph{sallama} (to greet). It reaches
past Arabic: Hebrew \textbf{shalom} is the \emph{same} Semitic root. English
meets it as the borrowing \emph{salaam} and the name \emph{Sal\=im}. A window
into the Semitic family, as Latin roots opened the Romance one.
\end{cousinweb}

\section{\ar{ال} \emph{(al-)} --- ``the,'' hiding in English \emph{algebra}}
\label{lesson:al}

\begin{cousinweb}
\ar{ال} (\emph{al-}, ``the''), attached to its noun. For $\sim$800 years Arabic
was the Mediterranean's language of science and trade, and English/Spanish
absorbed its words \emph{with al- attached}: \textbf{al}gebra (\emph{al-jabr}),
\textbf{al}cohol (\emph{al-ku\d{h}l}), \textbf{al}gorithm (\emph{al-Khw\=arizm\=\i}),
\textbf{al}chemy, \textbf{al}manac. Learning Arabic \emph{al-} is meeting an
old friend.
\end{cousinweb}

\begin{grammarlens}
\textbf{Sun letters.} Before certain consonants (\ar{س} \emph{s}, \ar{ش}
\emph{sh}, \ar{ن} \emph{n}, \ar{ر} \emph{r}, \ar{ت} \emph{t}\ldots) the \emph{l}
of \emph{al-} assimilates and the letter doubles: \emph{al-\,+\,sal\=am} $\to$
\emph{as-sal\=am}; \emph{al-\,+\,n\=ur} $\to$ \emph{an-n\=ur}. Before ``moon
letters'' (\ar{خ} \emph{kh}, \ar{ك} \emph{k}\ldots) the \emph{l} stays:
\emph{al-khayr}. You can hear the rule inside Spanish \emph{az\'{u}car} (from
\emph{as-sukkar}).
\end{grammarlens}

\section{\ar{السلام عليكم} \emph{(as-sal\=amu `alaykum)} --- ``peace be upon you''}
\label{lesson:as-salamu-alaykum}

\ar{السلام عليكم} $=$ \emph{as-sal\=amu} (\emph{al-}\,+\,\emph{sal\=am}, sun-letter
\emph{s} assimilated) $+$ \emph{`alaykum} (``upon you [all]''). Literally ``the
peace [be] upon you'' --- and note there is \textbf{no word for ``is''};
Arabic sets the two phrases side by side and lets ``is'' be understood (where
Spanish needs \emph{est\'{a}}).

\begin{culture}
The universal greeting; its reply is fixed: \ar{وعليكم السلام} (\emph{wa-`alaykum
as-sal\=am}) --- ``\emph{and} upon you the peace.'' You give peace; it is
returned. Any time, any register, real weight --- you are wishing \emph{peace},
not saying ``hi.''
\end{culture}

\begin{sounds}
\ar{ع} (\emph{`ayn}) in \emph{`alaykum} is a throat-tightening with no English
equivalent --- approximate it.
\end{sounds}

\section{\ar{صباح الخير} \emph{(\d{s}ab\=a\d{h} al-khayr)} --- ``morning of goodness''}
\label{lesson:sabah-al-khayr}

\begin{cousinweb}
\ar{صباح} (\emph{\d{s}ab\=a\d{h}}, ``morning''), root \textbf{\d{s}--b--\d{h}}
(dawn/waking: \emph{\d{s}ub\d{h}} dawn, \emph{a\d{s}ba\d{h}a} ``to become,
reach morning''). \ar{خير} (\emph{khayr}, ``good, goodness''), root
\textbf{kh--y--r}. Together \ar{صباح الخير} $=$ ``morning of the goodness'' ---
\emph{al-khayr} keeps its \emph{l} (moon letter \ar{خ}), a clean contrast with
\emph{as-sal\=am}.
\end{cousinweb}

\begin{culture}
Reply: \ar{صباح النور} (\emph{\d{s}ab\=a\d{h} an-n\=ur}) --- ``morning of the
\emph{light}'' (\emph{al-}\,+\,\emph{n\=ur}, sun-letter \emph{n} assimilated).
You wish goodness; they wish you light.
\end{culture}

\begin{sounds}
\ar{ص} (\d{s}) is an \emph{emphatic} (heavy) \emph{s}; \ar{خ} (kh) is the
throat-scrape, \textbf{= Spanish \emph{j}}.
\end{sounds}

\section{\ar{مساء الخير} \emph{(mas\=aʾ al-khayr)} --- ``evening of goodness''}
\label{lesson:masa-al-khayr}

\begin{cousinweb}
\ar{مساء} (\emph{mas\=aʾ}, ``evening''), root \textbf{m--s--w}: \emph{ams\=a}
``to become, reach evening'' --- the exact mirror of \emph{a\d{s}ba\d{h}a}.
Arabic names morning and evening as matched events (each with a ``to become''
verb), where French \emph{soir} and Spanish \emph{tarde} both merely meant
``late.'' \ar{مساء الخير} $=$ ``evening of goodness''; reply \ar{مساء النور}
(\emph{mas\=aʾ an-n\=ur}).
\end{cousinweb}

\begin{sounds}
Final \ar{ء} (hamza) is a glottal stop --- \emph{ma-S\=Aʾ}.
\end{sounds}

\section{\ar{شكرا} \emph{(shukran)} --- ``thank you''}
\label{lesson:shukran}

\begin{cousinweb}
\ar{شكرا} (\emph{shukran}), root \textbf{sh--k--r}, \emph{gratitude}:
\emph{shukr} (thanks), \emph{sh\=akir} (thankful), \emph{mashk\=ur} (thanked).
Look: \emph{sh\=akir} and \emph{mashk\=ur} sit in the \emph{exact} patterns as
\emph{k\=atib} (writer) and \emph{makt\=ub} (written) from the root lesson ---
learn a pattern once, it works on every root. Reply: \ar{عفوا}
(\emph{`afwan}), ``it's nothing.''
\end{cousinweb}
