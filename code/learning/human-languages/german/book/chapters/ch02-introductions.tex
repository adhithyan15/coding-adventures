\chapter{Introducing Yourself}
\label{ch:introductions}

Now a first real exchange --- giving your name, asking for someone else's, and
saying you're glad to meet them:

\begin{center}\itshape
--- Ich hei\ss{}e Mira.\\
--- Wie hei\ss{}en Sie?\\
--- Ich hei\ss{}e Arun.\\
--- Freut mich.
\end{center}

Built the same way as the greetings: each piece taken apart first, then
assembled --- and, because German is English's sibling, most pieces are direct
English cousins. \emph{Practice lessons: \texttt{lessons/GE-C02-*.md}.}

\section{\emph{ich} --- ``I,'' the same ancient word as Latin \emph{ego}}
\label{lesson:ich}

\begin{sounds}
\emph{ikh}: the \emph{ch} is soft here (a light hiss at the front of the mouth,
the ``ich-sound''), not the throat-scrape of \emph{Nacht}.
\end{sounds}

\begin{cousinweb}
\textbf{ich} (``I'') $\leftarrow$ Proto-Germanic \emph{*ik}, from
Proto-Indo-European \emph{*e\'{g}} --- the \emph{same} deep root that gave
Latin \textbf{ego} (and English \textbf{ego}), Greek \emph{eg\=o}, and, worn
down hardest of all, English \textbf{I}. So German \emph{ich}, Latin
\emph{ego}, and English \emph{I} are three survivals of one Ice-Age word for
``I.''
\end{cousinweb}

\section{\emph{hei\ss{}en} --- ``to be called,'' the verb English almost lost}
\label{lesson:heissen}

\begin{sounds}
\emph{HY-ssen}. The \ss{} (\emph{Eszett}) is just a sharp ``ss''; \emph{ei} is
the vowel of English \emph{eye}. So \emph{hei\ss{}e} $=$ \emph{HY-ssuh}.
\end{sounds}

\begin{cousinweb}
\textbf{hei\ss{}en} (``to be called / to be named'') $\leftarrow$ Proto-Germanic
\emph{*haitan\=a}. English had this verb too --- \textbf{hight}, ``named,'' now
archaic (``a knight \emph{hight} Gawain'') --- and it survives in \textbf{behest}
(a command, something ``called for''). Where the Romance languages say ``I call
\emph{myself}'' with a reflexive, German uses \emph{hei\ss{}en} as a plain verb
of \emph{being} called --- no ``myself'' needed.
\end{cousinweb}

\begin{grammarlens}
\textbf{One verb, changing its ending.} \emph{hei\ss{}en} is the dictionary
(infinitive) form. For each person the ending changes: \emph{ich hei\ss{}\textbf{e}}
(I am called), \emph{du hei\ss{}\textbf{t}} (you are called), \emph{Sie
hei\ss{}\textbf{en}} (you [formal] are called). This ending-swap is the same
machine you'll see on every German verb.
\end{grammarlens}

\section{\emph{ich hei\ss{}e\ldots} --- ``my name is\ldots,'' literally ``I am called''}
\label{lesson:ich-heisse}

\emph{ich} (I) $+$ \emph{hei\ss{}e} (am called) $=$ \textbf{ich hei\ss{}e\ldots}
--- word for word ``I am called\ldots,'' and the everyday German for \textbf{``my
name is\ldots''} \emph{Ich hei\ss{}e Arun.} $=$ ``My name is Arun.''

\begin{culture}
German also has the literal ``my name is'': \textbf{mein Name ist\ldots} ---
where \textbf{Name} $\leftarrow$ Proto-Germanic \emph{*nam\=o}, the \emph{same}
root as English \textbf{name} and Latin \emph{n\=omen} (English \textbf{noun},
\textbf{nom}inate). It's perfectly correct and common. But \emph{ich hei\ss{}e}
is warmer and more idiomatic --- German prefers the \emph{verb} (``I am
called'') to the \emph{noun} phrase. (And note \emph{Name} is Capitalized:
German capitalizes \emph{every} noun.)
\end{culture}

\section{\emph{du} / \emph{Sie} --- ``you,'' familiar and formal}
\label{lesson:du-sie}

\begin{cousinweb}
\textbf{du} (familiar ``you'') $\leftarrow$ Proto-Germanic \emph{*\th{}\=u},
from PIE \emph{*t\=u} --- the \emph{same} root as English archaic \textbf{thou},
Latin \emph{t\=u}, and (a cousin) French \emph{tu}. English kept it only in
old prayers and Yorkshire; German uses it every day.
\end{cousinweb}

\begin{grammarlens}
\textbf{Two ``you''s --- and German's is the odd one out.} \emph{du} is for
friends, family, children, peers. For respect, German uses \textbf{Sie} ---
which is really the word for ``\emph{they}'' (\emph{sie}), promoted to a polite
``you'' and always \textbf{Capitalized} to mark the difference. So German is
polite by addressing you in the \emph{third person plural} (``might they\ldots'')
--- a third route to politeness: Spanish coined \emph{usted} (``your grace''),
French uses the second-person plural \emph{vous} (``you all''), German borrows
the third-person plural \emph{Sie} (``they''). All three dodge the blunt,
intimate ``you.'' When unsure, use \emph{Sie}.
\end{grammarlens}

\section{\emph{wie} --- ``how,'' cousin of English \emph{how}}
\label{lesson:wie}

\begin{cousinweb}
\textbf{wie} (``how'') $\leftarrow$ Proto-Germanic \emph{*hw\=\i}, from the PIE
interrogative root \emph{*k\ensuremath{^w}o-} --- the very root behind the
English \emph{wh-} words \textbf{how}, \textbf{what}, \textbf{who}, \textbf{why}
(and, through Latin, the \emph{qu-} of \emph{quo modo}). German \emph{wie} and
English \emph{how} are the same question-word, lightly reshaped.
\end{cousinweb}

\begin{grammarlens}
Like French (\emph{comment}), German asks a name with ``\emph{how},'' not
``what'': \emph{wie hei\ss{}en Sie?} is literally ``\emph{how} are you called?''
Keep the German logic --- the question is about the manner of your being called.
\end{grammarlens}

\section{\emph{wie hei\ss{}en Sie?} --- ``what's your name?''}
\label{lesson:wie-heissen-sie}

\emph{wie} (how) $+$ \emph{hei\ss{}en Sie} (``are you [formal] called'') $=$
literally ``\textbf{how are you called?}'' --- the polite ``what's your name?''

\begin{grammarlens}
\textbf{Word order: verb second.} German puts the verb in the \emph{second}
slot: \emph{Wie} (1) \emph{hei\ss{}en} (2) \emph{Sie}? The informal version just
swaps in \emph{du} and its ending: \emph{Wie hei\ss{}t du?} So the same verb,
\emph{hei\ss{}en}, runs the whole exchange --- \emph{ich hei\ss{}e} to give your
name, \emph{wie hei\ss{}en Sie?} to ask for theirs.
\end{grammarlens}

\section{\emph{freut mich} --- ``pleased to meet you,'' i.e. ``it gladdens me''}
\label{lesson:freut-mich}

\begin{sounds}
\emph{froyt mikh}: \emph{eu} is the vowel of English \emph{boy}; the final
\emph{ch} is the soft ich-sound again.
\end{sounds}

\begin{cousinweb}
\textbf{freut mich} is short for \emph{es freut mich (Sie kennenzulernen)},
``it \emph{gladdens} me (to get to know you).'' The verb \textbf{freuen} (``to
gladden'') is built on \textbf{froh} (``glad, joyful'') --- the same Germanic
\emph{froh}/\emph{froh}- that surfaces in the English \emph{Frohe Weihnachten}
you see at Christmas. To say \emph{freut mich} is to say, warmly, ``this makes
me glad.''
\end{cousinweb}

\begin{cousinweb}
\textbf{mich} (``me,'' the object form) $\leftarrow$ Proto-Germanic \emph{*mek},
from PIE \emph{*me-} --- the \emph{same} first-person root as English
\textbf{me}, \textbf{my}, \textbf{mine} (and the French \emph{me} of \emph{je
m'appelle}). So German \emph{mich}, English \emph{me}, and French \emph{me} are
three cousins of one ancient word for ``me.'' Just as English shifts \emph{I}
(subject) to \emph{me} (object), German shifts \emph{ich} to \emph{mich}.
\end{cousinweb}

\begin{grammarlens}
German is literally saying ``[it] gladdens \emph{me}'' --- the meeting is what
does the gladdening, and you are on the receiving end (the object, \emph{mich}).
Fuller forms: \emph{sehr erfreut} (``much pleased''), or the plain
\emph{angenehm} (``pleasant / agreeable''), both usable as ``pleased to meet
you.''
\end{grammarlens}

\section{The whole exchange}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
German & English \\
\midrule
\emph{Ich hei\ss{}e Mira.} & My name is Mira. \\
\emph{Wie hei\ss{}en Sie?} & What's your name? \\
\emph{Ich hei\ss{}e Arun.} & My name is Arun. \\
\emph{Freut mich.} & Pleased to meet you. \\
\bottomrule
\end{tabular}
\end{center}

Every piece was a Germanic cousin of English: \emph{ich} (I/ego), \emph{hei\ss{}en}
(hight/behest), \emph{wie} (how), \emph{freuen} (froh). And German's two habits
to keep: it names with a plain verb of \emph{being} called (no ``myself''), and
it is polite by the \emph{third} person (\emph{Sie} $=$ ``they''). Next
chapter: \emph{Wie geht's?} (``how's it going?'') and answering with \emph{gut}
again --- the responding cycle.
