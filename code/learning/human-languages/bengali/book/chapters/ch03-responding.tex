\chapter{How Are You?}
\label{ch:responding}

After the name comes the exchange everyone actually has --- and, for the first
time, Bengali brings back a verb ``to be'':

\begin{center}
\bn{তুমি কেমন আছো?} \quad(\emph{tumi k\=emon \=achho} --- ``How are you?'')\\
\bn{আমি ভালো আছি}, \bn{ধন্যবাদ।} \quad(\emph{\=ami bh\=alo \=achhi, dh\^{o}nnobad} --- ``I'm well, thank you.'')
\end{center}

\emph{Practice lessons: \texttt{lessons/BN-C03-*.md}.}

\section{\bn{কেমন} \emph{(k\=emon)} --- ``how,'' another of the k- questions}
\label{lesson:kemon}

\begin{cousinweb}
\bn{কেমন} (\emph{k\=emon}, ``how, of what kind'') is from the interrogative stem
\emph{ka-}, PIE *k\ensuremath{^w}o- --- the root behind English \textbf{wh-} and
Latin \emph{qu-}. Bengali's questions nearly all begin with \emph{k-}: \emph{ki,
ke, k\^{o}be, k\=oth\=ay, k\=emon}.
\end{cousinweb}

\section{\bn{তুমি কেমন আছো?} --- ``how are you?''}
\label{lesson:tumi-kemon-achho}

The new word is \bn{আছো} (\emph{\=achho}, ``are you''), from the verb \bn{আছা}
(\emph{\=achh\=a}, ``to be, exist, have'') $+$ the familiar-you ending. So
\bn{তুমি কেমন আছো?} is ``you how are?'', the verb last. Respectful: \bn{আপনি কেমন
আছেন?} (\emph{\=apni k\=emon \=achhen?}).

\begin{grammarlens}
\textbf{Now the copula returns.} Chapter 2's zero copula covers \emph{equations}
(\emph{\=am\=ar n\=am Arun}). But ``how are you?'' asks about a \emph{state} ---
how you \emph{exist} --- and for that Bengali uses a real verb, \bn{আছা}
(\emph{\=achh\=a}). No verb for ``my name is Arun''; a real verb for ``how are
you.''
\end{grammarlens}

\section{\bn{আমি} \emph{(\=ami)} --- ``I,'' cousin of English \emph{am}}
\label{lesson:ami}

\begin{cousinweb}
\bn{আমি} (\emph{\=ami}, ``I'') descends from Sanskrit \bn{অস্মি} (\emph{asmi}, ``I
am''), from PIE *h\textsubscript{1}es- --- the ``to be'' root that gives English
\textbf{am} and Latin \emph{sum}. So Bengali's word for ``I'' is, at the root, a
form of ``\textbf{am}'': ``I'' and ``am'' were once one word. Its possessive is
\emph{\=am\=ar} (``my'').
\end{cousinweb}

\section{\bn{ভালো} \emph{(bh\=alo)} --- ``well,'' and how to answer}
\label{lesson:bhalo}

\begin{cousinweb}
\bn{ভালো} (\emph{bh\=alo}, ``good, well, fine'') is the everyday Bengali ``good,''
likely from Sanskrit \bn{ভদ্র} (\emph{bhadra}, ``good, auspicious''). It fills the
language: \emph{bh\=alo l\=age} (``[I] like it''), \emph{bh\=alob\=as\=a}
(``love,'' literally ``good-dwelling''). The reply: \bn{আমি ভালো আছি} (\emph{\=ami
bh\=alo \=achhi}), ``I am well.''
\end{cousinweb}

\section{\bn{কোনো ব্যাপার না} \emph{(kono b\ae{}p\=ar n\=a)} --- ``no matter''}
\label{lesson:kono-bepar-na}

\begin{cousinweb}
\bn{কোনো ব্যাপার না} literally ``\textbf{[it's] no matter}'' --- \emph{kono}
(``any'') $+$ \bn{ব্যাপার} (\emph{b\ae{}p\=ar}, ``matter, affair'') $+$ \bn{না}
(\emph{n\=a}, ``not''). It serves as ``it's nothing / you're welcome.''
\emph{B\ae{}p\=ar} is \textbf{Sanskrit} (\emph{vy\=ap\=ara}, ``dealing''), and
\emph{n\=a} you met in Chapter 1 --- the ancient negative, PIE *ne, behind English
\textbf{no}, \textbf{not}, \textbf{none}.
\end{cousinweb}

farewells.
