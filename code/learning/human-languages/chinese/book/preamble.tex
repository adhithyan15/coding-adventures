\usepackage{fontspec}

% Pinyin marks tone with a diacritic over the vowel, and the caron vowels are the
% ones Latin Modern is least likely to carry precomposed. Compose them from the
% accent command instead, which works in any font, so a tone mark can never be
% silently dropped -- and a dropped tone mark here would not be a cosmetic loss,
% it would change the word.
\usepackage{newunicodechar}
\newunicodechar{ǎ}{\v{a}}
\newunicodechar{ě}{\v{e}}
\newunicodechar{ǐ}{\v{\i}}
\newunicodechar{ǒ}{\v{o}}
\newunicodechar{ǔ}{\v{u}}
\newunicodechar{ǖ}{\={\"u}}
\newunicodechar{ǘ}{\'{\"u}}
\newunicodechar{ǚ}{\v{\"u}}
\newunicodechar{ǜ}{\`{\"u}}
\newunicodechar{ḿ}{\'{m}}
\newunicodechar{ǹ}{\`{n}}
\newunicodechar{ǵ}{\'{g}}

% The vendored Chinese font. It is a fonttools SUBSET of Noto Sans SC covering
% exactly the characters in data/scripts/chinese.json -- the full family is ~17 MB,
% which is too much to vendor for a handful of greetings. Every requested shape is
% mapped to the same file so a bold or italic teaching context cannot fall back to
% a system font and emit a substitution warning.
% (No `Script=` key: Han needs no complex shaping, and naming a script tag the
% subset does not declare would only produce a fontspec warning for no gain.)
\newfontfamily\chinesefont[
  Path=../../_fonts/,
  BoldFont=NotoSansSC-Subset.ttf,
  ItalicFont=NotoSansSC-Subset.ttf,
  BoldItalicFont=NotoSansSC-Subset.ttf
]{NotoSansSC-Subset.ttf}

% Inline Chinese snippet within English text. The book generator emits \zh{...}
% around every run of Han-script characters (see core/book-generation.json).
\newcommand{\zh}[1]{{\chinesefont #1}}
\pdfstringdefDisableCommands{%
  \def\zh#1{#1}% Keep the characters in PDF bookmarks, drop the font switch.
  \def\chinesefont{}%
  % The tone accents above expand to accent commands, which hyperref cannot put
  % in a PDF string. Drop the accent rather than emit a warning per bookmark;
  % the bookmark text stays readable and the printed page keeps its tone marks.
  \def\v#1{#1}%
  \def\=#1{#1}%
}
