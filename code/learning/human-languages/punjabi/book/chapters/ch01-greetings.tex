\chapter{Greetings}
\label{ch:greetings}

The Punjabi greetings --- and, taught through them, how to read the Gurmukhi
script and the two faiths and two vocabularies that meet in Punjabi. Each lesson
introduces the letters its word needs. \emph{Practice lessons:
\texttt{lessons/PA-C01-*.md}.}

\section{\pa{ਸਤਿ ਸ੍ਰੀ ਅਕਾਲ} \emph{(sat sr\=\i\ ak\=al)} --- the Sikh greeting}
\label{lesson:sat-sri-akal}

\begin{sounds}
\emph{(If you already read Gurmukhi, skim this.)} Gurmukhi runs \textbf{left to
right} under a top line, and every consonant carries a built-in ``a.'' \pa{ਸ}
sa, \pa{ਤ} ta, \pa{ਰ} ra, \pa{ਕ} ka, \pa{ਲ} la. A \textbf{vowel sign} changes
the vowel (\pa{ਾ} \=a, \pa{ਿ} i, \pa{ੀ} \=\i); a word-initial vowel rides a
carrier letter --- \pa{ਅ} is the ``a''-carrier. So \pa{ਅਕਾਲ} $=$ \emph{a-k\=a-l},
\emph{ak\=al}.
\end{sounds}

\begin{cousinweb}
The name \textbf{Gurmukhi} means ``from the mouth of the \emph{Guru}'' (\pa{ਗੁਰੂ}
\emph{gur\=u} $+$ \pa{ਮੁਖ} \emph{mukh}, ``mouth/face''); the script was
standardised by Guru Angad, the second Sikh Guru, to write the scriptures. The
greeting itself is a small creed: \pa{ਸਤਿ} (\emph{sat}, ``truth,'' from Sanskrit
\emph{satya} --- the same ``being/truth'' root behind English \emph{sooth} and
\emph{soothsayer}) $+$ \pa{ਸ੍ਰੀ} (\emph{sr\=\i}, an honorific ``revered,'' from
Sanskrit \emph{\'{s}r\=\i}, ``glory'') $+$ \pa{ਅਕਾਲ} (\emph{ak\=al},
``\textbf{timeless}'' $=$ \emph{a-} ``not'' $+$ \emph{k\=al} ``time/death,'' from
Sanskrit \emph{k\=ala}). \emph{Ak\=al} is a name of God in Sikhism, the Timeless
One --- so the greeting declares ``God is the eternal Truth.''
\end{cousinweb}

\begin{culture}
\emph{Sat sr\=\i\ ak\=al} is the greeting of Sikhs --- said with folded hands, and
serving as both hello and goodbye. It is a statement of faith as much as a
greeting, which is why it carries such weight. (A Sikh will often answer it by
repeating it back.)
\end{culture}

\section{\pa{ਨਮਸਤੇ} \emph{(namaste)} --- the shared greeting}
\label{lesson:namaste}

\begin{sounds}
New: \pa{ਨ} \emph{na}, \pa{ਮ} \emph{ma}, and \pa{ਤੇ} \emph{te} (\pa{ਤ} $+$ the
\emph{e}-sign \pa{ੇ}). With \pa{ਸ} from before, read \pa{ਨਮਸਤੇ} as
\emph{na\,·\,ma\,·\,s\,·\,te}.
\end{sounds}

\begin{cousinweb}
\pa{ਨਮਸਤੇ} (\emph{namaste}) $=$ Sanskrit \emph{namas} (``a bow'') $+$ \emph{te}
(``to you'') --- ``I bow to you,'' the very same word as Hindi \emph{namaste}.
Punjab is home to many communities; \emph{namaste} is the more general,
pan-Indian greeting, used across faiths, where \emph{sat sr\=\i\ ak\=al} is
specifically Sikh.
\end{cousinweb}

\section{\pa{ਧੰਨਵਾਦ} \emph{(dhannav\=ad)} --- ``thank you,'' the Sanskrit way}
\label{lesson:dhanvaad}

\begin{sounds}
New: \pa{ਧ} \emph{dha} (breathy \emph{d}), \pa{ਵ} \emph{va}, \pa{ਦ} \emph{da},
and the nasal dot \pa{ੰ} (\emph{\d{t}ippi}) on \pa{ਧੰ} \emph{dhaṁ}. Read
\pa{ਧੰਨਵਾਦ} as \emph{dhaṁ\,·\,na\,·\,vā\,·\,d}.
\end{sounds}

\begin{cousinweb}
\pa{ਧੰਨਵਾਦ} (\emph{dhannav\=ad}) $=$ Sanskrit \emph{dhanya} (``worthy'') $+$
\emph{v\=ada} (``a saying'') --- the same Sanskrit ``thank you'' behind Hindi
\emph{dhanyav\=ad} and Marathi's. It is the formal, Sanskritic thanks.
\end{cousinweb}

\section{\pa{ਸ਼ੁਕਰੀਆ} \emph{(shukr\=\i\=a)} --- ``thank you,'' the Persian way}
\label{lesson:shukriya}

\begin{sounds}
New: \pa{ਸ਼} \emph{sha} --- the letter \pa{ਸ} (\emph{sa}) with a dot beneath, a
\emph{pair bindi} that marks Perso-Arabic sounds. With \pa{ੁ} (the \emph{u}-sign)
and \pa{ੀ}, read \pa{ਸ਼ੁਕਰੀਆ} as \emph{shu\,·\,ka\,·\,rī\,·\,ā}.
\end{sounds}

\begin{cousinweb}
\pa{ਸ਼ੁਕਰੀਆ} (\emph{shukr\=\i\=a}) is \emph{not} Sanskrit --- it came through
Persian from \textbf{Arabic \emph{shukr}} (``gratitude''), the \emph{same}
Semitic root behind Arabic \emph{shukran}. Punjab sat for centuries on the road
between Persia and India, so Punjabi (like Urdu and Hindi) carries \textbf{two
vocabularies}: Sanskrit (\emph{dhannav\=ad}) and Perso-Arabic
(\emph{shukr\=\i\=a}). The dot under \pa{ਸ਼} is the script's own record of that
borrowing.
\end{cousinweb}

\section{\pa{ਹਾਂ} / \pa{ਨਹੀਂ} \emph{(h\=a\d{n} / nah\=\i\d{n})} --- ``yes'' and ``no''}
\label{lesson:han-nahin}

\begin{sounds}
New: \pa{ਹ} \emph{ha}, and the nasal sign \pa{ਂ} (\emph{bindi}). \pa{ਹਾਂ}
(\emph{h\=a\d{n}}) and \pa{ਨਹੀਂ} (\emph{nah\=\i\d{n}}) --- both end in a nasalised
vowel.
\end{sounds}

\begin{cousinweb}
\pa{ਹਾਂ} (\emph{h\=a\d{n}}, ``yes'') and \pa{ਨਹੀਂ} (\emph{nah\=\i\d{n}}, ``no'').
The ``no'' sits on the ancient negative \emph{na-} --- the same Proto-Indo-European
\emph{*ne} behind English \textbf{no}, \textbf{not}, Latin \emph{non}, and Hindi
\emph{nah\=\i\d{n}}. Punjabi's ``no'' and English's ``no'' are distant cousins.
\end{cousinweb}

\section{Recap}
\label{PA-C01-practice}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Punjabi & & Meaning \\
\midrule
\pa{ਸਤਿ ਸ੍ਰੀ ਅਕਾਲ} & \emph{sat sr\=\i\ ak\=al} & hello / goodbye (Sikh) \\
\pa{ਨਮਸਤੇ} & \emph{namaste} & hello (general) \\
\pa{ਧੰਨਵਾਦ} & \emph{dhannav\=ad} & thank you (Sanskritic) \\
\pa{ਸ਼ੁਕਰੀਆ} & \emph{shukr\=\i\=a} & thank you (Persian) \\
\pa{ਹਾਂ} / \pa{ਨਹੀਂ} & \emph{h\=a\d{n} / nah\=\i\d{n}} & yes / no \\
\bottomrule
\end{tabular}
\end{center}

Two threads run through even these first words: the \textbf{Gurmukhi} script,
``from the Guru's mouth,'' carrying a Sikh greeting that is a small creed; and
Punjabi's \textbf{two vocabularies} --- Sanskrit (\emph{dhannav\=ad}) and
Perso-Arabic (\emph{shukr\=\i\=a}), the script itself marking the Persian words
with a dot. Next chapter: introducing yourself --- \emph{mer\=a n\=a\d{n}\ldots}
(``my name\ldots'') --- and Punjabi's \emph{t\=u\d{n}} / \emph{tus\=\i\d{n}}
(familiar / respectful ``you'').
