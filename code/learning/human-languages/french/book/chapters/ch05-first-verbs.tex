\chapter{The First Verbs}
\label{ch:first-verbs}

Until now you have assembled fixed phrases. This chapter hands you a
\emph{pattern} instead: the regular \emph{-er} verbs, the largest verb family in
French by a wide margin. Learn the one template and thousands of verbs come with
it --- and with it you build your first sentence that nobody handed you
pre-made. \emph{Practice lessons: \texttt{lessons/FR-C05-*.md}.}

\section{\emph{parler} --- ``to speak,'' the model \emph{-er} verb}
\label{lesson:parler}

\begin{sounds}
The infinitive ending \emph{-er} is a clean ``ay'': \emph{parler} $=$
\emph{par-LAY} (the \emph{r} of that ending is silent). What is left when you
strip the ending is the \textbf{stem}, \emph{parl-}.
\end{sounds}

\begin{cousinweb}
\textbf{parler} (``to speak, to talk'') $\leftarrow$ Late Latin
\emph{parabol\=are}, ``to tell parables, to discourse,'' from \emph{parabola}
(``a comparison, a speech''), itself from Greek \emph{parabol\=e}, ``a
throwing-beside.'' So to \emph{speak}, in French, is at root to \emph{tell
parables}.\\[3pt]
That root fans out into English: \textbf{parable}, the story-comparison itself;
\textbf{parole}, a spoken word of honour (as a prisoner's parole);
\textbf{palaver}, a long talk; \textbf{parley}, to talk terms; and even
\textbf{parliament}, a talking-place.
\end{cousinweb}

\begin{grammarlens}
\textbf{The \emph{-er} present tense.} Drop \emph{-er} from the infinitive, then
add the ending that belongs to the person. (\emph{je} $=$ I, \emph{tu} $=$ you
familiar, \emph{il / elle} $=$ he / she, \emph{nous} $=$ we, \emph{vous} $=$ you
formal or plural, \emph{ils / elles} $=$ they.)

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Person & Ending & \emph{parler} & Sounds like \\
\midrule
\emph{je} & \emph{-e} & \emph{je parle} & \emph{parl} \\
\emph{tu} & \emph{-es} & \emph{tu parles} & \emph{parl} \\
\emph{il / elle} & \emph{-e} & \emph{il parle} & \emph{parl} \\
\emph{nous} & \emph{-ons} & \emph{nous parlons} & \emph{par-LON} \\
\emph{vous} & \emph{-ez} & \emph{vous parlez} & \emph{par-LAY} \\
\emph{ils / elles} & \emph{-ent} & \emph{ils parlent} & \emph{parl} \\
\bottomrule
\end{tabular}
\end{center}

Here is the great French surprise: \textbf{\emph{-e}, \emph{-es} and
\emph{-ent} are all silent}. \emph{je parle}, \emph{tu parles} and \emph{ils
parlent} \emph{sound identical} --- just \emph{parl}. Only the \emph{nous} form
(\emph{-ons}) and the \emph{vous} form (\emph{-ez}) are audible.
\end{grammarlens}

\begin{grammarlens}
\textbf{Why French keeps its pronouns.} That silence has a consequence which
shapes the whole language. In Spanish, the ending alone tells you who is
speaking --- \emph{hablo} (``I speak'') against \emph{hablas} (``you speak'')
--- so the pronoun can be dropped. In French your ear cannot hear any
difference between \emph{parle}, \emph{parles} and \emph{parlent}, so the
\textbf{pronoun must stay} to carry the person. \emph{Je} and \emph{tu} are not
optional decoration in French; they are the only thing telling you who is doing
the speaking.
\end{grammarlens}

\section{\emph{habiter} --- ``to live,'' and it is about \emph{having a place}}
\label{lesson:habiter}

\begin{sounds}
The \textbf{h} is silent --- French \emph{h} is never pronounced. So
\emph{habiter} $=$ \emph{a-bee-TAY}, with the same \emph{-er} $=$ ``ay'' ending
as \emph{parler}.
\end{sounds}

\begin{cousinweb}
\textbf{habiter} (``to live, to dwell'') $\leftarrow$ Latin \emph{habit\=are},
``to dwell, to keep on having [a place]'' --- the \textbf{frequentative} of
\emph{hab\=ere}, ``to have.'' A frequentative is a verb of repeated action, so
to \emph{dwell} somewhere is, at root, to \emph{keep having} it as your
place.\\[3pt]
The root is thoroughly at home in English: \textbf{habitat}, the place a
creature dwells; \textbf{inhabit}, \textbf{inhabit}ant, \textbf{habit}ation. And
through \emph{hab\=ere} (``to have'') come \textbf{habit}, what you keep having,
and \textbf{able} / \textbf{ability}, what you have it in you to do.
\end{cousinweb}

\begin{grammarlens}
\textbf{The same template, silent trap and all.} Drop \emph{-er}, add the
endings --- identical to \emph{parler}:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Person & \emph{habiter} & Sounds like \\
\midrule
\emph{je} & \emph{j'habite} & \emph{a-BEET} \\
\emph{tu} & \emph{tu habites} & \emph{a-BEET} \\
\emph{il / elle} & \emph{il habite} & \emph{a-BEET} \\
\emph{nous} & \emph{nous habitons} & \emph{a-bee-TON} \\
\emph{vous} & \emph{vous habitez} & \emph{a-bee-TAY} \\
\emph{ils / elles} & \emph{ils habitent} & \emph{a-BEET} \\
\bottomrule
\end{tabular}
\end{center}

Note \emph{j'habite}: \emph{je} elides to \emph{j'} before the vowel of
\emph{habite}. Because the \emph{h} is silent, the word \emph{counts} as
beginning with a vowel. With this verb you can answer an \emph{o\`{u}}
(``where'') question: \emph{J'habite \`{a} Paris.} --- ``I live in Paris.''
\end{grammarlens}

\section{\emph{travailler} --- ``to work,'' and the torture behind \emph{travel}}
\label{lesson:travailler}

\begin{sounds}
The letters \emph{-ill-} carry a \emph{y} glide: \emph{travailler} $=$
\emph{tra-va-YAY}, with the same \emph{-er} $=$ ``ay'' ending. The conjugated
\emph{je travaille} is \emph{tra-VYE}.
\end{sounds}

\begin{cousinweb}
\textbf{travailler} (``to work'') $\leftarrow$ Vulgar Latin \emph{tripali\=are},
``to torture,'' from \emph{tripalium}, a Roman instrument of torture made of
\textbf{three stakes} (\emph{tri-} ``three'' $+$ \emph{p\=alus} ``stake'' $\to$
English \textbf{pale}, im\textbf{pale}).\\[3pt]
The slide of meaning is grim and perfectly logical: \emph{torture} $\to$
\emph{painful toil} $\to$ \emph{work}. Two English words came down that same
road: \textbf{travail}, painful and laborious effort; and \textbf{travel},
because journeying was once such an ordeal that the word for toil became the
word for taking a trip. Etymologically, you are tortured every time you travel.
\end{cousinweb}

\begin{culture}
Spanish \emph{trabajar} (``to work'') comes from the very same
\emph{tripali\=are} --- a perfect cross-language twin, torture and all. The
middle of the word went one way in each language: the \emph{-al-} stayed in
French (\emph{travail}, \emph{travailler}) and softened to \emph{-aj-} in
Spanish (\emph{trabajo}, \emph{trabajar}).
\end{culture}

\begin{grammarlens}
\textbf{Nothing new --- and that is the point.} One template, every \emph{-er}
verb:

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Person & \emph{travailler} & Sounds like \\
\midrule
\emph{je} & \emph{je travaille} & \emph{tra-VYE} \\
\emph{tu} & \emph{tu travailles} & \emph{tra-VYE} \\
\emph{il / elle} & \emph{il travaille} & \emph{tra-VYE} \\
\emph{nous} & \emph{nous travaillons} & \emph{tra-va-YON} \\
\emph{vous} & \emph{vous travaillez} & \emph{tra-va-YAY} \\
\emph{ils / elles} & \emph{ils travaillent} & \emph{tra-VYE} \\
\bottomrule
\end{tabular}
\end{center}

\noindent \emph{Je travaille \`{a} Paris.} --- ``I work in Paris.''
\end{grammarlens}

\section{\emph{Je parle fran\c{c}ais} --- your first real sentence}
\label{lesson:je-parle-francais}

\emph{Je} (I, \S\ref{lesson:je}) $+$ \emph{parle} (speak, from \emph{parler},
\S\ref{lesson:parler}) $+$ \emph{fran\c{c}ais} (French) $=$ \textbf{Je parle
fran\c{c}ais.} A complete, grammatical French sentence --- and the first one you
assembled yourself rather than learned whole.

\begin{sounds}
\emph{fran\c{c}ais} $=$ \emph{frahn-SEH}. The cedilla under the \emph{\c{c}}
forces a soft \emph{s} sound where a plain \emph{c} would have been hard; the
opening \emph{fran-} is nasal, and the final \emph{-ais} is a plain \emph{eh}.
\end{sounds}

\begin{cousinweb}
\textbf{fran\c{c}ais} (``French'' --- both the language and a French person)
$\leftarrow$ \emph{Francia}, ``land of the \textbf{Franks},'' the Germanic
people who settled Roman Gaul and gave France its name. And here is the twist:
the Franks' own name meant \textbf{``free.''} That is why English
\textbf{frank} and \textbf{frankly} mean open and unguarded, and why the old
French coin was called a \emph{franc}.
\end{cousinweb}

\begin{grammarlens}
\textbf{Two rules hiding in four words.} First, \textbf{you keep the
\emph{je}}. Spanish drops its pronoun --- \emph{hablo espa\~{n}ol} says ``I
speak Spanish'' with no word for ``I,'' because the ending \emph{-o} carries the
person. French cannot do that: \emph{parle}, \emph{parles} and \emph{parlent}
all sound alike (\S\ref{lesson:parler}), so the pronoun must stay. This is the
single biggest structural difference between French and its Iberian cousins.\\[3pt]
Second, \textbf{no article on the language} here: \emph{je parle fran\c{c}ais},
not \emph{je parle le fran\c{c}ais}.
\end{grammarlens}

Swap the pieces and the pattern generates new sentences: \emph{J'habite \`{a}
Paris.} (``I live in Paris.'') and \emph{Tu parles fran\c{c}ais?} (``Do you
speak French?'' --- no extra words needed, just raise the tone at the end).

\section{Making sentences with \emph{-er} verbs}
\label{lesson:ch5-practice}

No new words. For the first time you are not reciting phrases --- you are
\emph{building} them from a pattern. Run each verb through \emph{je / tu / il}
and listen: nothing changes.

\begin{center}
\begin{tabular}{@{}llll@{}}
\toprule
Verb & \emph{je} & \emph{tu} & \emph{il} \\
\midrule
\emph{parler} & \emph{je parle} & \emph{tu parles} & \emph{il parle} \\
\emph{habiter} & \emph{j'habite} & \emph{tu habites} & \emph{il habite} \\
\emph{travailler} & \emph{je travaille} & \emph{tu travailles} & \emph{il travaille} \\
\bottomrule
\end{tabular}
\end{center}

Every singular form --- and the \emph{ils} form too --- \emph{sounds} the same,
because \emph{-e}, \emph{-es} and \emph{-ent} are silent. That is exactly why
\emph{je}, \emph{tu} and \emph{il} stay put: they carry the person your ear
cannot hear.

\medskip
\noindent Now say something real:

\begin{center}\itshape
--- Tu parles fran\c{c}ais? --- Oui, je parle un peu.\\
--- O\`{u} habites-tu? --- J'habite \`{a} Lyon. Et je travaille \`{a} Paris.\\
--- Merci! --- De rien.
\end{center}

\noindent (\emph{oui} $=$ ``yes''; \emph{un peu} $=$ ``a little''; \emph{o\`{u}}
$=$ ``where''; and \emph{merci} / \emph{de rien} come straight back from
Chapter~\ref{ch:how-are-you}.)

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
French & English \\
\midrule
\emph{Je parle fran\c{c}ais.} & I speak French. \\
\emph{Tu parles fran\c{c}ais?} & Do you speak French? \\
\emph{Oui, je parle un peu.} & Yes, I speak a little. \\
\emph{J'habite \`{a} Paris.} & I live in Paris. \\
\emph{Je travaille \`{a} Lyon.} & I work in Lyon. \\
\bottomrule
\end{tabular}
\end{center}

What this chapter built: the \textbf{regular \emph{-er} present tense} --- drop
\emph{-er}, add \emph{-e / -es / -e / -ons / -ez / -ent}, where the singular
endings and the \emph{ils} ending are silent, which is why the subject pronoun
is mandatory. Three verbs to drill it: \emph{parler} (from
\emph{parabol\=are} --- parable, parole), \emph{habiter} (from
\emph{habit\=are} --- habitat, inhabit), \emph{travailler} (from
\emph{tripalium} --- travail, travel; twin of Spanish \emph{trabajar}). And
\emph{Je parle fran\c{c}ais}, your first self-assembled sentence, with
\emph{fran\c{c}ais} from \emph{Francia}, the land of the Franks, whose name
meant ``free.'' French now moves in real sentences.
