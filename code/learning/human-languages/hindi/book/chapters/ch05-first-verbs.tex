\chapter{The First Verbs}
\label{ch:first-verbs}

Until now, ``to be.'' Now verbs that \emph{act} --- and sentences that move:

\begin{center}
\hi{मैं हिंदी बोलता हूँ।} \quad(\emph{mai\d{m} hind\=\i\ bolt\=a h\=u\d{m}} --- ``I speak Hindi.'')
\end{center}

\emph{Practice lessons: \texttt{lessons/HI-C05-*.md}.}

\section{\hi{बोलना} \emph{(boln\=a)} --- ``to speak,'' your first full verb}
\label{lesson:bolna}

\begin{sounds}
\hi{ब} (\emph{ba}) $+$ \hi{ो} (the \emph{o}-m\=atr\=a) $\to$ \hi{बो} (\emph{bo});
\hi{ल} (\emph{la}); \hi{ना} (\emph{n\=a}). Read \hi{बो·ल·ना} $\to$ \emph{boln\=a}.
\end{sounds}

\begin{cousinweb}
\hi{बोलना} (\emph{boln\=a}, ``to speak'') is a native Indo-Aryan verb (Prakrit
\emph{bolla-}, ``to utter''). Its ending \hi{-ना} (\emph{-n\=a}) is the point:
\textbf{every} Hindi infinitive ends in \emph{-n\=a} --- \emph{bol-n\=a} ``to
speak,'' \emph{cal-n\=a} ``to move,'' \emph{mil-n\=a} ``to meet.'' Strip the
\emph{-n\=a} and you have the \textbf{stem} (\emph{bol-}) that takes every other
ending.
\end{cousinweb}

\begin{grammarlens}
\textbf{Stem $+$ ending.} The stem \emph{bol-} holds the action; endings pin down
who and when: \emph{bol-t\=a} (speaking, m.), \emph{bol-t\=\i} (f.), \emph{bol-o}
(speak!, familiar). Every tense is built by adding to a stem like this.
\end{grammarlens}

\section{\hi{मैं हिंदी बोलता हूँ} \emph{(mai\d{m} hind\=\i\ bolt\=a h\=u\d{m})} --- ``I speak Hindi''}
\label{lesson:main-hindi-bolta-hun}

\hi{मैं} (I) $+$ \hi{हिंदी} (Hindi, the object) $+$ \hi{बोलता} (speak, m.) $+$
\hi{हूँ} (am) $=$ literally ``\textbf{I Hindi speaking am}'' --- verb-frame last,
as always.

\begin{cousinweb}
The name \hi{हिंदी} (\emph{hind\=\i}) is a traveller: from \emph{sindhu} (the
river Indus) $\to$ Persian \emph{hind} (the land beyond it) $\to$ \emph{hind\=\i}
(``of Hind''). The same \emph{sindhu} gave Greek \emph{Indos} $\to$ English
\textbf{India}, \textbf{Indus}, \textbf{Hindu}. The language is named, at two
removes, after a river.
\end{cousinweb}

\begin{grammarlens}
\textbf{The present habitual $=$ stem $+$ \emph{-t\=a/-t\=\i} $+$ ``to be.''}
\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
 & Hindi & \\
\midrule
a man & \hi{मैं हिंदी बोलता हूँ} & \emph{bolt\=a h\=u\d{m}} \\
a woman & \hi{मैं हिंदी बोलती हूँ} & \emph{bolt\=\i\ h\=u\d{m}} \\
you (resp.) & \hi{आप हिंदी बोलते हैं} & \emph{bolte hai\d{m}} \\
\bottomrule
\end{tabular}
\end{center}
The \emph{-t\=a/-t\=\i/-te} agrees in gender and number; \emph{h\=u\d{m}/hai/hai\d{m}}
agrees in person. Two agreements, one small sentence.
\end{grammarlens}

\section{\hi{रहना} \emph{(rahn\=a)} --- ``to live, to stay''}
\label{lesson:rahna}

\begin{sounds}
\hi{र} (\emph{ra}) $+$ \hi{ह} (\emph{ha}) $+$ \hi{ना} (\emph{n\=a}) $\to$
\hi{रहना}. The familiar infinitive \hi{-ना}.
\end{sounds}

\begin{cousinweb}
\hi{रहना} (\emph{rahn\=a}, ``to live, remain, stay''), from Sanskrit \emph{rah-}
(``to remain''), covers both ``reside'' and ``keep on'': \emph{mai\d{m} Dill\=\i\
me\d{m} raht\=a h\=u\d{m}} (``I live in Delhi''), \emph{\d{t}h\=\i k rahn\=a}
(``stay well'').
\end{cousinweb}

\begin{grammarlens}
\textbf{``In'' comes after the noun.} \hi{मैं दिल्ली में रहता हूँ} --- the word
\hi{में} (\emph{me\d{m}}, ``in'') sits \emph{after} Delhi. Hindi uses
\textbf{post}positions where English uses prepositions: ``Delhi-in I live,'' verb
last.
\end{grammarlens}

\section{\hi{करना} \emph{(karn\=a)} --- ``to do,'' the most useful verb in Hindi}
\label{lesson:karna}

\begin{sounds}
\hi{क} (\emph{ka}) $+$ \hi{र} (\emph{ra}) $+$ \hi{ना} (\emph{n\=a}) $\to$
\hi{करना}.
\end{sounds}

\begin{cousinweb}
\hi{करना} (\emph{karn\=a}, ``to do, make'') is from the Sanskrit root
\emph{k\d{r}} (``to do, make''), PIE *k\ensuremath{^w}er- --- one of the most productive roots
in Sanskrit. Its family is already all around you: \emph{karma} (``a thing
\emph{done}''), \emph{namask\=ar} (``the \emph{making} of a bow,'' Chapter 1),
and the name \emph{Sanskrit} itself (\emph{sa\d{m}sk\d{r}ta}, ``put together,
perfected'').
\end{cousinweb}

\begin{grammarlens}
\textbf{\hi{करना} makes verbs out of anything.} Noun $+$ \emph{karn\=a}:
\emph{k\=am karn\=a} (``work-do'' $=$ to work), \emph{b\=at karn\=a} (``talk-do''
$=$ to converse), \emph{inti\-z\=ar karn\=a} (``to wait''). So \emph{mai\d{m} k\=am
kart\=a h\=u\d{m}} $=$ ``I work.''
\end{grammarlens}

\section{Sentences about yourself}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Hindi & English \\
\midrule
\hi{मैं हिंदी बोलता/बोलती हूँ।} & I speak Hindi. \\
\hi{मैं दिल्ली में रहता/रहती हूँ।} & I live in Delhi. \\
\hi{मैं काम करता/करती हूँ।} & I work. \\
\bottomrule
\end{tabular}
\end{center}

Three verbs, one engine: a stem, the \emph{-t\=a/-t\=\i/-te} that agrees in
gender/number, and the \emph{h\=u\d{m}/hai/hai\d{m}} that agrees in person --- the
verb always last, ``in'' always after its noun. Roots banked: \emph{boln\=a}
(Prakrit \emph{bolla-}), \emph{rahn\=a} (\emph{rah-} ``remain''), \emph{karn\=a}
(\emph{k\d{r}} --- the root of \emph{karma}, \emph{namask\=ar}, \emph{Sanskrit}),
and \emph{hind\=\i} (\emph{sindhu}, the Indus). From here, Hindi's sentences only
grow.
