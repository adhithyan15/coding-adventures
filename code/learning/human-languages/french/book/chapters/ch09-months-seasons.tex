\chapter{Months and Seasons}
\label{ch:months-seasons}

The French calendar is a procession of Roman gods and emperors, carried almost
unchanged out of Latin --- and the four seasons are four small Latin poems about
heat, sleep, and first things. \emph{Practice lessons:
\nolinkurl{lessons/FR-C09-*.md}.}

\section{\emph{les mois} --- the calendar's gods and Caesars}
\label{lesson:mois}

The twelve French months are a \textbf{procession of Roman gods and emperors},
almost unchanged from Latin. Two of them pay off twice: \emph{mars} is the
war-god who also names Tuesday, and \emph{septembre}--\emph{d\'{e}cembre} are
still the Latin numbers seven to ten.

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
French & $\leftarrow$ Latin & who / what \\
\midrule
\emph{janvier} & \emph{Januarius} & \textbf{Janus}, two-faced god of doors and beginnings \\
\emph{f\'{e}vrier} & \emph{Februarius} & \emph{Februa}, the Roman \textbf{purification} festival \\
\emph{mars} & \emph{Martius} & \textbf{Mars}, the war-god \\
\emph{avril} & \emph{Aprilis} & perhaps \emph{aper\={\i}re}, ``to \textbf{open}'' (buds opening) \\
\emph{mai} & \emph{Maius} & \textbf{Maia}, goddess of growth \\
\emph{juin} & \emph{Junius} & \textbf{Juno}, queen of the gods \\
\emph{juillet} & \emph{Julius} & \textbf{Julius Caesar} (he inserted this month) \\
\emph{ao\^{u}t} & \emph{Augustus} & the emperor \textbf{Augustus} (he inserted this one) \\
\emph{septembre} & \emph{september} & ``\textbf{7th}'' (\emph{septem}) \\
\emph{octobre} & \emph{october} & ``\textbf{8th}'' (\emph{octo}) \\
\emph{novembre} & \emph{november} & ``\textbf{9th}'' (\emph{novem}) \\
\emph{d\'{e}cembre} & \emph{december} & ``\textbf{10th}'' (\emph{decem}) \\
\bottomrule
\end{tabular}
\end{center}

\begin{cousinweb}
\textbf{mars} $=$ \textbf{Mars} $=$ \textbf{mardi}. The month and the weekday
honour the \emph{same} war-god --- \emph{Martius} (his month) and \emph{Martis
di\=es} (his day). Spot one, and you know the other.\\[3pt]
\textbf{The 7--10 months.} \emph{Septembre} through \emph{d\'{e}cembre} still
carry \emph{septem, octo, novem, decem} --- because the old Roman year began in
\textbf{March}, so September really \emph{was} the seventh month. Then
\textbf{Julius} and \textbf{Augustus} shoved two months in (\emph{juillet},
\emph{ao\^{u}t}), the counting slid out of place --- but the names stuck.
\end{cousinweb}

\begin{culture}
Two of the twelve are named for \emph{people} rather than gods:
\emph{juillet} for Julius Caesar and \emph{ao\^{u}t} for the emperor Augustus.
Each man put his own month into the calendar, and two thousand years later
France still says their names every summer.
\end{culture}

\section{\emph{les saisons} --- the four seasons}
\label{lesson:saisons}

Four words for the turning year: \textbf{le printemps}, \textbf{l'\'{e}t\'{e}},
\textbf{l'automne}, \textbf{l'hiver}. Each traces to Latin, and the first one is
a small poem --- \emph{printemps} means, almost literally, ``the first time.''

\begin{center}
\begin{tabularx}{\linewidth}{@{}ll>{\raggedright\arraybackslash}X@{}}
\toprule
French & $\leftarrow$ Latin & literally \\
\midrule
\emph{le printemps} (spring) & \emph{pr\={\i}mum tempus} & ``the \textbf{first time / prime season}'' \\
\emph{l'\'{e}t\'{e}} (summer) & \emph{aest\=as / aest\=atem} & ``heat, the hot season'' \\
\emph{l'automne} (autumn) & \emph{autumnus} & ``autumn'' --- the word English borrowed \\
\emph{l'hiver} (winter) & \emph{h\={\i}bernum (tempus)} & ``the \textbf{wintry} season'' \\
\bottomrule
\end{tabularx}
\end{center}

\begin{cousinweb}
\textbf{printemps} $=$ \emph{prime} $+$ \emph{temps} (``time, weather''): the
year's \emph{first} season, its ``prime time.'' Older French used
\emph{primev\`{e}re} --- the same \emph{prima vera} (``first spring'') that
Italian keeps as \emph{primavera}.\\[3pt]
\textbf{\'{e}t\'{e}} $\leftarrow$ \emph{aest\=as}, ``heat'' --- summer as the
hot season; English \textbf{estival} (also spelled \textbf{aestival}) means ``of
summer.''\\[3pt]
\textbf{automne} $\leftarrow$ \emph{autumnus}: the \emph{m} stayed in the
spelling, the \emph{n} went silent.\\[3pt]
\textbf{hiver} $\leftarrow$ \emph{h\={\i}bernum}, ``wintry'' --- and English
\textbf{hibernate} is literally ``to spend the winter.'' Same root: one a
season, one a bear's sleep.
\end{cousinweb}

\begin{grammarlens}
\textbf{Saying ``in'' a season.} Spring is the odd one out: \emph{au printemps},
but \emph{en \'{e}t\'{e}} / \emph{en automne} / \emph{en hiver}. Only spring
takes \emph{au}.
\end{grammarlens}
