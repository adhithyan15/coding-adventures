\chapter{Farewells}
\label{ch:farewells}

Chapter 1 gave you the grave, formal \hi{अलविदा} (\emph{alvid\=a}). Real days end
more warmly --- with a promise to meet again:

\begin{center}
\hi{फिर मिलेंगे!} \quad(\emph{phir milenge} --- ``We'll meet again!'')
\end{center}

\emph{Practice lessons: \texttt{lessons/HI-C04-*.md}.}

\section{\hi{फिर} \emph{(phir)} --- ``again''}
\label{lesson:phir}

\begin{sounds}
\hi{फ} (\emph{pha}, an aspirated \emph{p}) $+$ \hi{ि} (the \emph{i}-m\=atr\=a,
written before but read after) $+$ \hi{र} (\emph{ra}) $\to$ \hi{फिर}.
\end{sounds}

\begin{cousinweb}
\hi{फिर} (\emph{phir}) means ``again, then, afterwards'' --- a native Indo-Aryan
adverb with no tidy English cognate, but a familiar \emph{job}: it turns a parting
into a promise. English ``see you \emph{again}''; Hindi \emph{phir milenge}.
\end{cousinweb}

\section{\hi{मिलेंगे} \emph{(milenge)} --- ``(we) will meet''}
\label{lesson:milenge}

\begin{sounds}
\hi{मि} (\emph{mi}) $+$ \hi{ले} (\emph{le}) $+$ the \emph{anusv\=ara} $+$ \hi{गे}
(\emph{ge}) $\to$ \hi{मिलेंगे} (\emph{milenge}).
\end{sounds}

\begin{cousinweb}
\hi{मिलेंगे} (\emph{milenge}, ``we will meet'') is the future of \hi{मिलना}
(\emph{miln\=a}, ``to meet, to be found''), from Sanskrit \emph{mil-} (``to join,
unite'') --- the root of \emph{milan} (``a meeting, union''). The ending
\hi{-एंगे} (\emph{-enge}) is the plural future.
\end{cousinweb}

\begin{grammarlens}
\textbf{The future is an ending, not a helper.} \emph{mil-} (``meet'') $+$
\emph{-enge} (``we/they will'') $\to$ \emph{milenge}. No word like English
``will'' --- the tense lives in the ending, as person lives in
\emph{h\=u\d{m}/hai/hai\d{m}}.
\end{grammarlens}

\section{\hi{फिर मिलेंगे} \emph{(phir milenge)} --- ``we'll meet again''}
\label{lesson:phir-milenge}

\hi{फिर} (again) $+$ \hi{मिलेंगे} (we will meet) $=$ ``\textbf{we'll meet
again}.'' Where \hi{अलविदा} (\emph{alvid\=a}) is grave and final --- and, fittingly,
a \textbf{Perso-Arabic} word (\emph{al-wid\=a}, ``the farewell'') --- \emph{phir
milenge} is native, everyday, and hopeful, the sibling of the ``promise of
return'' goodbyes across South Asia (Tamil, Bengali, Marathi all say ``I'll come
again'').

\begin{grammarlens}
Two registers: \hi{अलविदा} (formal, final, Perso-Arabic) versus \hi{फिर मिलेंगे}
(everyday, warm, native). Even more casual: \emph{phir milte hai\d{m}} (``(we)
meet again,'' present for a near future).
\end{grammarlens}

\section{\hi{कल मिलते हैं} \emph{(kal milte hai\d{m})} --- ``see you tomorrow''}
\label{lesson:kal-milte-hain}

\begin{sounds}
\hi{क} (\emph{ka}) $+$ \hi{ल} (\emph{la}) $\to$ \hi{कल} (\emph{kal}). Two simple
letters, one tricky word.
\end{sounds}

\begin{cousinweb}
\hi{कल} (\emph{kal}) means \textbf{both ``tomorrow'' and ``yesterday''} --- the
day on either side of today --- from Sanskrit \emph{k\=ala} (``time''), the very
root behind Punjabi's \emph{ak\=al} (``timeless''). Context and the verb's tense
tell which: \emph{kal milte hai\d{m}} (``tomorrow''), \emph{kal mil\=a}
(``yesterday'').
\end{cousinweb}

\begin{grammarlens}
\hi{मिलते हैं} (\emph{milte hai\d{m}}) is literally ``(we) meet'' --- Hindi, like
English (``I'm seeing you tomorrow''), uses the present for a scheduled future.
\end{grammarlens}

\section{\hi{चलता हूँ} \emph{(calt\=a h\=u\d{m})} --- ``I'll be off''}
\label{lesson:chalta-hun}

\begin{sounds}
\hi{च} (\emph{ca}) $+$ \hi{ल} (\emph{la}) $+$ long \emph{\=a} $\to$ \hi{चलता}
(\emph{calt\=a}), with \hi{हूँ} (\emph{h\=u\d{m}}).
\end{sounds}

\begin{cousinweb}
\hi{चलता हूँ} literally ``\textbf{I go / I move}'' --- used like English ``I'll be
off.'' The verb \hi{चलना} (\emph{caln\=a}, ``to move, walk, go'') is from Sanskrit
\emph{cal-} (``to move, stir'') --- the root of \emph{calan} (``circulation'') and
\emph{ca\~{n}cal} (``restless''). Like Bengali \emph{\=ashi} and Marathi
\emph{yeto}, Hindi leaves by saying ``I'm \emph{moving},'' never a blunt ``I'm
leaving.''
\end{cousinweb}

\begin{grammarlens}
\textbf{The verb agrees with the speaker's gender.} A man says \hi{चलता हूँ}
(\emph{calt\=a h\=u\d{m}}); a woman says \hi{चलती हूँ} (\emph{calt\=\i\ h\=u\d{m}}).
The \emph{-t\=a / -t\=\i} ending marks the speaker --- a feature English lacks (compare
Marathi \emph{yeto} m.\ / \emph{yete} f.).
\end{grammarlens}

\section{The farewells}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Hindi & & Register \\
\midrule
\hi{अलविदा} & \emph{alvid\=a} & formal, final (Perso-Arabic) \\
\hi{फिर मिलेंगे} & \emph{phir milenge} & everyday, ``we'll meet again'' \\
\hi{कल मिलते हैं} & \emph{kal milte hai\d{m}} & ``see you tomorrow'' \\
\hi{चलता हूँ} & \emph{calt\=a/calt\=\i\ h\=u\d{m}} & casual, ``I'll be off'' \\
\bottomrule
\end{tabular}
\end{center}

Roots banked: \emph{milenge} (\emph{mil-} ``to join''), \emph{kal} (\emph{k\=ala}
``time,'' cousin of Punjabi \emph{ak\=al}), \emph{calt\=a} (\emph{cal-} ``to
move,'' the \emph{-t\=a/-t\=\i} marking the speaker's gender). Next chapter: the
first real verbs --- \emph{mai\d{m} hind\=\i\ bolt\=a h\=u\d{m}} --- and Hindi's
sentences begin to move.
