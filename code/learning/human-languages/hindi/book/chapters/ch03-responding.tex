\chapter{How Are You?}
\label{ch:responding}

After the name comes the exchange everyone actually has --- asking, and
answering, how someone is:

\begin{center}
\hi{आप कैसे हैं?} \quad(\emph{\=ap kaise hai\d{m}} --- ``How are you?'')\\
\hi{मैं ठीक हूँ, धन्यवाद।} \quad(\emph{mai\d{m} \d{t}h\=\i k h\=u\d{m}, dhanyav\=ad} --- ``I'm fine, thank you.'')
\end{center}

Every atom traced, then assembled. \emph{Practice lessons:
\texttt{lessons/HI-C03-*.md}.}

\section{\hi{कैसे} \emph{(kaise)} --- ``how,'' another of the k- questions}
\label{lesson:kaise}

\begin{sounds}
\hi{क} (\emph{ka}) $+$ \hi{ै} (the \emph{ai}-m\=atr\=a) $\to$ \hi{कै} (\emph{kai});
\hi{स} (\emph{sa}) $+$ \hi{े} (the \emph{e}-m\=atr\=a) $\to$ \hi{से} (\emph{se}).
Read \hi{कै·से} $\to$ \emph{kaise}.
\end{sounds}

\begin{cousinweb}
\hi{कैसे} (\emph{kaise}, ``how'') $\leftarrow$ Sanskrit \emph{k\=\i d\d{r}\'{s}a}
(``of what kind'') / the interrogative stem \emph{ka-}, from Proto-Indo-European
*k\ensuremath{^w}o- --- the same question-root behind English \textbf{wh-} words
and Latin \emph{qu-}. Hindi's questions nearly all begin with this \emph{k-}:
\emph{ky\=a} (what), \emph{kaun} (who), \emph{kab} (when), \emph{kah\=a\d{m}}
(where), \emph{kaise} (how).
\end{cousinweb}

\begin{grammarlens}
The adjective \hi{कैसा} (\emph{kais\=a}) agrees: \emph{kais\=a} (m.), \emph{kais\=\i}
(f.), \emph{kaise} (m.\ plural / respectful). Because the respectful ``you''
(\emph{\=ap}) takes a plural verb, ``how are \emph{you}?'' uses the plural
\hi{कैसे}.
\end{grammarlens}

\section{\hi{आप कैसे हैं?} \emph{(\=ap kaise hai\d{m})} --- ``how are you?''}
\label{lesson:aap-kaise-hain}

\hi{हैं} is \hi{है} (\emph{hai}, ``is'') $+$ the \emph{anusv\=ara} \hi{ं} $\to$
\hi{हैं} (\emph{hai\d{m}}), the nasalised plural. So \hi{आप कैसे हैं?} is literally
``\textbf{you how are?}'' --- verb last, and because \emph{\=ap} is grammatically
plural (that is how Hindi shows respect), its verb is the plural \emph{hai\d{m}},
not \emph{hai}.

\begin{grammarlens}
\textbf{Respect is plural.} To an elder or stranger: \emph{\=ap kaise hai\d{m}?}
To a friend: \emph{tum kaise ho?} (\emph{ho} is the familiar ``are''). Match the
``you'' to the verb: \emph{\=ap \ldots\ hai\d{m}}, \emph{tum \ldots\ ho}.
\end{grammarlens}

\section{\hi{मैं} \emph{(mai\d{m})} --- ``I,'' the m- of the first person}
\label{lesson:main}

\begin{sounds}
\hi{म} (\emph{ma}) $+$ \hi{ै} (\emph{ai}) $+$ the \emph{anusv\=ara} \hi{ं} $\to$
\hi{मैं} (\emph{mai\d{m}}). One nasal syllable.
\end{sounds}

\begin{cousinweb}
\hi{मैं} (\emph{mai\d{m}}, ``I'') $\leftarrow$ Sanskrit \emph{may\=a} (``by me''),
on the first-person root \emph{ma-} --- the same root behind Chapter 2's
\emph{mer\=a} (``my'') and behind English \textbf{me}, \textbf{my}, \textbf{mine}
and Latin \emph{m\=e}. Hindi's ``I,'' its ``my,'' and English's ``me/my'' grow
from one ancient first-person sound.
\end{cousinweb}

\section{\hi{हूँ} \emph{(h\=u\d{m})} --- ``am,'' cousin of English \emph{am}}
\label{lesson:hun}

\begin{sounds}
\hi{ह} (\emph{ha}) $+$ \hi{ू} (long \emph{u}) $\to$ \hi{हू} (\emph{h\=u}); the
\emph{chandrabindu} \hi{ँ} nasalises $\to$ \hi{हूँ} (\emph{h\=u\d{m}}).
\end{sounds}

\begin{cousinweb}
\hi{हूँ} (\emph{h\=u\d{m}}, ``am'') $\leftarrow$ Sanskrit \emph{asmi} (``I am'') ---
the first person of the root that also gave you \emph{hai}. That root, PIE
*h\textsubscript{1}es-, is ``to be'' across the family: \emph{asmi} $\to$ English
\textbf{am}, Latin \emph{sum}; \emph{asti} $\to$ \emph{hai} / English \emph{is}.
The copula lines up person by person: \hi{हूँ} \emph{am}, \hi{है} \emph{is},
\hi{हैं} \emph{are}.
\end{cousinweb}

\section{\hi{ठीक} \emph{(\d{t}h\=\i k)} --- ``fine,'' and how to answer}
\label{lesson:thik}

\begin{sounds}
\hi{ठ} (\emph{\d{t}ha}, a \textbf{retroflex} \emph{th} --- tongue curled to the
roof of the mouth) $+$ \hi{ी} (long \emph{\=\i}) $+$ \hi{क} (\emph{ka}) $\to$
\hi{ठीक}. The retroflex \hi{ट}-\hi{ठ}-\hi{ड} series is a hallmark of Indian
sound.
\end{sounds}

\begin{cousinweb}
\hi{ठीक} (\emph{\d{t}h\=\i k}) means ``correct, right, fine, exactly, OK.'' It is
a genuinely \textbf{native} Indian word --- no European cousin --- and one of the
most useful in the language. Not every word has a far-flung cognate, and that is
worth seeing too: \emph{\d{t}h\=\i k} is Indian to the core.
\end{cousinweb}

Put the atoms together: \hi{मैं ठीक हूँ} (\emph{mai\d{m} \d{t}h\=\i k h\=u\d{m}}) ---
``\textbf{I am fine}'' (literally ``I fine am''). The whole exchange:

\begin{center}
--- \emph{\=ap kaise hai\d{m}?} \quad --- \emph{mai\d{m} \d{t}h\=\i k h\=u\d{m},
dhanyav\=ad.}
\end{center}

\section{\hi{आपका स्वागत है} \emph{(\=apk\=a sv\=agat hai)} --- ``you're welcome''}
\label{lesson:aapka-swagat-hai}

\begin{sounds}
The key word \hi{स्वागत} (\emph{sv\=agat}): the conjunct \hi{स्व} (\hi{स्} \emph{s}
$+$ \hi{व} \emph{va} $\to$ \emph{sva}) $+$ \hi{ग} (\emph{ga}) $+$ \hi{त}
(\emph{ta}), with \hi{आपका} (\emph{\=apk\=a}, ``your'') and \hi{है} (\emph{hai}).
\end{sounds}

\begin{cousinweb}
\hi{आपका स्वागत है} $=$ ``your welcome is'' $\to$ ``\textbf{you are welcome}.''
The heart, \hi{स्वागत} (\emph{sv\=agat}), is Sanskrit \emph{su} (``well, good'')
$+$ \emph{\=agata} (``come'') --- ``\textbf{well come},'' exactly like English
\emph{welcome}. Inside it: \emph{su-} the twin of Greek \emph{eu-}
(\emph{euphoria}), and \emph{\=agata} from the root \emph{gam} (``go/come''),
cousin of English \textbf{come}. You will meet \emph{sv\=agatam} again, unchanged,
in the Sanskrit track.
\end{cousinweb}

\section{The whole exchange}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Hindi & English \\
\midrule
\hi{आप कैसे हैं?} & How are you? \\
\hi{मैं ठीक हूँ, धन्यवाद।} & I'm fine, thank you. \\
\hi{आपका स्वागत है।} & You're welcome. \\
\bottomrule
\end{tabular}
\end{center}

Every atom traced: \emph{kaise} (the interrogative \emph{k-}, English \emph{how}),
\emph{mai\d{m}} (root \emph{ma-}, English \emph{me}), \emph{h\=u\d{m}} (Sanskrit
\emph{asmi}, English \emph{am}), \emph{sv\=agat} (\emph{su} $+$ \emph{\=agata} $=$
``well come''). You now hold the copula trio --- \emph{h\=u\d{m} / hai / hai\d{m}}.
