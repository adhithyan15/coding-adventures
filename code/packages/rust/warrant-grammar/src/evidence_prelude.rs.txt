/// Dynamic value of the Warrant pure profile (VER04e section 2.1 subset).
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum V {
    Bool(bool),
//@if v0
    Nat(u64),
    U64(u64),
//@endif
//@if v1
    /// `Int` and `Nat` (the numeric tower below the fixed-width types).
    Int(i128),
    /// A fixed-width integer: its type and canonical value.
    Fixed(F, i128),
    /// `Option<T>` (`None` carries no type).
    Opt(Option<Box<V>>),
//@endif
    Bytes(Vec<u8>),
    Ok(Box<V>),
    Err(String),
    Name(String),
//@if comp
    /// WX-A: a tuple (VER04j section 4.5).
    Tup(Vec<V>),
    /// WX-A: a record or variant value: type name, constructor index,
    /// constructor name, fields in declaration order (a record is constructor 0).
    Con(&'static str, usize, &'static str, Vec<V>),
    /// A list.
    List(Vec<V>),
//@endif
}

//@if v0
impl V {
    pub fn truthy(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn length(&self) -> V {
        match self {
            V::Bytes(b) => V::Nat(b.len() as u64),
            _ => panic!("len of a non-bytes value"),
        }
    }
    pub fn err_of(v: V) -> V {
        match v {
            V::Name(n) => V::Err(n),
            _ => panic!("Err needs an error constructor"),
        }
    }
    fn num(&self) -> u64 {
        match self {
            V::Nat(n) | V::U64(n) => *n,
            _ => panic!("numeric operand expected"),
        }
    }
    pub fn compare(l: &V, r: &V, op: &str) -> V {
        let (a, b) = (l.num(), r.num());
        V::Bool(match op {
            "<" => a < b,
            ">" => a > b,
            "<=" => a <= b,
            _ => a >= b,
        })
    }
    pub fn arith(l: &V, r: &V, op: char) -> V {
        let (a, b) = (l.num(), r.num());
        let n = match op {
            '+' => a.checked_add(b),
            '-' => a.checked_sub(b),
            _ => a.checked_mul(b),
        };
        V::Nat(n.expect("model arithmetic overflow"))
    }
    fn bytes(&self) -> &[u8] {
        match self {
            V::Bytes(b) => b,
            _ => panic!("bytes expected"),
        }
    }
    fn u64v(&self) -> u64 {
        self.num()
    }
    fn boolv(&self) -> bool {
        matches!(self, V::Bool(true))
    }
}

//@if v1
/// The eight fixed-width integer types of Warrant v1 (VER04j section 2.2).
///
/// A value is stored as the mathematical integer it denotes (an `i128`, so
/// `U64` and `I64` both fit) and is always canonical, i.e. inside the range of
/// its type.  Bitwise functions work on the width-`w` two's-complement
/// *pattern*, held in a `u128` and masked to `w` bits.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum F {
    U8,
    U16,
    U32,
    U64,
    I8,
    I16,
    I32,
    I64,
}

impl F {
    pub fn bits(self) -> u32 {
        match self {
            F::U8 | F::I8 => 8,
            F::U16 | F::I16 => 16,
            F::U32 | F::I32 => 32,
            F::U64 | F::I64 => 64,
        }
    }
    pub fn signed(self) -> bool {
        matches!(self, F::I8 | F::I16 | F::I32 | F::I64)
    }
    pub fn min(self) -> i128 {
        if self.signed() {
            -(1i128 << (self.bits() - 1))
        } else {
            0
        }
    }
    pub fn max(self) -> i128 {
        if self.signed() {
            (1i128 << (self.bits() - 1)) - 1
        } else {
            (1i128 << self.bits()) - 1
        }
    }
    pub fn mask(self) -> u128 {
        (1u128 << self.bits()) - 1
    }
    /// The two's-complement pattern of `v` masked to `w` bits (any `v`).
    pub fn pattern(self, v: i128) -> u128 {
        (v as u128) & self.mask()
    }
    /// Read a pattern back as a canonical value of the type.
    pub fn canon(self, p: u128) -> i128 {
        let p = p & self.mask();
        if self.signed() && p >> (self.bits() - 1) == 1 {
            (p as i128) - (1i128 << self.bits())
        } else {
            p as i128
        }
    }
    fn contains(self, v: i128) -> bool {
        self.min() <= v && v <= self.max()
    }
}

/// Edge values of a fixed-width type (VER04j section 2.6), in a fixed order.
/// Unsigned width `w`: `0..=255`, `1<<i` for `8 <= i < w`, `MAX`, then
/// `MAX ^ (1<<i)` for `0 <= i < w`.  Signed: the same patterns read signed,
/// plus `-1 ..= -255`, deduplicated, ascending.
pub fn fx_edge(f: F) -> Vec<i128> {
    let w = f.bits();
    let umax = f.mask();
    let mut u: Vec<u128> = (0..=255u128).filter(|v| *v <= umax).collect();
    u.extend((8..w).map(|i| 1u128 << i));
    u.push(umax);
    u.extend((0..w).map(|i| umax ^ (1u128 << i)));
    if !f.signed() {
        return u.into_iter().map(|p| p as i128).collect();
    }
    let mut s: Vec<i128> = u.into_iter().map(|p| f.canon(p)).collect();
    s.extend((1..=255i128).map(|k| -k).filter(|v| f.contains(*v)));
    s.sort();
    s.dedup();
    s
}

/// Every value of a narrow type (`U8 I8 U16 I16`), ascending.
pub fn fx_full(f: F) -> Vec<i128> {
    (f.min()..=f.max()).collect()
}

impl V {
    pub fn truthy(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn length(&self) -> V {
        match self {
            V::Bytes(b) => V::Int(b.len() as i128),
//@if comp
            V::List(l) => V::Int(l.len() as i128),
//@endif
            _ => panic!("len of a non-bytes value"),
        }
    }
    pub fn err_of(v: V) -> V {
        match v {
            V::Name(n) => V::Err(n),
            _ => panic!("Err needs an error constructor"),
        }
    }
    /// The mathematical value of an `Int`/`Nat` or fixed-width operand.
    pub fn int(&self) -> i128 {
        match self {
            V::Int(n) | V::Fixed(_, n) => *n,
            _ => panic!("numeric operand expected"),
        }
    }
    pub fn fx(&self) -> (F, i128) {
        match self {
            V::Fixed(f, n) => (*f, *n),
            _ => panic!("fixed-width operand expected"),
        }
    }
    /// A shift count or bit index: a mathematical natural number.
    pub fn count(&self) -> u128 {
        u128::try_from(self.int()).expect("count must be a natural number")
    }
    fn bytes(&self) -> &[u8] {
        match self {
            V::Bytes(b) => b,
            _ => panic!("bytes expected"),
        }
    }
    fn boolv(&self) -> bool {
        matches!(self, V::Bool(true))
    }
    pub fn compare(l: &V, r: &V, op: &str) -> V {
        let (a, b) = (l.int(), r.int());
        V::Bool(match op {
            "<" => a < b,
            ">" => a > b,
            "<=" => a <= b,
            _ => a >= b,
        })
    }
    /// `+ - *`: wrapping on equal fixed-width operands (the pattern product
    /// keeps `U64 * U64` exact), checked `i128` on `Int`/`Nat`.
    pub fn arith(l: &V, r: &V, op: char) -> V {
        if let (V::Fixed(f, a), V::Fixed(g, b)) = (l, r) {
            assert_eq!(f, g, "mixed fixed-width operands");
            let (pa, pb) = (f.pattern(*a), f.pattern(*b));
            let p = match op {
                '+' => pa.wrapping_add(pb),
                '-' => pa.wrapping_sub(pb),
                _ => pa.wrapping_mul(pb),
            };
            return V::Fixed(*f, f.canon(p));
        }
        let (a, b) = (l.int(), r.int());
        let n = match op {
            '+' => a.checked_add(b),
            '-' => a.checked_sub(b),
            _ => a.checked_mul(b),
        };
        V::Int(n.expect("model arithmetic overflow"))
    }
    /// Unary minus: wrapping on a signed fixed-width value, exact on `Int`.
    pub fn neg(v: &V) -> V {
        match v {
            V::Fixed(f, a) => V::Fixed(*f, f.canon(f.pattern(*a).wrapping_neg())),
            _ => V::Int(v.int().checked_neg().expect("model arithmetic overflow")),
        }
    }
    fn opt(v: Option<i128>, f: F) -> V {
        V::Opt(v.map(|x| Box::new(V::Fixed(f, x))))
    }
    pub fn widen(t: F, x: &V) -> V {
        let v = x.int();
        assert!(t.contains(v), "widen operand outside the target range");
        V::Fixed(t, v)
    }
    pub fn wrap(t: F, x: &V) -> V {
        V::Fixed(t, t.canon(t.pattern(x.int())))
    }
    pub fn try_(t: F, x: &V) -> V {
        let v = x.int();
        V::Opt(t.contains(v).then(|| Box::new(V::Fixed(t, v))))
    }
    pub fn to_int(x: &V) -> V {
        V::Int(x.fx().1)
    }
    pub fn to_nat(x: &V) -> V {
        let (f, v) = x.fx();
        assert!(!f.signed(), "to_nat of a signed value");
        V::Int(v)
    }
    pub fn max_value(t: F) -> V {
        V::Fixed(t, t.max())
    }
    pub fn min_value(t: F) -> V {
        V::Fixed(t, t.min())
    }
    pub fn checked(op: char, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let exact = match op {
            '+' => x.checked_add(y),
            '-' => x.checked_sub(y),
            _ => x.checked_mul(y),
        };
        V::opt(exact.filter(|v| f.contains(*v)), f)
    }
    /// `div`/`rem`: unsigned operands, or `Int`/`Nat` with floor semantics.
    pub fn divrem(div: bool, a: &V, b: &V) -> V {
        if let V::Fixed(f, x) = a {
            let (_, y) = b.fx();
            assert!(!f.signed(), "div/rem of signed operands");
            return V::opt((y != 0).then(|| if div { x / y } else { x % y }), *f);
        }
        let (x, y) = (a.int(), b.int());
        if y == 0 || (x == i128::MIN && y == -1) {
            return V::Opt(None);
        }
        let r = if div {
            let q = x / y;
            if x % y != 0 && ((x < 0) != (y < 0)) {
                q - 1
            } else {
                q
            }
        } else if y == -1 {
            0
        } else {
            let m = x % y;
            if m != 0 && ((m < 0) != (y < 0)) {
                m + y
            } else {
                m
            }
        };
        V::Opt(Some(Box::new(V::Int(r))))
    }
    /// `div_trunc`/`rem_trunc` on signed operands: `None` for `b == 0` and for
    /// `min / -1`.
    pub fn divrem_trunc(div: bool, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let ok = y != 0 && !(y == -1 && x == f.min());
        V::opt(ok.then(|| if div { x / y } else { x % y }), f)
    }
//@if concat
    /// `concat(a, b)` of two byte strings.
    pub fn concat(a: &V, b: &V) -> V {
        match (a, b) {
            (V::Bytes(x), V::Bytes(y)) => {
                let mut out = Vec::with_capacity(x.len() + y.len());
                out.extend_from_slice(x);
                out.extend_from_slice(y);
                V::Bytes(out)
            }
            _ => panic!("concat of byte strings expected"),
        }
    }
//@endif
//@if minmax
    /// `imin`/`imax` (WX-H): the smaller / larger of two `Int` values.
    pub fn imin(a: &V, b: &V) -> V {
        V::Int(a.int().min(b.int()))
    }
    pub fn imax(a: &V, b: &V) -> V {
        V::Int(a.int().max(b.int()))
    }
//@endif
    pub fn unwrap_or(o: &V, d: &V) -> V {
        match o {
            V::Opt(Some(v)) => (**v).clone(),
            V::Opt(None) => d.clone(),
            _ => panic!("Option operand expected"),
        }
    }
    /// `bit_and` (`'&'`), `bit_or` (`'|'`), `bit_xor` (`'^'`).
    pub fn bitop(op: char, a: &V, b: &V) -> V {
        let (f, x) = a.fx();
        let (_, y) = b.fx();
        let (px, py) = (f.pattern(x), f.pattern(y));
        V::Fixed(
            f,
            f.canon(match op {
                '&' => px & py,
                '|' => px | py,
                _ => px ^ py,
            }),
        )
    }
    pub fn bit_not(a: &V) -> V {
        let (f, x) = a.fx();
        V::Fixed(f, f.canon(!f.pattern(x)))
    }
    /// `shl shr sar rotl rotr`.  A count `>= w` is handled *before* any host
    /// shift (VER04j section 3.5), so the host never sees an over-wide shift.
    pub fn shift(op: &str, a: &V, n: &V) -> V {
        let (f, x) = a.fx();
        let (n, w, p) = (n.count(), u128::from(f.bits()), f.pattern(x));
        let v = match op {
            "shl" => {
                if n >= w {
                    0
                } else {
                    f.canon(p << n)
                }
            }
            "shr" => {
                if n >= w {
                    0
                } else {
                    f.canon(p >> n)
                }
            }
            "sar" => {
                assert!(f.signed(), "sar of an unsigned value");
                if n >= w {
                    if x < 0 {
                        -1
                    } else {
                        0
                    }
                } else {
                    x >> n
                }
            }
            _ => {
                let k = if op == "rotl" { n % w } else { (w - n % w) % w };
                if k == 0 {
                    x
                } else {
                    f.canon((p << k) | (p >> (w - k)))
                }
            }
        };
        V::Fixed(f, v)
    }
    /// `popcount clz ctz` of the width-`w` pattern.
    pub fn bitcount(op: &str, a: &V) -> V {
        let (f, x) = a.fx();
        let p = f.pattern(x);
        let w = f.bits();
        V::Int(match op {
            "popcount" => i128::from(p.count_ones()),
            "clz" => {
                if p == 0 {
                    i128::from(w)
                } else {
                    i128::from(p.leading_zeros() - (128 - w))
                }
            }
            _ => {
                if p == 0 {
                    i128::from(w)
                } else {
                    i128::from(p.trailing_zeros())
                }
            }
        })
    }
    pub fn bit(a: &V, i: &V) -> V {
        let (f, x) = a.fx();
        let i = i.count();
        V::Bool(i < u128::from(f.bits()) && (f.pattern(x) >> i) & 1 == 1)
    }
    /// `be_bytes` / `le_bytes`.
    pub fn endian_bytes(big: bool, a: &V) -> V {
        let (f, x) = a.fx();
        let p = f.pattern(x);
        let n = (f.bits() / 8) as usize;
        let mut v: Vec<u8> = (0..n).rev().map(|i| (p >> (8 * i)) as u8).collect();
        if !big {
            v.reverse();
        }
        V::Bytes(v)
    }
    /// `from_be<T>` / `from_le<T>`.
    pub fn from_endian(big: bool, t: F, b: &V) -> V {
        let mut bytes = b.bytes().to_vec();
        assert_eq!(bytes.len() as u32 * 8, t.bits(), "wrong byte width");
        if !big {
            bytes.reverse();
        }
        let p = bytes.iter().fold(0u128, |acc, x| (acc << 8) | u128::from(*x));
        V::Fixed(t, t.canon(p))
    }
}

/// The panic payload an adapter raises for a value its host integer type cannot
/// hold: the case is *unreachable by type*, skipped (never narrowed) and counted
/// (VER04j section 4.6).
#[derive(Debug)]
pub struct Unreachable;

thread_local! {
    static UNREACH: std::cell::Cell<u64> = const { std::cell::Cell::new(0) };
}

fn count_unreach() {
    UNREACH.with(|c| c.set(c.get() + 1));
}

fn take_unreach() -> u64 {
    UNREACH.with(|c| c.replace(0))
}

/// Convert a model integer to a host integer, or raise [`Unreachable`].
fn host<T: TryFrom<i128>>(v: i128) -> T {
    T::try_from(v).unwrap_or_else(|_| std::panic::panic_any(Unreachable))
}

/// A match pattern, built by the generated model code (VER04j section 4.3).
#[derive(Clone, Copy, Debug)]
pub enum Pat {
    Wild,
    /// A binder: the matched value is appended to the bindings.
    Bind,
    Bool(bool),
    Tup(&'static [Pat]),
    /// Constructor index and field patterns of a record or variant.
    Con(usize, &'static [Pat]),
    None,
    Some(&'static Pat),
    Ok(&'static Pat),
    /// `Err(p)`: `p` is matched against the error value (`V::Name`).
    Err(&'static Pat),
    /// `E.CASE`.
    Case(&'static str),
}

/// Does `v` match `p`?  Binders are pushed to `out` left to right.
pub fn pat_match(p: &Pat, v: &V, out: &mut Vec<V>) -> bool {
    match (p, v) {
        (Pat::Wild, _) => true,
        (Pat::Bind, v) => {
            out.push(v.clone());
            true
        }
        (Pat::Bool(b), V::Bool(x)) => b == x,
        (Pat::Tup(ps), V::Tup(vs)) => {
            ps.len() == vs.len() && ps.iter().zip(vs).all(|(p, v)| pat_match(p, v, out))
        }
        (Pat::Con(i, ps), V::Con(_, j, _, vs)) => {
            i == j && ps.len() == vs.len() && ps.iter().zip(vs).all(|(p, v)| pat_match(p, v, out))
        }
        (Pat::None, V::Opt(None)) => true,
        (Pat::Some(p), V::Opt(Some(b))) => pat_match(p, b, out),
        (Pat::Ok(p), V::Ok(b)) => pat_match(p, b, out),
        (Pat::Err(p), V::Err(n)) => pat_match(p, &V::Name(n.clone()), out),
        (Pat::Case(c), V::Name(n)) => c == n,
        _ => false,
    }
}

/// The result of an operation with its `Ok` (or `Some`) payload rewritten by `f`.
fn map_payload(v: V, f: impl FnOnce(V) -> V) -> V {
    match v {
        V::Ok(x) => V::Ok(Box::new(f(*x))),
        V::Opt(Some(x)) => V::Opt(Some(Box::new(f(*x)))),
        other => f(other),
    }
}

/// Fault `drop-consumed`: the second component of a tuple result becomes `0`.
fn fault_drop_consumed(v: V) -> V {
    map_payload(v, |x| match x {
        V::Tup(mut vs) if vs.len() >= 2 => {
            vs[1] = zero_like(&vs[1]);
            V::Tup(vs)
        }
        other => other,
    })
}

/// Fault `swap-components`: the components of a homogeneous tuple are reversed
/// (exchanged, for a pair).
fn fault_swap_components(v: V) -> V {
    map_payload(v, |x| match x {
        V::Tup(mut vs) => {
            vs.reverse();
            V::Tup(vs)
        }
        other => other,
    })
}

/// Fault `field-zero`: the first numeric field of a record result becomes `0`.
fn fault_field_zero(v: V) -> V {
    map_payload(v, |x| match x {
        V::Con(t, i, c, mut vs) => {
            if let Some(slot) = vs.iter_mut().find(|f| matches!(f, V::Int(_) | V::Fixed(..))) {
                *slot = zero_like(slot);
            }
            V::Con(t, i, c, vs)
        }
        other => other,
    })
}

/// Fault `off-by-one-consumed`: the second component of a `(value, consumed)`
/// pair is off by one, `+ 1` when even and `- 1` when odd, so both directions
/// occur and a claim that pins `consumed` on every input catches the fault.
fn fault_off_by_one_consumed(v: V) -> V {
    map_payload(v, |x| match x {
        V::Tup(mut vs) if vs.len() >= 2 => {
            vs[1] = match &vs[1] {
                V::Int(n) => V::Int(if n % 2 == 0 { n + 1 } else { n - 1 }),
                V::Fixed(f, n) => {
                    let p = f.pattern(*n);
                    let q = if n % 2 == 0 { p.wrapping_add(1) } else { p.wrapping_sub(1) };
                    V::Fixed(*f, f.canon(q))
                }
                other => other.clone(),
            };
            V::Tup(vs)
        }
        other => other,
    })
}

/// Fault `wrong-ctor`: a variant result is reported as the *previous*
/// constructor (the last one for the first), with the first element of each of
/// its fields' domains as fields.
fn fault_wrong_ctor(v: V, ty: Ty) -> V {
    map_payload(v, |x| match (x, ty) {
        (V::Con(name, i, _, _), Ty::Data(_, ctors)) if !ctors.is_empty() => {
            let j = (i + ctors.len() - 1) % ctors.len();
            let (cname, fields) = ctors[j];
            V::Con(name, j, cname, fields.iter().map(|t| domain_take(*t, 1).remove(0)).collect())
        }
        (other, _) => other,
    })
}

/// Fault `ok-as-err`: a successful result is reported as the first error case.
fn fault_ok_as_err(v: V, first_case: &str) -> V {
    match v {
        V::Ok(_) => V::Err(first_case.to_string()),
        other => other,
    }
}

/// A zero of the same numeric type as `v` (any other value is unchanged).
fn zero_like(v: &V) -> V {
    match v {
        V::Int(_) => V::Int(0),
        V::Fixed(f, _) => V::Fixed(*f, 0),
        other => other.clone(),
    }
}

fn mask_last_byte(bytes: &[u8]) -> Vec<u8> {
    let mut v = bytes.to_vec();
    if let Some(last) = v.last_mut() {
        *last = 0;
    }
    v
}

/// Run `body`, converting a panic into `None` (used for declared panic errors).
fn contained<T>(body: impl FnOnce() -> T) -> Option<T> {
    catch_unwind(AssertUnwindSafe(body)).ok()
}

//@if scoped
thread_local! {
    /// The one operation a negative control is currently attacking, if any.
    static ATTACK_OP: std::cell::RefCell<Option<&'static str>> = const { std::cell::RefCell::new(None) };
}

/// The fault `op`'s adapter should obey: the requested one unless a negative
/// control is attacking a different operation.
fn fault_for(op: &str, fault: Fault) -> Fault {
    ATTACK_OP.with(|a| match *a.borrow() {
        Some(target) if target != op => Fault::None,
        _ => fault,
    })
}

/// Run `body` with only `op` obeying injected faults.
fn attacking<T>(op: &'static str, body: impl FnOnce() -> T) -> T {
    struct Reset;
    impl Drop for Reset {
        fn drop(&mut self) {
            ATTACK_OP.with(|a| *a.borrow_mut() = None);
        }
    }
    ATTACK_OP.with(|a| *a.borrow_mut() = Some(op));
    let _reset = Reset;
    body()
}

//@if obs
// ---------------------------------------------------------------------------
// WX-H: the allocation observer (VER04j sections 6.3 and 14.2 item 33).
//
// Every adapter call of an operation that has a `heap` or `output` bound runs
// on a worker thread inside an *allocation window*.  A counting global
// allocator charges every allocation made on a thread whose window is open.
//
//   supervisor thread            worker thread (window open)
//   -----------------            ---------------------------
//   obs_exec(job) --job--------> job()      alloc  alloc  free ...
//        |                                     |
//        |  (waits)                            +-- over the cap? record the
//        v                                         attempted size, wake the
//   Finished(result, peak)  <---- done            supervisor, park FOREVER
//   or Tripped(attempted)   <---- trip
//
// Why park instead of failing the allocation?  A `GlobalAlloc` may not unwind,
// and on stable Rust an allocation that returns null aborts the process
// (`handle_alloc_error`), which is exactly the CI failure a bomb must not
// cause.  The over-cap request is therefore never made: its thread is parked,
// the supervisor fails the instance, and the parked worker (holding at most
// the cap) is abandoned.  A test that has tripped once fails its later observed
// instances without running them (so a shrinker cannot abandon a worker per
// candidate), and at most `OBS_MAX_ABANDONED` workers, holding at most
// `OBS_MAX_HELD` bytes between them, are ever abandoned in one process.
// ---------------------------------------------------------------------------

use std::alloc::{GlobalAlloc, Layout, System};

/// Workers that may be abandoned before observed calls are refused outright.
const OBS_MAX_ABANDONED: usize = 64;
/// Bytes the abandoned workers may hold between them (they are never freed).
const OBS_MAX_HELD: u64 = 1 << 30;

/// The panic payload an observed instance raises when the observer fails it
/// (`bound-heap-exceeded`, `bound-negative`, `output-bound-exceeded`).
#[derive(Debug)]
pub struct ObsFailure(pub String);

struct Window {
    active: Cell<bool>,
    cur: Cell<u64>,
    peak: Cell<u64>,
    slot: Cell<Option<&'static Slot>>,
}

thread_local! {
    // `const` and without a destructor: touching it from inside the allocator
    // can neither allocate nor register a destructor.
    static WIN: Window = const {
        Window {
            active: Cell::new(false),
            cur: Cell::new(0),
            peak: Cell::new(0),
            slot: Cell::new(None),
        }
    };
}

enum ObsReply {
    Done(Result<V, Box<dyn std::any::Any + Send>>, u64),
    /// The size the call asked for in total, and what it already held.
    Tripped(u64, u64),
}

struct Slot {
    state: Mutex<Option<ObsReply>>,
    ready: std::sync::Condvar,
}

struct Observer;

fn obs_charge(n: u64) {
    let _ = WIN.try_with(|w| {
        if !w.active.get() {
            return;
        }
        let cur = w.cur.get().saturating_add(n);
        if cur > OBSERVER_CAP_BYTES {
            // Nothing below this line is observed (it must not charge itself).
            w.active.set(false);
            obs_trip(w, cur, w.cur.get());
        }
        w.cur.set(cur);
        if cur > w.peak.get() {
            w.peak.set(cur);
        }
    });
}

fn obs_release(n: u64) {
    let _ = WIN.try_with(|w| {
        if w.active.get() {
            w.cur.set(w.cur.get().saturating_sub(n));
        }
    });
}

/// Report the attempted peak to the supervisor and park this thread forever.
fn obs_trip(w: &Window, attempted: u64, held: u64) -> ! {
    if let Some(slot) = w.slot.get() {
        if let Ok(mut g) = slot.state.lock() {
            *g = Some(ObsReply::Tripped(attempted, held));
        }
        slot.ready.notify_one();
    }
    loop {
        std::thread::park();
    }
}

// SAFETY: every method forwards to `System` unchanged after accounting; the
// accounting touches only `const` thread-locals and never allocates or unwinds.
#[allow(unsafe_op_in_unsafe_fn)]
unsafe impl GlobalAlloc for Observer {
    unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
        obs_charge(layout.size() as u64);
        System.alloc(layout)
    }
    unsafe fn alloc_zeroed(&self, layout: Layout) -> *mut u8 {
        obs_charge(layout.size() as u64);
        System.alloc_zeroed(layout)
    }
    unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) {
        obs_release(layout.size() as u64);
        System.dealloc(ptr, layout)
    }
    unsafe fn realloc(&self, ptr: *mut u8, layout: Layout, new_size: usize) -> *mut u8 {
        if new_size > layout.size() {
            obs_charge((new_size - layout.size()) as u64);
        } else {
            obs_release((layout.size() - new_size) as u64);
        }
        System.realloc(ptr, layout, new_size)
    }
}

#[global_allocator]
static OBSERVER: Observer = Observer;

type ObsJob = Box<dyn FnOnce() + Send>;

struct Worker {
    jobs: std::sync::mpsc::Sender<ObsJob>,
    slot: &'static Slot,
}

fn obs_spawn() -> Worker {
    let slot: &'static Slot = Box::leak(Box::new(Slot {
        state: Mutex::new(None),
        ready: std::sync::Condvar::new(),
    }));
    let (jobs, rx) = std::sync::mpsc::channel::<ObsJob>();
    std::thread::Builder::new()
        .name("warrant-observer".into())
        .stack_size(16 << 20)
        .spawn(move || {
            WIN.with(|w| w.slot.set(Some(slot)));
            while let Ok(job) = rx.recv() {
                job();
            }
        })
        .expect("cannot start the allocation observer worker");
    Worker { jobs, slot }
}

static OBS_ABANDONED: std::sync::atomic::AtomicUsize = std::sync::atomic::AtomicUsize::new(0);
static OBS_HELD: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);

thread_local! {
    static OBS_WORKER: std::cell::RefCell<Option<Worker>> = const { std::cell::RefCell::new(None) };
}

/// The result of one observed call.
enum Exec {
    /// The call returned (or panicked) inside the cap; the peak it reached.
    Finished(Result<V, Box<dyn std::any::Any + Send>>, u64),
    /// The call asked for more than the cap; the size it reached.
    Tripped(u64),
    /// Too many workers (or bytes) are already abandoned: the call was not started.
    Refused,
}

/// Run `job` on the observer worker inside an allocation window.
fn obs_exec(job: impl FnOnce() -> V + Send + 'static) -> Exec {
    if OBS_ABANDONED.load(std::sync::atomic::Ordering::SeqCst) >= OBS_MAX_ABANDONED
        || OBS_HELD.load(std::sync::atomic::Ordering::SeqCst) >= OBS_MAX_HELD
    {
        return Exec::Refused;
    }
    OBS_WORKER.with(|cell| {
        let mut held = cell.borrow_mut();
        let worker = held.get_or_insert_with(obs_spawn);
        let slot = worker.slot;
        *slot.state.lock().unwrap() = None;
        let wrapped: ObsJob = Box::new(move || {
            WIN.with(|w| {
                w.cur.set(0);
                w.peak.set(0);
                w.active.set(true);
            });
            let r = catch_unwind(AssertUnwindSafe(job));
            let peak = WIN.with(|w| {
                w.active.set(false);
                w.peak.get()
            });
            *slot.state.lock().unwrap() = Some(ObsReply::Done(r, peak));
            slot.ready.notify_one();
        });
        worker.jobs.send(wrapped).expect("the observer worker is alive");
        let mut g = slot.state.lock().unwrap();
        while g.is_none() {
            g = slot.ready.wait(g).unwrap();
        }
        let reply = g.take().expect("reply present");
        drop(g);
        match reply {
            ObsReply::Done(r, peak) => Exec::Finished(r, peak),
            ObsReply::Tripped(peak, bytes) => {
                // The worker is parked inside the allocator for good.
                *held = None;
                OBS_ABANDONED.fetch_add(1, std::sync::atomic::Ordering::SeqCst);
                OBS_HELD.fetch_add(bytes, std::sync::atomic::Ordering::SeqCst);
                Exec::Tripped(peak)
            }
        }
    })
}

/// What the observer saw during one test (an example, a suite, a table).
#[derive(Clone, Debug, Default)]
struct ObsAgg {
    instances: u64,
    max_peak: u64,
    bound_at_max: Option<i128>,
    /// The first failure the observer raised during the test.
    note: String,
    /// An earlier instance of this test exceeded the cap: later observed
    /// instances fail without running.
    latched: bool,
}

thread_local! {
    static OBS_AGG: std::cell::RefCell<ObsAgg> = std::cell::RefCell::new(ObsAgg::default());
}

static OBS_STATS: Mutex<BTreeMap<String, ObsAgg>> = Mutex::new(BTreeMap::new());

fn obs_begin_test() {
    OBS_AGG.with(|a| *a.borrow_mut() = ObsAgg::default());
}

fn obs_end_test() -> ObsAgg {
    OBS_AGG.with(|a| std::mem::take(&mut *a.borrow_mut()))
}

fn obs_record(peak: u64, bound: Option<i128>) {
    OBS_AGG.with(|a| {
        let mut a = a.borrow_mut();
        a.instances += 1;
        if peak >= a.max_peak {
            a.max_peak = peak;
            a.bound_at_max = bound;
        }
    });
}

fn obs_note(msg: &str) {
    OBS_AGG.with(|a| {
        let mut a = a.borrow_mut();
        if a.note.is_empty() {
            a.note = msg.to_string();
        }
    });
}

/// Fail the current instance (caught by the claim or example wrapper).
fn obs_fail(msg: String) -> ! {
    std::panic::panic_any(ObsFailure(msg))
}

/// A claim or example wrapper caught an [`ObsFailure`]: remember why, and
/// report the instance as failed.
fn obs_failed(payload: Box<dyn std::any::Any + Send>) -> bool {
    match payload.downcast::<ObsFailure>() {
        Ok(f) => obs_note(&f.0),
        Err(other) => std::panic::resume_unwind(other),
    }
    false
}

/// Evaluate a bound expression; a negative or unevaluable bound is a defect of
/// the document and fails the instance (VER04j section 14.2 item 31).
fn obs_bound(what: &str, f: impl FnOnce() -> V) -> i128 {
    match catch_unwind(AssertUnwindSafe(|| f().int())) {
        Ok(n) if n >= 0 => n,
        Ok(n) => obs_fail(format!("bound-negative: the {what} is {n} at this instance")),
        Err(_) => obs_fail(format!("bound-negative: the {what} cannot be evaluated at this instance")),
    }
}

/// An operation's adapter, observed: run `job` (the real adapter) in a window,
/// then compare the measured peak with the `heap` row evaluated at the
/// arguments.  A call that exceeds the cap or the row fails the instance.
fn obs_observe(op: &str, heap: Option<i128>, job: impl FnOnce() -> V + Send + 'static) -> V {
    if OBS_AGG.with(|a| a.borrow().latched) {
        obs_fail(format!(
            "bound-heap-exceeded: `{op}` not run, an earlier instance of this test exceeded the allocation cap"
        ));
    }
    match obs_exec(job) {
        Exec::Finished(Ok(v), peak) => {
            obs_record(peak, heap);
            if let Some(b) = heap {
                if i128::from(peak) > b {
                    obs_fail(format!(
                        "bound-heap-exceeded: `{op}` reached {peak} bytes against a heap bound of {b}"
                    ));
                }
            }
            v
        }
        Exec::Finished(Err(p), peak) => {
            obs_record(peak, heap);
            std::panic::resume_unwind(p)
        }
        Exec::Tripped(peak) => {
            obs_record(peak, heap);
            OBS_AGG.with(|a| a.borrow_mut().latched = true);
            obs_fail(format!(
                "bound-heap-exceeded: `{op}` asked for {peak} bytes; the allocation cap is {OBSERVER_CAP_BYTES}"
            ))
        }
        Exec::Refused => {
            OBS_AGG.with(|a| a.borrow_mut().latched = true);
            obs_fail(format!(
                "bound-heap-exceeded: `{op}` not run, earlier calls in this process already exceeded the allocation cap"
            ))
        }
    }
}

/// The size of a successful result (VER04j section 6.2): the byte length of a
/// `Bytes`, the element count of a list, the size of the `Ok` payload of a
/// `Result`; an `Err` has size 0.
fn obs_size(v: &V) -> i128 {
    match v {
        V::Bytes(b) => b.len() as i128,
        V::List(l) => l.len() as i128,
        V::Ok(x) => obs_size(x),
        V::Err(_) => 0,
        other => panic!("`output` is not defined for {other:?}"),
    }
}

/// The generated property of an `output` row: `size(result) <= bound`.
fn obs_output_ok(result: V, bound: i128) -> bool {
    let size = obs_size(&result);
    if size > bound {
        obs_note(&format!("output-bound-exceeded: result size {size} is over the bound {bound}"));
        return false;
    }
    true
}

/// A failed observed suite keeps the observer's first note in its message.
fn obs_annotate(v: Verdict, note: &str) -> Verdict {
    match v {
        Verdict::Fail { failure_id, message } if !note.is_empty() => {
            Verdict::fail(failure_id, format!("{note}; {message}"))
        }
        other => other,
    }
}

/// Fault `truncate-silently`: an `Err` becomes `Ok` of the first `limit` bytes
/// of `source` (VER04j section 14.2 item 36).
fn fault_truncate_silently(v: V, source: &[u8], limit: i128) -> V {
    match v {
        V::Err(_) => {
            let n = usize::try_from(limit.max(0)).unwrap_or(usize::MAX).min(source.len());
            V::Ok(Box::new(V::Bytes(source[..n].to_vec())))
        }
        other => other,
    }
}

/// Relation bias for a limit parameter (VER04j section 6.3): when a claim has a
/// `Bytes` binder and an unsigned numeric binder, half of the draws replace the
/// number by a value that sits on the boundary of the byte string's length.
fn obs_limit_bias(tys: &[Ty], env: &mut [V], src: &mut dyn Src) {
    let Some(b) = tys.iter().position(|t| matches!(t, Ty::Bytes)) else {
        return;
    };
    let len = match &env[b] {
        V::Bytes(x) => x.len() as i128,
        _ => return,
    };
    for (j, t) in tys.iter().enumerate() {
        let max = match t {
            Ty::Fx(f) | Ty::FxEdge(f) if matches!(f, F::U16 | F::U32 | F::U64) => f.max(),
            Ty::Nat => i128::from(u32::MAX),
            _ => continue,
        };
        if src.bounded(2) != 0 {
            continue;
        }
        let pick = match src.bounded(9) {
            0 => 0,
            1 => 1,
            2 => (len - 1).max(0),
            3 => len,
            4 => 2 * len,
            5 => 255,
            6 => 256,
            7 => 65535,
            _ => max,
        }
        .min(max);
        env[j] = match t {
            Ty::Fx(f) | Ty::FxEdge(f) => V::Fixed(*f, pick),
            _ => V::Int(pick),
        };
    }
}

// WX-E: pinned known-answer tables (VER04j section 5).

/// Counters of one table run, for the manifest.
#[derive(Clone, Debug, Default)]
pub struct TableStats {
    /// Rows run.
    pub cases: u64,
    /// Rows the document pins.
    pub rows_pinned: u64,
    /// SHA-256 of the checked-in copy.
    pub sha256: String,
    /// Rows whose `holds` was false (or that panicked).
    pub failed: u64,
    /// The first 16 failing rows as `vectors/<table>/<row id>`; empty in CI mode
    /// (VER00 section 7.3: the data is not echoed).
    pub failed_ids: Vec<String>,
}

/// Stats of the fault-free run of each table, by table name.
static TABLE_STATS: Mutex<BTreeMap<String, TableStats>> = Mutex::new(BTreeMap::new());

/// Decode one TFV value as a model value of type `ty` (VER04j section 4.9).
/// Integers are range-checked, `Bytes<N>` length-checked; constructors are
/// found by *name*.  The generator already verified every row of the file, so
/// a failure here means the checked-in copy was altered after generation.
fn table_value(v: &Value, ty: Ty, depth: usize) -> Result<V, String> {
    if depth > 64 {
        return Err("nests too deeply".into());
    }
    let int = |t: &str| t.parse::<i128>().map_err(|_| "integer out of range".to_string());
    match (ty, v) {
        (Ty::Bool, Value::Bool(b)) => Ok(V::Bool(*b)),
        (Ty::Fx(f) | Ty::FxEdge(f), Value::Int(t)) => {
            let n = int(t)?;
            if n < f.min() || n > f.max() {
                return Err("integer out of range".into());
            }
            Ok(V::Fixed(f, n))
        }
        (Ty::Nat | Ty::NatCount, Value::Int(t)) => {
            let n = int(t)?;
            if n < 0 {
                return Err("negative Nat".into());
            }
            Ok(V::Int(n))
        }
        (Ty::Bytes, Value::Bytes(b)) => Ok(V::Bytes(b.clone())),
        (Ty::BytesN(n), Value::Bytes(b)) if b.len() == n => Ok(V::Bytes(b.clone())),
//@if comp
        (Ty::Tuple(cs), Value::List(items)) if items.len() == cs.len() => {
            let mut out = Vec::with_capacity(cs.len());
            for (c, x) in cs.iter().zip(items) {
                out.push(table_value(x, *c, depth + 1)?);
            }
            Ok(V::Tup(out))
        }
        (Ty::Opt(t), Value::List(items)) => match items.as_slice() {
            [] => Ok(V::Opt(None)),
            [x] => Ok(V::Opt(Some(Box::new(table_value(x, *t, depth + 1)?)))),
            _ => Err("an Option is L0 or L1 x".into()),
        },
        (Ty::List(t), Value::List(items)) => {
            let mut out = Vec::with_capacity(items.len());
            for x in items {
                out.push(table_value(x, *t, depth + 1)?);
            }
            Ok(V::List(out))
        }
        // A record (one constructor) is its field list; a variant is
        // `L2 S<ctor> L<k> fields`, by constructor *name*.
        (Ty::Data(name, ctors), Value::List(items)) if ctors.len() == 1 => {
            let (cname, shapes) = ctors[0];
            if items.len() != shapes.len() {
                return Err("wrong number of fields".into());
            }
            let mut out = Vec::with_capacity(shapes.len());
            for (s, x) in shapes.iter().zip(items) {
                out.push(table_value(x, *s, depth + 1)?);
            }
            Ok(V::Con(name, 0, cname, out))
        }
        (Ty::Data(name, ctors), Value::List(items)) => {
            let [Value::Text(c), Value::List(fields)] = items.as_slice() else {
                return Err("a variant is L2 S<ctor> L<k> ...".into());
            };
            let Some(i) = ctors.iter().position(|(n, _)| *n == c.as_str()) else {
                return Err("unknown constructor".into());
            };
            let (cname, shapes) = ctors[i];
            if fields.len() != shapes.len() {
                return Err("wrong number of fields".into());
            }
            let mut out = Vec::with_capacity(shapes.len());
            for (s, x) in shapes.iter().zip(fields) {
                out.push(table_value(x, *s, depth + 2)?);
            }
            Ok(V::Con(name, i, cname, out))
        }
        (Ty::Res(t, cases), Value::Map(pairs)) => match pairs.as_slice() {
            [(Value::Text(k), payload)] if k == "ok" => {
                Ok(V::Ok(Box::new(table_value(payload, *t, depth + 1)?)))
            }
            [(Value::Text(k), Value::Text(case))] if k == "err" && cases.contains(&case.as_str()) => {
                Ok(V::Err(case.clone()))
            }
            _ => Err("a Result is M1 S\"ok\" v or M1 S\"err\" S<CASE>".into()),
        },
//@endif
        _ => Err("a value that does not fit its column".into()),
    }
}

/// Run one table: check the pin, read the rows, evaluate `holds` for each.
pub fn run_table(
    name: &str,
    bytes: &[u8],
    pinned: &str,
    rows_pinned: usize,
    cols: &[(&str, Ty)],
    holds: Holds,
    fault: Fault,
) -> (Verdict, TableStats) {
    let mut stats = TableStats {
        rows_pinned: rows_pinned as u64,
        sha256: sha256_hex(bytes),
        ..TableStats::default()
    };
    let malformed = |why: String| Verdict::fail("corpus-malformed", format!("table {name}: {why}"));
    // The pin comes first: nothing in the copy is read until it matches.
    if stats.sha256 != pinned {
        return (malformed("the checked-in copy does not match its sha256 pin".into()), stats);
    }
    let Ok(text) = std::str::from_utf8(bytes) else {
        return (malformed("the checked-in copy is not UTF-8".into()), stats);
    };
    let records = match tfv::parse_file(text) {
        Ok(r) => r,
        Err(e) => return (malformed(format!("malformed TFV, {e}")), stats),
    };
    if records.len() != rows_pinned {
        return (
            malformed(format!("{} records, {rows_pinned} pinned", records.len())),
            stats,
        );
    }
    let in_ci = std::env::var("CI").is_ok();
    for rec in &records {
        if rec.fields.len() != cols.len() {
            return (malformed(format!("row {:?} does not have the declared columns", rec.id)), stats);
        }
        let mut env = Vec::with_capacity(cols.len());
        for (cname, ty) in cols {
            let Some(value) = rec.field(cname) else {
                return (malformed(format!("row {:?} is missing a column", rec.id)), stats);
            };
            match table_value(value, *ty, 0) {
                Ok(v) => env.push(v),
                Err(why) => return (malformed(format!("row {:?}: {why}", rec.id)), stats),
            }
        }
        stats.cases += 1;
        // A panicking row fails (the subject is not trusted to be total); one
        // that cannot be handed to the host at all fails too.
        let held = catch_unwind(AssertUnwindSafe(|| holds(&env, fault))).unwrap_or(false);
        if !held {
            stats.failed += 1;
            if !in_ci && stats.failed_ids.len() < 16 {
                stats.failed_ids.push(format!("vectors/{name}/{}", rec.id));
            }
        }
    }
    if fault == Fault::None {
        TABLE_STATS.lock().unwrap().insert(name.to_string(), stats.clone());
    }
    if stats.failed == 0 {
        return (Verdict::Pass, stats);
    }
    let first = if stats.failed_ids.is_empty() {
        String::new()
    } else {
        format!("; first: {}", stats.failed_ids.join(", "))
    };
    (
        Verdict::fail(
            "not-deep-equal",
            format!("table {name}: {} of {} rows failed{first}", stats.failed, stats.cases),
        ),
        stats,
    )
}

//@if bytefaults
/// Fault `flip-result-bit`: the lowest bit of the first byte of a byte-string
/// result (inside an `Ok` or a `Some`) is flipped; an empty result gains a byte.
fn fault_flip_result_bit(v: V) -> V {
    match v {
        V::Bytes(mut b) => {
            match b.first_mut() {
                Some(x) => *x ^= 1,
                None => b.push(1),
            }
            V::Bytes(b)
        }
        V::Ok(x) => V::Ok(Box::new(fault_flip_result_bit(*x))),
        V::Opt(Some(x)) => V::Opt(Some(Box::new(fault_flip_result_bit(*x)))),
        other => other,
    }
}

/// Fault `drop-last-result-byte`: the last byte of a byte-string result is
/// lost; an empty result gains a byte.
fn fault_drop_last_result_byte(v: V) -> V {
    match v {
        V::Bytes(mut b) => {
            if b.pop().is_none() {
                b.push(0);
            }
            V::Bytes(b)
        }
        V::Ok(x) => V::Ok(Box::new(fault_drop_last_result_byte(*x))),
        V::Opt(Some(x)) => V::Opt(Some(Box::new(fault_drop_last_result_byte(*x)))),
        other => other,
    }
}

/// Binder types the generator understands.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Ty {
    Bool,
    U8,
    U64,
    Nat,
    Bytes,
    BytesN(usize),
//@if v1
    /// A fixed-width binder at its default domain: every value of a narrow
    /// type (`U8 I8 U16 I16`), the edge set of a wide one.
    Fx(F),
    /// A narrow binder demoted to its edge set by the budget rule (section 2.6).
    FxEdge(F),
    /// A `Nat` that flows directly into a shift count: `0..=64` plus the
    /// boundary counts `65 127 128 129 255 256 65535 65536` (section 3.6).
    NatCount,
//@endif
//@if comp
    /// WX-A: a tuple, the Cartesian product of its components.
    Tuple(&'static [Ty]),
    /// WX-A: a record (one constructor named like the type) or variant: the
    /// constructors with their field types, in declaration order.
    Data(&'static str, &'static [(&'static str, &'static [Ty])]),
    /// `Option<T>`: `None`, then `Some(x)` for every `x` of `T`.
    Opt(&'static Ty),
    /// `Result<T, E>`: `Ok(x)` for every `x` of `T`, then each error case.
    Res(&'static Ty, &'static [&'static str]),
    /// `List<T>`: every list of length 0..=2 over the first 16 elements of `T`,
    /// plus length-3 lists over its first 4.
    List(&'static Ty),
//@endif
}

const ALPHABET: [u8; 4] = [0x00, 0x01, 0x80, 0xff];

fn alpha_strings(len: usize) -> Vec<Vec<u8>> {
    let total = 4usize.pow(len as u32);
    (0..total)
        .map(|mut n| {
            let mut s = vec![0u8; len];
            for slot in s.iter_mut() {
                *slot = ALPHABET[n % 4];
                n /= 4;
            }
            s
        })
        .collect()
}

/// The finite domain of one binder (VER04h section 3).
pub fn domain(ty: Ty) -> Vec<V> {
    match ty {
        Ty::Bool => vec![V::Bool(false), V::Bool(true)],
//@if v0
        Ty::U8 => (0..=255u64).map(V::Nat).collect(),
        Ty::Nat => (0..=64u64).map(V::Nat).collect(),
        Ty::U64 => {
            let mut v: Vec<u64> = (0..=255u64).collect();
            v.extend((8..64).map(|i| 1u64 << i));
            v.push(u64::MAX);
            v.extend((0..64).map(|i| u64::MAX ^ (1u64 << i)));
            v.into_iter().map(V::U64).collect()
        }
//@endif
//@if v1
        Ty::U8 | Ty::U64 => unreachable!("a v1 document binds U8/U64 as Ty::Fx"),
        Ty::Nat => (0..=64i128).map(V::Int).collect(),
        Ty::NatCount => (0..=64i128)
            .chain([65, 127, 128, 129, 255, 256, 65535, 65536])
            .map(V::Int)
            .collect(),
        Ty::Fx(f) if f.bits() <= 16 => fx_full(f).into_iter().map(|v| V::Fixed(f, v)).collect(),
        Ty::Fx(f) | Ty::FxEdge(f) => fx_edge(f).into_iter().map(|v| V::Fixed(f, v)).collect(),
//@endif
//@if comp
        Ty::Tuple(_) | Ty::Data(..) | Ty::Opt(_) | Ty::Res(..) | Ty::List(_) => {
            domain_take(ty, domain_size(ty).map_or(0, |n| n as usize))
        }
//@endif
        Ty::Bytes => {
            let mut v: Vec<Vec<u8>> = vec![Vec::new()];
            v.extend((0..=255u8).map(|b| vec![b]));
            v.extend(alpha_strings(2));
            v.extend(alpha_strings(3));
            for len in 4..=8 {
                v.push(vec![0x00; len]);
                v.push(vec![0xa5; len]);
            }
            v.into_iter().map(V::Bytes).collect()
        }
        Ty::BytesN(0) => vec![V::Bytes(Vec::new())],
        Ty::BytesN(1) => (0..=255u8).map(|b| V::Bytes(vec![b])).collect(),
        Ty::BytesN(n) if n <= 3 => alpha_strings(n).into_iter().map(V::Bytes).collect(),
        Ty::BytesN(n) => {
            let mut v: Vec<Vec<u8>> = ALPHABET.iter().map(|&s| vec![s; n]).collect();
            for pos in 0..n {
                for &s in &ALPHABET[1..] {
                    let mut x = vec![0u8; n];
                    x[pos] = s;
                    v.push(x);
                }
            }
            v.into_iter().map(V::Bytes).collect()
        }
    }
}

/// Closed-form size of [`domain`], `None` on `u64` overflow or an over-cap
/// width.  Must agree with the generator's `Ty::domain_size`; the generator
/// test suite checks `domain_size(t) == domain(t).len()` for every type.
pub fn domain_size(ty: Ty) -> Option<u64> {
    match ty {
        Ty::Bool => Some(2),
//@if v0
        Ty::U8 => Some(256),
        Ty::Nat => Some(65),
        Ty::U64 => Some(256 + 56 + 1 + 64),
//@endif
//@if v1
        Ty::U8 | Ty::U64 => None,
        Ty::Nat => Some(65),
        Ty::NatCount => Some(65 + 8),
        Ty::Fx(f) if f.bits() <= 16 => Some(1u64 << f.bits()),
        Ty::Fx(f) | Ty::FxEdge(f) => Some(fx_edge(f).len() as u64),
//@endif
//@if comp
        Ty::Tuple(cs) => product_size(cs),
        Ty::Data(_, ctors) => {
            let mut total = 0u64;
            for (_, fields) in ctors {
                total = total.checked_add(product_size(fields)?)?;
            }
            Some(total)
        }
        Ty::Opt(t) => domain_size(*t)?.checked_add(1),
        Ty::Res(t, cases) => domain_size(*t)?.checked_add(cases.len() as u64),
        Ty::List(t) => {
            let n = domain_size(*t)?;
            let (d, e) = (n.min(16), n.min(4));
            1u64.checked_add(d)?.checked_add(d * d)?.checked_add(e * e * e)
        }
//@endif
        Ty::Bytes => Some(1 + 256 + 16 + 64 + 10),
        Ty::BytesN(n) if n as u64 > MAX_FIXED_WIDTH => None,
        Ty::BytesN(0) => Some(1),
        Ty::BytesN(1) => Some(256),
        Ty::BytesN(n) if n <= 3 => 4u64.checked_pow(n as u32),
        Ty::BytesN(n) => 3u64.checked_mul(n as u64)?.checked_add(4),
    }
}

/// Upper bound on the memory [`domain`] materialises, in bytes.
pub fn domain_bytes(ty: Ty) -> Option<u64> {
//@if comp
    if is_comp(ty) {
        return domain_size(ty)?.checked_mul(elem_cost(ty)?);
    }
//@endif
    let payload = match ty {
        Ty::Bytes => 8,
        Ty::BytesN(n) => n as u64,
        _ => 0,
    };
    domain_size(ty)?.checked_mul(payload.checked_add(VALUE_OVERHEAD)?)
}

//@if comp
fn is_comp(ty: Ty) -> bool {
    matches!(ty, Ty::Tuple(_) | Ty::Data(..) | Ty::Opt(_) | Ty::Res(..) | Ty::List(_))
}

/// Size of the Cartesian product of component domains (`None` on overflow).
fn product_size(cs: &[Ty]) -> Option<u64> {
    let mut n = 1u64;
    for c in cs {
        n = n.checked_mul(domain_size(*c)?)?;
    }
    Some(n)
}

/// The most bytes one element of the domain of `ty` can occupy: every node
/// costs [`VALUE_OVERHEAD`], a `Bytes` payload its length.  An upper bound, so
/// `size x elem_cost` over-estimates the memory a domain needs (VER04j
/// section 14.1, item 20).
fn elem_cost(ty: Ty) -> Option<u64> {
    let node = VALUE_OVERHEAD;
    match ty {
        Ty::Bytes => node.checked_add(8),
        Ty::BytesN(n) => node.checked_add(n as u64),
        Ty::Tuple(cs) => {
            let mut c = node;
            for t in cs {
                c = c.checked_add(elem_cost(*t)?)?;
            }
            Some(c)
        }
        Ty::Data(_, ctors) => {
            let mut worst = 0u64;
            for (_, fields) in ctors {
                let mut c = 0u64;
                for t in *fields {
                    c = c.checked_add(elem_cost(*t)?)?;
                }
                worst = worst.max(c);
            }
            node.checked_add(worst)
        }
        Ty::Opt(t) => node.checked_add(elem_cost(*t)?),
        Ty::Res(t, _) => node.checked_add(elem_cost(*t)?.max(2 * node)),
        Ty::List(t) => node.checked_add(elem_cost(*t)?.checked_mul(3)?),
        _ => Some(node),
    }
}

/// The first `k` elements of the domain of `ty`, in domain order, without
/// materialising the rest: tuples and records are the Cartesian product of
/// their components (lexicographic, the last component varying fastest), a
/// variant the concatenation of its constructors' products in declaration
/// order, `Option<T>` is `None` then `Some(x)`, `Result<T, E>` is `Ok(x)` then
/// each error case, `List<T>` is the lists of length 0, 1, 2 over the first 16
/// elements of `T` and then length 3 over its first 4 (VER04j section 4.7).
pub fn domain_take(ty: Ty, k: usize) -> Vec<V> {
    if k == 0 {
        return Vec::new();
    }
    match ty {
        Ty::Tuple(cs) => product(cs, k).into_iter().map(V::Tup).collect(),
        Ty::Data(name, ctors) => {
            let mut out = Vec::new();
            for (i, (cname, fields)) in ctors.iter().enumerate() {
                if out.len() >= k {
                    break;
                }
                for fs in product(fields, k - out.len()) {
                    out.push(V::Con(name, i, cname, fs));
                }
            }
            out
        }
        Ty::Opt(t) => {
            let mut out = vec![V::Opt(None)];
            out.extend(domain_take(*t, k - 1).into_iter().map(|x| V::Opt(Some(Box::new(x)))));
            out
        }
        Ty::Res(t, cases) => {
            let mut out: Vec<V> = domain_take(*t, k).into_iter().map(|x| V::Ok(Box::new(x))).collect();
            for c in cases.iter() {
                if out.len() >= k {
                    break;
                }
                out.push(V::Err(c.to_string()));
            }
            out
        }
        Ty::List(t) => {
            let wide = domain_take(*t, 16);
            let narrow = domain_take(*t, 4);
            let mut out: Vec<V> = vec![V::List(Vec::new())];
            for a in &wide {
                out.push(V::List(vec![a.clone()]));
            }
            for a in &wide {
                for b in &wide {
                    out.push(V::List(vec![a.clone(), b.clone()]));
                }
            }
            for a in &narrow {
                for b in &narrow {
                    for c in &narrow {
                        out.push(V::List(vec![a.clone(), b.clone(), c.clone()]));
                    }
                }
            }
            out.truncate(k);
            out
        }
        _ => {
            let mut d = domain(ty);
            d.truncate(k);
            d
        }
    }
}

/// The first `k` tuples of the lexicographic product of the components.
fn product(comps: &[Ty], k: usize) -> Vec<Vec<V>> {
    let Some((first, rest_comps)) = comps.split_first() else {
        return vec![Vec::new()];
    };
    let rest_size = product_size(rest_comps).map_or(usize::MAX, |n| n.min(usize::MAX as u64) as usize);
    let rest = product(rest_comps, k.min(rest_size));
    let heads = domain_take(*first, k.div_ceil(rest.len().max(1)));
    let mut out = Vec::new();
    'outer: for a in heads {
        for r in &rest {
            let mut t = Vec::with_capacity(1 + r.len());
            t.push(a.clone());
            t.extend(r.iter().cloned());
            out.push(t);
            if out.len() == k {
                break 'outer;
            }
        }
    }
    out
}

//@endif
/// Whether the Cartesian product of the binder domains fits both budgets.
/// Every step is checked: a wrapped product (eight `U8` binders is `2^64`,
/// which is `0` in `u64`) must count as over budget, never as zero.
pub fn within_budget(tys: &[Ty]) -> bool {
    let mut product = 1u64;
    let mut bytes = 0u64;
    for t in tys {
        let (Some(size), Some(b)) = (domain_size(*t), domain_bytes(*t)) else {
            return false;
        };
        let (Some(p), Some(total)) = (product.checked_mul(size), bytes.checked_add(b)) else {
            return false;
        };
        product = p;
        bytes = total;
    }
    product <= EXHAUSTIVE_BUDGET && bytes <= DOMAIN_BYTES_BUDGET
}

fn describe(tys: &[Ty]) -> String {
    tys.iter()
        .map(|t| format!("{:?}[{}]", t, domain_size(*t).map_or_else(|| "over-budget".to_string(), |n| n.to_string())))
        .collect::<Vec<_>>()
        .join(" x ")
}

//@if v1
/// The manifest `domain` text of a claim (VER04j section 2.6): per binder its
/// type, whether the domain is `full` (every value), `edge` (the biased
/// boundary set), `count` (the shift-count domain) or `set` (a fixed finite
/// set), and its size, for example `a: U16 full(65536); b: U16 edge(281)`.
fn describe(tys: &[Ty], names: &[&str]) -> String {
    tys.iter()
        .zip(names)
        .map(|(t, n)| {
//@if comp
            if is_comp(*t) {
                return format!("{n}: {}", describe_ty(*t));
            }
//@endif
            let size = domain_size(*t).map_or_else(|| "over-budget".to_string(), |n| n.to_string());
            let (ty, kind) = match t {
                Ty::Fx(f) if f.bits() <= 16 => (format!("{f:?}"), "full"),
                Ty::Fx(f) | Ty::FxEdge(f) => (format!("{f:?}"), "edge"),
                Ty::NatCount => ("Nat".to_string(), "count"),
                Ty::Nat => ("Nat".to_string(), "set"),
                other => (format!("{other:?}"), "set"),
            };
            format!("{n}: {ty} {kind}({size})")
        })
        .collect::<Vec<_>>()
        .join("; ")
}
//@endif

//@if comp
/// The manifest text of one type's domain (VER04j section 4.7), for example
/// `tuple(U64 edge(377), Nat set(65)) size(24505)`.
fn describe_ty(ty: Ty) -> String {
    let size = domain_size(ty).map_or_else(|| "over-budget".to_string(), |n| n.to_string());
    let leaf = |name: String, kind: &str| format!("{name} {kind}({size})");
    match ty {
        Ty::Fx(f) if f.bits() <= 16 => leaf(format!("{f:?}"), "full"),
        Ty::Fx(f) | Ty::FxEdge(f) => leaf(format!("{f:?}"), "edge"),
        Ty::NatCount => leaf("Nat".to_string(), "count"),
        Ty::Nat => leaf("Nat".to_string(), "set"),
        Ty::Tuple(cs) => {
            let parts: Vec<String> = cs.iter().map(|c| describe_ty(*c)).collect();
            format!("tuple({}) size({size})", parts.join(", "))
        }
        Ty::Data(name, ctors) if ctors.len() == 1 => {
            let parts: Vec<String> = ctors[0].1.iter().map(|c| describe_ty(*c)).collect();
            format!("record {name}({}) size({size})", parts.join(", "))
        }
        Ty::Data(name, ctors) => format!("variant {name}[{} ctors] size({size})", ctors.len()),
        Ty::Opt(t) => format!("Option<{}> size({size})", describe_ty(*t)),
        Ty::Res(t, cases) => {
            format!("Result<{}, {} cases> size({size})", describe_ty(*t), cases.len())
        }
        Ty::List(t) => format!(
            "List<{}> size({size}) [len 0..=2 over the first 16, len 3 over the first 4]",
            describe_ty(*t)
        ),
        other => leaf(format!("{other:?}"), "set"),
    }
}

/// Per-test counters that end up in the manifest.
#[derive(Clone, Debug, Default)]
pub struct Stats {
    pub cases: u64,
    pub discarded: u64,
    pub iterations: u64,
    pub admitted: u64,
    /// Property cases whose first two binders were equal (the equal branch).
    pub equal: u64,
//@if comp
    /// Cases skipped because a host integer cannot hold a generated value.
    pub unreachable: u64,
//@endif
}

fn counterexample_message(what: &str, env: &[V]) -> String {
    let repr = format!("{env:?}");
    if std::env::var("CI").is_ok() {
        format!("{what}; repro-fnv: {:016x}", fnv1a64(repr.as_bytes()))
    } else {
        format!("{what}; counterexample: {repr}")
    }
}

type Admit = fn(&[V]) -> bool;
type Holds = fn(&[V], Fault) -> bool;

/// Bounded-exhaustive enumeration of the Cartesian product of binder domains.
pub fn exhaustive(tys: &[Ty], admit: Admit, holds: Holds, fault: Fault) -> (Verdict, Stats) {
    let mut stats = Stats::default();
    // Check the budget BEFORE materialising any domain: a hostile width would
    // otherwise allocate gigabytes first and be refused afterwards.
    if !within_budget(tys) {
        return (
            Verdict::fail("corpus-malformed", "exhaustive domain exceeds the budget"),
            stats,
        );
    }
    let domains: Vec<Vec<V>> = tys.iter().map(|t| domain(*t)).collect();
    let mut idx = vec![0usize; domains.len()];
    loop {
        let env: Vec<V> = idx.iter().zip(&domains).map(|(i, d)| d[*i].clone()).collect();
        if admit(&env) {
            stats.cases += 1;
            if !holds(&env, fault) {
                return (
                    Verdict::fail(
                        "property-counterexample",
                        counterexample_message("exhaustive suite falsified the claim", &env),
                    ),
                    stats,
                );
            }
        } else {
            stats.discarded += 1;
        }
        let mut pos = idx.len();
        loop {
            if pos == 0 {
                if stats.cases == 0 {
                    return (
                        Verdict::fail("property-exhausted", "refinement admitted no case"),
                        stats,
                    );
                }
                return (Verdict::Pass, stats);
            }
            pos -= 1;
            idx[pos] += 1;
            if idx[pos] < domains[pos].len() {
                break;
            }
            idx[pos] = 0;
        }
    }
}

// ----- relation-biased generation (shared by property and fuzz) -----------

trait Src {
    fn byte(&mut self) -> u8;
    fn bounded(&mut self, n: u64) -> u64;
    fn word(&mut self) -> u64;
}

struct RngSrc<'a>(&'a mut SplitMix64);
impl Src for RngSrc<'_> {
    fn byte(&mut self) -> u8 {
        self.0.bounded(256) as u8
    }
    fn bounded(&mut self, n: u64) -> u64 {
        self.0.bounded(n)
    }
    fn word(&mut self) -> u64 {
        self.0.next_u64()
    }
}

/// Fuzz input as a randomness source; exhausted input reads as zeros.
struct StreamSrc<'a> {
    data: &'a [u8],
    pos: usize,
}
impl Src for StreamSrc<'_> {
    fn byte(&mut self) -> u8 {
        let b = self.data.get(self.pos).copied().unwrap_or(0);
        self.pos += 1;
        b
    }
    fn bounded(&mut self, n: u64) -> u64 {
        u64::from(self.byte()) % n
    }
    fn word(&mut self) -> u64 {
        let mut w = 0u64;
        for i in 0..8 {
            w |= u64::from(self.byte()) << (8 * i);
        }
        w
    }
}

//@if v1
/// Draw a fixed-width value (VER04j section 2.6).  `U8` is uniform and `U64`
/// keeps the v0 generator (zero, MAX, powers of two, small values, uniform);
/// every other type draws 3/4 uniformly over its full range and 1/4 from its
/// edge set, so boundaries are reached without exhaustive enumeration.
fn fresh_fx(f: F, src: &mut dyn Src) -> V {
    let v = match f {
        F::U8 => i128::from(src.byte()),
        F::U64 => i128::from(match src.bounded(8) {
            0 => 0,
            1 => u64::MAX,
            2 => 1u64 << src.bounded(64),
            3..=5 => src.bounded(300),
            _ => src.word(),
        }),
        _ => {
            if src.bounded(4) == 0 {
                let edge = fx_edge(f);
                edge[(src.word() % edge.len() as u64) as usize]
            } else {
                f.canon(u128::from(src.word()))
            }
        }
    };
    V::Fixed(f, v)
}

//@endif
fn fresh(ty: Ty, src: &mut dyn Src) -> V {
    match ty {
        Ty::Bool => V::Bool(src.bounded(2) == 1),
//@if v0
        Ty::U8 => V::Nat(u64::from(src.byte())),
        Ty::Nat => V::Nat(src.bounded(65)),
        Ty::U64 => V::U64(match src.bounded(8) {
            0 => 0,
            1 => u64::MAX,
            2 => 1u64 << src.bounded(64),
//@if u8
            // Documents that mix U8 and U64 binders (an exponent next to a
            // byte) need small U64 values often enough for refinements such as
            // `e < 255` to be admitted.
            3..=5 => src.bounded(300),
//@endif
            _ => src.word(),
        }),
//@endif
//@if v1
        Ty::U8 | Ty::U64 => unreachable!("a v1 document binds U8/U64 as Ty::Fx"),
        Ty::Nat => V::Int(i128::from(src.bounded(65))),
        // Half of all draws come from 0..=130 so that n = w and n = w + 1 are
        // reached for every width including 64 (VER04j section 3.6).
        Ty::NatCount => V::Int(i128::from(if src.bounded(2) == 0 {
            src.bounded(131)
        } else {
            src.bounded(65)
        })),
        Ty::Fx(f) | Ty::FxEdge(f) => fresh_fx(f, src),
//@endif
//@if comp
        // Components are drawn independently; constructors uniformly, so a rare
        // constructor is reached as often as a common one (section 4.7).
        Ty::Tuple(cs) => V::Tup(cs.iter().map(|c| fresh(*c, src)).collect()),
        Ty::Data(name, ctors) => {
            let i = src.bounded(ctors.len() as u64) as usize;
            let (cname, fields) = ctors[i];
            V::Con(name, i, cname, fields.iter().map(|t| fresh(*t, src)).collect())
        }
        Ty::Opt(t) => {
            if src.bounded(2) == 0 {
                V::Opt(None)
            } else {
                V::Opt(Some(Box::new(fresh(*t, src))))
            }
        }
        Ty::Res(t, cases) => {
            if cases.is_empty() || src.bounded(2) == 0 {
                V::Ok(Box::new(fresh(*t, src)))
            } else {
                V::Err(cases[src.bounded(cases.len() as u64) as usize].to_string())
            }
        }
        Ty::List(t) => {
            let n = src.bounded(4) as usize;
            V::List((0..n).map(|_| fresh(*t, src)).collect())
        }
//@endif
        Ty::Bytes => {
            let len = src.bounded(65) as usize;
            V::Bytes((0..len).map(|_| src.byte()).collect())
        }
        Ty::BytesN(n) => V::Bytes((0..n).map(|_| src.byte()).collect()),
    }
}

/// Derive a second binder from the first (mode 0..8, see VER04h section 4).
fn relate(prev: &V, ty: Ty, mode: u8, src: &mut dyn Src) -> V {
//@if comp
    if mode >= 4 {
        if let Some(v) = relate_comp(prev, ty, mode, src) {
            return v;
        }
    }
//@endif
    match (prev, mode) {
        (_, 0) | (_, 1) => fresh(ty, src),
        (p, 2) | (p, 3) => p.clone(),
        (V::Bytes(b), 4) | (V::Bytes(b), 5) if !b.is_empty() => {
            let mut o = b.clone();
            let bit = src.bounded(8 * o.len() as u64) as usize;
            o[bit / 8] ^= 1 << (bit % 8);
            V::Bytes(o)
        }
        (V::Bytes(b), 6) if !b.is_empty() => {
            let mut o = b.clone();
            let at = src.bounded(o.len() as u64) as usize;
            o[at] = o[at].wrapping_add(1 + src.bounded(255) as u8);
            V::Bytes(o)
        }
        (V::Bytes(b), 7) if !matches!(ty, Ty::BytesN(_)) => {
            let mut o = b.clone();
            if o.is_empty() || src.bounded(2) == 0 {
                o.push(src.byte());
            } else {
                o.pop();
            }
            V::Bytes(o)
        }
//@if v0
//@if u8
        // A small U64 pairs with another small U64, so refinements that relate
        // two exponents (`e < 255 and f < 255 and e != f`) are admitted.
        (V::U64(n), 6..=7) if *n < 1000 => V::U64(src.bounded(300)),
//@endif
        (V::U64(n), 4) | (V::U64(n), 5) => V::U64(n ^ (1u64 << src.bounded(64))),
        (V::U64(n), 6) => V::U64(n | (1u64 << src.bounded(64))),
//@if u8
        // A U8 binder stays inside 0..=255: flip one bit, or step by one mod 256.
        (V::Nat(n), 4..=5) if ty == Ty::U8 => V::Nat(n ^ (1u64 << src.bounded(8))),
        (V::Nat(n), 6..=7) if ty == Ty::U8 => V::Nat((n + 1) % 256),
//@endif
        (V::Nat(n), 4..=7) => V::Nat(n.wrapping_add(1)),
//@endif
//@if v1
        // A small U64 pairs with another small U64 (refinements such as
        // `e < 255 and f < 255 and e != f` must be admitted); every other
        // fixed-width value flips one bit (modes 4-5) or steps by one with
        // wrap-around (modes 6-7), so related pairs straddle every boundary.
        (V::Fixed(F::U64, n), 6..=7) if *n < 1000 => V::Fixed(F::U64, i128::from(src.bounded(300))),
        (V::Fixed(f, n), 4..=5) => V::Fixed(*f, f.canon(f.pattern(*n) ^ (1u128 << src.bounded(u64::from(f.bits()))))),
        (V::Fixed(f, n), 6..=7) => V::Fixed(*f, f.canon(f.pattern(*n).wrapping_add(1))),
        (V::Int(n), 4..=7) => V::Int(n + 1),
//@endif
        (p, _) => p.clone(),
    }
}

//@if comp
/// Relation bias on a composite: perturb *one* component (a leaf, recursively)
/// and keep everything else, so the related value differs in a single place
/// (section 4.7).  `None` when `prev` has nothing to perturb.
fn relate_comp(prev: &V, ty: Ty, mode: u8, src: &mut dyn Src) -> Option<V> {
    match (prev, ty) {
        (V::Tup(vs), Ty::Tuple(cs)) if !vs.is_empty() => {
            let i = src.bounded(vs.len() as u64) as usize;
            let mut o = vs.clone();
            o[i] = relate(&vs[i], cs[i], mode, src);
            Some(V::Tup(o))
        }
        (V::Con(n, i, c, vs), Ty::Data(_, ctors)) if !vs.is_empty() => {
            let j = src.bounded(vs.len() as u64) as usize;
            let mut o = vs.clone();
            o[j] = relate(&vs[j], ctors[*i].1[j], mode, src);
            Some(V::Con(n, *i, c, o))
        }
        (V::Opt(Some(x)), Ty::Opt(t)) => Some(V::Opt(Some(Box::new(relate(x, *t, mode, src))))),
        (V::Ok(x), Ty::Res(t, _)) => Some(V::Ok(Box::new(relate(x, *t, mode, src)))),
        (V::List(vs), Ty::List(t)) if !vs.is_empty() => {
            let i = src.bounded(vs.len() as u64) as usize;
            let mut o = vs.clone();
            o[i] = relate(&vs[i], *t, mode, src);
            Some(V::List(o))
        }
        _ => None,
    }
}

//@endif
fn build(tys: &[Ty], src: &mut dyn Src, mode: u8) -> Vec<V> {
    let mut env: Vec<V> = Vec::with_capacity(tys.len());
    for (i, ty) in tys.iter().enumerate() {
        let partner = (0..i).find(|j| tys[*j] == *ty);
        let v = match partner {
            Some(j) => relate(&env[j].clone(), *ty, mode, src),
            None => fresh(*ty, src),
        };
        env.push(v);
    }
//@if obs
    obs_limit_bias(tys, &mut env, src);
//@endif
    env
}

fn decode(input: &[u8], tys: &[Ty]) -> Vec<V> {
    let mode = input.first().copied().unwrap_or(0) % 8;
    let mut src = StreamSrc { data: input, pos: 1 };
    build(tys, &mut src, mode)
}

fn shrink_vals(cand: Vec<V>, prop_raw: &dyn Fn(&Vec<V>) -> bool) -> (Vec<V>, usize) {
    // A shrink candidate may leave the operation's domain (a `Bytes<N>` binder
    // made shorter panics inside the adapter).  Such a candidate is not a
    // counterexample, so a panic counts as "the claim holds".
    let prop = |t: &Vec<V>| catch_unwind(AssertUnwindSafe(|| prop_raw(t))).unwrap_or(true);
    let mut cur = cand;
    let mut steps = 0;
    for i in 0..cur.len() {
        let base = cur.clone();
        match &base[i] {
            V::Bytes(bytes) => {
                let (s, n) = shrink_bytes(
                    bytes,
                    |b| {
                        let mut t = base.clone();
                        t[i] = V::Bytes(b.to_vec());
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::Bytes(s);
                steps += n;
            }
//@if v0
            V::U64(v) => {
                let (s, n) = shrink_int(
                    *v as i128,
                    0,
                    0,
                    u64::MAX as i128,
                    |x| {
                        let mut t = base.clone();
                        t[i] = V::U64(x as u64);
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::U64(s as u64);
                steps += n;
            }
//@endif
//@if v1
            // Unsigned values shrink toward 0 (bisection, then single steps);
            // a signed value first tries its negation, then shrinks toward 0
            // from its own side.
            V::Fixed(f, v) => {
                let f = *f;
                let start = if *v < 0 && f.contains(-*v) {
                    let mut t = base.clone();
                    t[i] = V::Fixed(f, -*v);
                    if prop(&t) { *v } else { -*v }
                } else {
                    *v
                };
                let (s, n) = shrink_int(
                    start,
                    0,
                    f.min(),
                    f.max(),
                    |x| {
                        let mut t = base.clone();
                        t[i] = V::Fixed(f, x);
                        prop(&t)
                    },
                    1000,
                );
                cur[i] = V::Fixed(f, s);
                steps += n;
            }
//@endif
            _ => {}
        }
    }
    (cur, steps)
}

//@if comp
/// Shrink one value of type `ty` while `holds` keeps *failing* (a candidate is
/// accepted when the property is false for it).  Components shrink with their
/// own shrinkers; a variant first tries an earlier constructor with minimal
/// fields, then shrinks field-wise; `Option` shrinks toward `None`; a list
/// toward `[]`, then loses elements, then shrinks them (section 4.7).
fn shrink_value(ty: Ty, v: &V, holds: &dyn Fn(&V) -> bool, steps: &mut usize) -> V {
    if *steps >= 1000 {
        return v.clone();
    }
    let still_fails = |c: &V, steps: &mut usize| {
        *steps += 1;
        !holds(c)
    };
    match (v, ty) {
        (V::Tup(vs), Ty::Tuple(cs)) => {
            let mut cur = vs.clone();
            for i in 0..cur.len() {
                let base = cur.clone();
                let shrunk = shrink_value(
                    cs[i],
                    &base[i],
                    &|c: &V| {
                        let mut t = base.clone();
                        t[i] = c.clone();
                        holds(&V::Tup(t))
                    },
                    steps,
                );
                cur[i] = shrunk;
            }
            V::Tup(cur)
        }
        (V::Con(name, idx, cname, vs), Ty::Data(_, ctors)) => {
            for (j, (jname, jfields)) in ctors.iter().enumerate().take(*idx) {
                let minimal: Vec<V> = jfields.iter().map(|t| domain_take(*t, 1).remove(0)).collect();
                let cand = V::Con(name, j, jname, minimal);
                if still_fails(&cand, steps) {
                    return cand;
                }
            }
            let mut cur = vs.clone();
            for i in 0..cur.len() {
                let base = cur.clone();
                let shrunk = shrink_value(
                    ctors[*idx].1[i],
                    &base[i],
                    &|c: &V| {
                        let mut t = base.clone();
                        t[i] = c.clone();
                        holds(&V::Con(name, *idx, cname, t))
                    },
                    steps,
                );
                cur[i] = shrunk;
            }
            V::Con(name, *idx, cname, cur)
        }
        (V::Opt(Some(x)), Ty::Opt(t)) => {
            let none = V::Opt(None);
            if still_fails(&none, steps) {
                return none;
            }
            let inner = shrink_value(*t, x, &|c: &V| holds(&V::Opt(Some(Box::new(c.clone())))), steps);
            V::Opt(Some(Box::new(inner)))
        }
        (V::Ok(x), Ty::Res(t, _)) => {
            let inner = shrink_value(*t, x, &|c: &V| holds(&V::Ok(Box::new(c.clone()))), steps);
            V::Ok(Box::new(inner))
        }
        (V::List(vs), Ty::List(t)) => {
            let empty = V::List(Vec::new());
            if !vs.is_empty() && still_fails(&empty, steps) {
                return empty;
            }
            let mut cur = vs.clone();
            let mut i = 0;
            while i < cur.len() {
                let mut shorter = cur.clone();
                shorter.remove(i);
                if still_fails(&V::List(shorter.clone()), steps) {
                    cur = shorter;
                } else {
                    i += 1;
                }
            }
            for i in 0..cur.len() {
                let base = cur.clone();
                let shrunk = shrink_value(
                    *t,
                    &base[i],
                    &|c: &V| {
                        let mut l = base.clone();
                        l[i] = c.clone();
                        holds(&V::List(l))
                    },
                    steps,
                );
                cur[i] = shrunk;
            }
            V::List(cur)
        }
        // A scalar leaf: the v1 shrinker of fixed-width values, byte strings, ...
        _ => {
            let (r, n) = shrink_vals(vec![v.clone()], &|t: &Vec<V>| holds(&t[0]));
            *steps += n;
            r.into_iter().next().unwrap_or_else(|| v.clone())
        }
    }
}

/// Shrink a candidate environment binder by binder (composite-aware).
fn shrink_comp(tys: &[Ty], cand: Vec<V>, prop_raw: &dyn Fn(&Vec<V>) -> bool) -> (Vec<V>, usize) {
    // As in `shrink_vals`: a candidate outside the operation's domain panics in
    // the adapter and then counts as "the claim holds".
    let prop = |t: &Vec<V>| catch_unwind(AssertUnwindSafe(|| prop_raw(t))).unwrap_or(true);
    let mut cur = cand;
    let mut steps = 0;
    for i in 0..cur.len() {
        let base = cur.clone();
        cur[i] = shrink_value(
            tys[i],
            &base[i],
            &|c: &V| {
                let mut t = base.clone();
                t[i] = c.clone();
                prop(&t)
            },
            &mut steps,
        );
    }
    (cur, steps)
}

//@endif
/// L2 property suite over relation-biased generators.
pub fn property(tag: &str, tys: &[Ty], admit: Admit, holds: Holds, fault: Fault) -> (Verdict, Stats) {
    let seed = SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64();
    let options = PbtOptions {
        num_tests: PROPERTY_CASES as usize,
        max_retries: 100,
        max_shrink_steps: 1000,
        is_ci: std::env::var("CI").is_ok(),
    };
    let cases = Cell::new(0u64);
    let discarded = Cell::new(0u64);
    let equal = Cell::new(0u64);
    let gen = |rng: &mut SplitMix64| -> Result<Vec<V>, String> {
        for _ in 0..100 {
            let mode = rng.bounded(8) as u8;
            let env = build(tys, &mut RngSrc(rng), mode);
            if admit(&env) {
                cases.set(cases.get() + 1);
                if env.len() >= 2 && env[0] == env[1] {
                    equal.set(equal.get() + 1);
                }
                return Ok(env);
            }
            discarded.set(discarded.get() + 1);
        }
        Err(FID_EXHAUSTED.to_string())
    };
    let verdict = check_property(
        seed,
        &options,
        gen,
        move |env: &Vec<V>| holds(env, fault),
//@if plain
        shrink_vals,
//@endif
//@if comp
        |c: Vec<V>, p: &dyn Fn(&Vec<V>) -> bool| shrink_comp(tys, c, p),
//@endif
    );
    let stats = Stats {
        cases: cases.get(),
        discarded: discarded.get(),
        equal: equal.get(),
        ..Stats::default()
    };
    (verdict, stats)
}

fn cover(cov: &mut CoverageBitmap, env: &[V]) {
    for (i, v) in env.iter().enumerate() {
        match v {
            V::Bytes(b) => cov.record_edge(300 + b.len().min(64)),
//@if v0
            V::U64(n) => cov.record_edge(500 + n.count_ones() as usize),
//@endif
            V::Bool(x) => cov.record_edge(600 + usize::from(*x)),
//@if v0
//@if u8
            V::Nat(n) => cov.record_edge(2000 + (*n as usize).min(512)),
//@endif
//@endif
//@if v1
            V::Fixed(f, n) => cov.record_edge(500 + f.pattern(*n).count_ones() as usize),
            V::Int(n) => cov.record_edge(2000 + (*n).clamp(0, 512) as usize),
//@endif
//@if comp
            V::Con(_, c, _, _) => cov.record_edge(5000 + (*c).min(255)),
            V::Opt(o) => cov.record_edge(5300 + usize::from(o.is_some())),
            V::List(l) => cov.record_edge(5400 + l.len().min(16)),
//@endif
            _ => {}
        }
        for w in &env[i + 1..] {
            match (v, w) {
                (V::Bytes(a), V::Bytes(b)) => {
                    cov.record_edge(if a == b { 1 } else { 2 });
                    cov.record_edge(if a.len() == b.len() { 3 } else { 4 + (a.len() ^ b.len()) % 64 });
                    let pre = a.iter().zip(b.iter()).take_while(|(x, y)| x == y).count();
                    cov.record_edge(1000 + pre.min(64));
                }
//@if v0
                (V::U64(a), V::U64(b)) => {
                    cov.record_edge(if a == b { 700 } else { 701 + (a ^ b).count_ones() as usize });
                }
//@if u8
                (V::Nat(a), V::Nat(b)) => {
                    cov.record_edge(if a == b { 3000 } else { 3001 + (a ^ b).count_ones() as usize });
                }
//@endif
//@endif
//@if v1
                (V::Fixed(f, a), V::Fixed(_, b)) => {
                    cov.record_edge(if a == b { 700 } else { 701 + (f.pattern(*a) ^ f.pattern(*b)).count_ones() as usize });
                }
//@endif
                _ => {}
            }
        }
    }
}

/// Coverage-guided fuzz campaign; the claim body is the oracle.
pub fn fuzz_claim(
    tag: &str,
    tys: &[Ty],
    admit: Admit,
    holds: Holds,
    fault: Fault,
    iterations: u64,
) -> (Verdict, Stats) {
    let opts = FuzzOptions {
        max_iterations: iterations as usize,
        seed: SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64(),
        max_input_len: 256,
        base_energy: 10,
    };
    let corpus: Vec<Vec<u8>> = (0..8u8)
        .map(|m| {
            let mut s = vec![m, 4];
            s.extend((0..40u8).map(|i| i.wrapping_mul(37).wrapping_add(m)));
            s
        })
        .collect();
    let admitted = Cell::new(0u64);
    let report = fuzz::fuzz_target(&corpus, &opts, |input, cov| {
        let env = decode(input, tys);
        cover(cov, &env);
        if !admit(&env) {
            cov.record_edge(4000);
            return FuzzOutcome::Pass;
        }
        admitted.set(admitted.get() + 1);
        match catch_unwind(AssertUnwindSafe(|| holds(&env, fault))) {
            Ok(true) => FuzzOutcome::Pass,
            Ok(false) => FuzzOutcome::Crash {
                kind: "claim-violated".into(),
                frames: vec![tag.to_string()],
            },
            Err(_) => FuzzOutcome::Crash {
                kind: "panic".into(),
                frames: vec![tag.to_string()],
            },
        }
    });
    let stats = Stats {
        iterations: report.iterations as u64,
        admitted: admitted.get(),
        ..Stats::default()
    };
    if let Some(c) = report.crashes.first() {
        return (
            Verdict::fail(FID_FUZZ_CRASH, format!("fuzz campaign found {} ({} hits)", c.kind, c.hits)),
            stats,
        );
    }
    if stats.admitted == 0 {
        return (Verdict::fail("corpus-mismatch", "fuzz campaign admitted no input"), stats);
    }
    (Verdict::Pass, stats)
}

#[derive(Clone, Debug)]
struct Meta {
    id: String,
    kind: &'static str,
    op: String,
    claim: String,
    fault: String,
    target: String,
    domain: String,
    seed: String,
//@if v1
    /// Binders the budget rule demoted from `full` to `edge` (section 2.6).
    demoted: Vec<String>,
//@endif
}

static META: Mutex<Vec<Meta>> = Mutex::new(Vec::new());
static STATS: Mutex<BTreeMap<String, (Stats, String)>> = Mutex::new(BTreeMap::new());
/// SHA-256 of the crate source the target was built against (`run_main`'s
/// argument), recorded so a checker can detect a manifest older than the source.
static SOURCE_SHA256: Mutex<String> = Mutex::new(String::new());

fn reg(
    suite: &mut Suite,
    meta: Meta,
    body: impl Fn() -> (Verdict, Stats, String) + 'static,
) {
    // WARRANT_EVIDENCE_FILTER=<substring> keeps only matching test ids; it
    // exists so BUILD can run a sabotage mode against a cheap subset.  A
    // filtered run never writes a manifest (see `write_manifest`).
    if let Ok(filter) = std::env::var("WARRANT_EVIDENCE_FILTER") {
        if !meta.id.contains(&filter) {
            return;
        }
    }
    let id = meta.id.clone();
//@if obs
    let observed = meta.kind != "negctl";
//@endif
    META.lock().unwrap().push(meta);
    suite.add(id.clone(), move |_ctx| {
//@if obs
        obs_begin_test();
//@endif
        let (verdict, stats, note) = body();
//@if obs
        let agg = obs_end_test();
        let verdict = if observed { obs_annotate(verdict, &agg.note) } else { verdict };
        OBS_STATS.lock().unwrap().insert(id.clone(), agg);
//@endif
        STATS.lock().unwrap().insert(id.clone(), (stats, note));
        verdict
    });
}

fn is_detected(v: &Verdict) -> bool {
    match v {
        Verdict::Fail { failure_id, .. } => {
            coding_adventures_test_bootstrap::verdict::is_registered_failure_id(failure_id)
        }
        Verdict::Pass => false,
    }
}

fn seed_of(tag: &str) -> u64 {
    SplitMix64::new(EVIDENCE_SEED ^ fnv1a64(tag.as_bytes())).next_u64()
}

fn write_manifest(outcomes: &[Outcome]) {
    let Some(path) = manifest::output_path("WARRANT_EVIDENCE_OUT") else {
        return;
    };
    if std::env::var("WARRANT_EVIDENCE_FAULT").is_ok()
        || std::env::var("WARRANT_EVIDENCE_FILTER").is_ok()
    {
        eprintln!("refusing to write an evidence manifest from a sabotaged or filtered run");
        return;
    }
    let metas = META.lock().unwrap().clone();
    let stats = STATS.lock().unwrap().clone();
//@if obs
    let obs_stats = OBS_STATS.lock().unwrap().clone();
//@endif
    let mut tests: Vec<Json> = Vec::new();
    for m in metas {
        let verdict = outcomes
            .iter()
            .find(|o| o.id == m.id)
            .map(|o| if o.passed() { "pass" } else { "fail" })
            .unwrap_or("not-run");
        let (s, note) = stats.get(&m.id).cloned().unwrap_or_default();
//@if obs
        let obs_entry = obs_stats.get(&m.id).cloned();
//@endif
//@if vec
        let table_stat = TABLE_STATS.lock().unwrap().get(&m.claim).cloned();
//@endif
        let mut e: BTreeMap<String, Json> = BTreeMap::new();
        e.insert("id".into(), Json::Str(m.id));
        e.insert("kind".into(), Json::str(m.kind));
        e.insert("op".into(), Json::Str(m.op));
        e.insert("claim".into(), Json::Str(m.claim));
        e.insert("verdict".into(), Json::str(verdict));
        match m.kind {
            "example" => {
                e.insert("cases".into(), Json::Num(1));
            }
            "exhaustive" => {
                e.insert("cases".into(), Json::Num(s.cases));
                e.insert("discarded".into(), Json::Num(s.discarded));
                e.insert("domain".into(), Json::Str(m.domain));
//@if comp
                e.insert("unreachable_by_type".into(), Json::Num(s.unreachable));
//@endif
//@if v1
                e.insert(
                    "demoted".into(),
                    Json::Arr(m.demoted.into_iter().map(Json::Str).collect()),
                );
//@endif
            }
            "property" => {
                e.insert("cases".into(), Json::Num(s.cases));
                e.insert("discarded".into(), Json::Num(s.discarded));
                e.insert("equal_cases".into(), Json::Num(s.equal));
//@if comp
                e.insert("unreachable_by_type".into(), Json::Num(s.unreachable));
//@endif
                e.insert("seed".into(), Json::Str(m.seed));
            }
            "fuzz" => {
                e.insert("iterations".into(), Json::Num(s.iterations));
                e.insert("admitted".into(), Json::Num(s.admitted));
//@if comp
                e.insert("unreachable_by_type".into(), Json::Num(s.unreachable));
//@endif
                e.insert("seed".into(), Json::Str(m.seed));
            }
//@if vec
            "table" => {
                let t = table_stat.unwrap_or_default();
                e.insert("cases".into(), Json::Num(t.cases));
                e.insert("rows_pinned".into(), Json::Num(t.rows_pinned));
                e.insert("table_sha256".into(), Json::Str(t.sha256));
                if t.failed > 0 {
                    e.insert("failed_rows".into(), Json::Num(t.failed));
                    e.insert(
                        "failed_ids".into(),
                        Json::Arr(t.failed_ids.into_iter().map(Json::Str).collect()),
                    );
                }
            }
//@endif
            _ => {
                e.insert("fault".into(), Json::Str(m.fault));
                e.insert("target".into(), Json::Str(m.target));
                e.insert("caught_by".into(), Json::Str(note));
            }
        }
//@if obs
        if m.kind != "negctl" {
            if let Some(a) = obs_entry.filter(|a| a.instances > 0) {
                e.insert("instances_observed".into(), Json::Num(a.instances));
                e.insert("max_observed_heap".into(), Json::Num(a.max_peak));
                if let Some(b) = a.bound_at_max {
                    e.insert("bound_at_max".into(), Json::Num(u64::try_from(b).unwrap_or(u64::MAX)));
                }
            }
        }
//@endif
        tests.push(Json::Obj(e));
    }
    let doc = Json::obj([
        ("schema", Json::str("warrant-test-evidence-v1")),
//@if v1
        ("warrant_version", Json::str("v1")),
//@endif
        ("module", Json::str(MODULE)),
        ("spec", Json::str(SPEC_PATH)),
        ("spec_sha256", Json::str(SPEC_SHA256)),
        ("source_sha256", Json::Str(SOURCE_SHA256.lock().unwrap().clone())),
        ("evidence_seed", Json::Str(EVIDENCE_SEED.to_string())),
        ("tests", Json::Arr(tests)),
    ]);
    if let Err(e) = manifest::write(&path, &doc) {
        eprintln!("cannot write evidence manifest: {e}");
    }
}

/// Write the manifest of an in-process run (`Suite::run_recorded`) to
/// `WARRANT_EVIDENCE_OUT`; the generated test target does this through
/// `run_main`, a test of the generator does it through this wrapper.
pub fn write_manifest_for(outcomes: &[Outcome]) {
    write_manifest(outcomes);
}

//@endif
/// Decide the injected fault from the two sabotage variables.
///
/// `WARRANT_EVIDENCE_FILTER` exists only so BUILD can aim a *sabotage* run at
/// a cheap subset.  Without a real fault it would turn the target into a suite
/// that silently runs fewer tests and still exits 0, so a filter is an error
/// unless the fault is a real (non-`none`) one (audit A3; `FAULT=none` plus a
/// filter was the residual hole).  Pure function so it can be tested without
/// touching the process environment.
pub fn resolve_sabotage(fault: Option<&str>, filter: Option<&str>) -> Result<Fault, String> {
    let resolved = match fault {
        None => Fault::None,
        Some(text) => Fault::parse(text).ok_or_else(|| format!("unknown WARRANT_EVIDENCE_FAULT {text:?}"))?,
    };
    match (resolved, filter) {
        (Fault::None, Some(f)) => Err(format!(
            "WARRANT_EVIDENCE_FILTER={f:?} requires WARRANT_EVIDENCE_FAULT to name a real fault \
             (not unset or \"none\"); a filtered run is only meaningful as a sabotage run"
        )),
        (fault, _) => Ok(fault),
    }
}

/// A filtered suite that selects nothing proves nothing and must not pass.
pub fn require_tests(count: usize, filter: Option<&str>) -> Result<(), String> {
    match filter {
        Some(f) if count == 0 => Err(format!("WARRANT_EVIDENCE_FILTER={f:?} matched no test")),
        _ => Ok(()),
    }
}

/// Entry point of the generated test target. Returns the process exit code.
///
/// `src_dir` is the crate's `src/` directory (the hand-written shim passes
/// `CARGO_MANIFEST_DIR/src`); the digest of **every** `src/**/*.rs` file, framed
/// by `manifest::frame_sources`, goes into the manifest as `source_sha256`, the
/// same definition the proof and bench manifests and the checker use.
pub fn run_main(src_dir: &std::path::Path) -> i32 {
    match manifest::source_tree_framing(src_dir) {
        Ok(framed) => *SOURCE_SHA256.lock().unwrap() = sha256_hex(&framed),
        Err(e) => {
            eprintln!("cannot bind evidence to {}: {e}", src_dir.display());
            return 2;
        }
    }
    std::panic::set_hook(Box::new(|_| ()));
    let fault_var = std::env::var("WARRANT_EVIDENCE_FAULT").ok();
    let filter_var = std::env::var("WARRANT_EVIDENCE_FILTER").ok();
    let fault = match resolve_sabotage(fault_var.as_deref(), filter_var.as_deref()) {
        Ok(f) => f,
        Err(msg) => {
            eprintln!("{msg}");
            return 2;
        }
    };
    let suite = build_suite(fault);
    let registered = META.lock().unwrap().len();
    if let Err(msg) = require_tests(registered, filter_var.as_deref()) {
        eprintln!("{msg}");
        return 2;
    }
    run::main_recorded(&suite, write_manifest)
}

impl Meta {
    fn new(id: &str, kind: &'static str, op: &str, claim: &str) -> Meta {
        Meta {
            id: id.to_string(),
            kind,
            op: op.to_string(),
            claim: claim.to_string(),
            fault: String::new(),
            target: String::new(),
            domain: String::new(),
            seed: String::new(),
//@if v1
            demoted: Vec::new(),
//@endif
        }
    }
}

fn example_verdict(ok: bool) -> Verdict {
    if ok {
        Verdict::Pass
    } else {
        Verdict::fail("not-deep-equal", "example expression did not evaluate to true")
    }
}

/// A negative control passes iff at least one suite caught the injected fault.
fn negctl_result(caught: Vec<&str>, fault: &str) -> (Verdict, Stats, String) {
    if caught.is_empty() {
        (
            Verdict::fail(
                "corpus-mismatch",
                format!("fault {fault} was accepted by every suite of the operation"),
            ),
            Stats::default(),
            String::new(),
        )
    } else {
        (Verdict::Pass, Stats::default(), caught.join(","))
    }
}
