\chapter{The First Verbs}
\label{ch:first-verbs}

Until now, ``to be.'' Now verbs that \emph{act} --- and the gloriously simple way
Bengali conjugates them:

\begin{center}
\bn{আমি বাংলা বলি।} \quad(\emph{\=ami b\=angl\=a b\^{o}li} --- ``I speak Bengali.'')
\end{center}

\emph{Practice lessons: \texttt{lessons/BN-C05-*.md}.}

\section{\bn{বলা} \emph{(b\^{o}l\=a)} --- ``to speak,'' your first full verb}
\label{lesson:bola}

\begin{cousinweb}
\bn{বলা} (\emph{b\^{o}l\=a}, ``to speak, say'') is a native Indo-Aryan verb (the
same \emph{bol-} as Hindi \emph{boln\=a}, Punjabi \emph{bol\d{n}\=a}). The ending
-\bn{আ} (\emph{-\=a}) is the infinitive; add -\bn{ই} (\emph{-i}) to the stem for
``I speak'': \emph{b\^{o}l-} $+$ \emph{-i} $\to$ \bn{বলি} (\emph{b\^{o}li}).
\end{cousinweb}

\begin{grammarlens}
\textbf{The verb changes for person --- but \emph{never} for gender.} \emph{\=ami
b\^{o}li} (I say), \emph{tumi b\^{o}lo} (you say), \emph{se b\^{o}le} (he/she
says). There is no ``he-form'' vs. ``she-form'': \emph{se b\^{o}le} is ``he
says'' \emph{and} ``she says.'' Where Hindi and Punjabi make even ``I speak''
agree with the speaker's gender (\emph{bolt\=a/bolt\=\i}), Bengali dropped
grammatical gender entirely --- one less thing to track, forever.
\end{grammarlens}

\section{\bn{আমি বাংলা বলি} --- ``I speak Bengali''}
\label{lesson:ami-bangla-boli}

\bn{আমি} (I) $+$ \bn{বাংলা} (\emph{b\=angl\=a}, the object) $+$ \bn{বলি} (speak) ---
``I Bengali speak,'' verb last. The name \bn{বাংলা} (\emph{b\=angl\=a}) is the
language's own word for itself (English ``Bengali'' comes from it), descending
from \bn{বঙ্গ} (\emph{b\^{o}ngo}), the ancient name of the region --- so the
language is named after its land, the great delta of the Ganges.

\begin{grammarlens}
Because Bengali verbs carry no gender, this sentence is the \emph{same} whether a
man or a woman says it --- unlike Hindi. And since \emph{-i} already means ``I,''
you could drop \emph{\=ami}: \emph{b\=angl\=a b\^{o}li} is a complete ``I speak
Bengali.''
\end{grammarlens}

\section{\bn{থাকা} \emph{(th\=ak\=a)} --- ``to live, to stay''}
\label{lesson:thaka}

\begin{cousinweb}
\bn{থাকা} (\emph{th\=ak\=a}, ``to live, stay, remain'') is from Sanskrit \bn{স্থা}
(\emph{sth\=a}, ``to stand, stay''), the PIE root *steh\textsubscript{2}- that
also gives English \textbf{stand}, \textbf{stay}, \textbf{state} and Latin
\emph{st\=are}. ``I live in Kolkata'' is \bn{আমি কলকাতায় থাকি} (\emph{\=ami
Kolk\=at\=ay th\=aki}) --- stem $+$ the \emph{-i} ``I,'' no gender.
\end{cousinweb}

\begin{grammarlens}
\textbf{``In'' comes after the noun.} \bn{কলকাতায়} (\emph{Kolk\=at\=ay}) $=$
\emph{Kolkata} $+$ \emph{-y} (``in, at''), on the noun's \emph{end}. Bengali uses
\textbf{post}positions where English uses prepositions.
\end{grammarlens}

\section{\bn{কাজ করা} \emph{(k\=aj k\^{o}r\=a)} --- ``to work''}
\label{lesson:kaj-kora}

\begin{cousinweb}
\bn{করা} (\emph{k\^{o}r\=a}, ``to do, make'') is from Sanskrit \emph{k\d{r}}
(``to do''), PIE *k\ensuremath{^w}er- --- the root of Chapter 1's \emph{n\^{o}moshkar}
(\emph{namas} $+$ \emph{k\=ara}, ``the \emph{making} of a bow'') and of
\emph{karma}. And \bn{কাজ} (\emph{k\=aj}, ``work'') is itself from Sanskrit
\emph{k\=arya} (also from \emph{k\d{r}}). Bengali builds compounds as
\textbf{noun $+$ \bn{করা}}: \bn{কাজ করা} (\emph{k\=aj k\^{o}r\=a}, ``work-do'' $=$
``to work''). So \emph{\=ami k\=aj k\^{o}ri} $=$ ``I work.''
\end{cousinweb}

\section{Sentences about yourself}
\label{BN-C05-practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Bengali & English \\
\midrule
\bn{আমি বাংলা বলি।} & I speak Bengali. \\
\bn{আমি কলকাতায় থাকি।} & I live in Kolkata. \\
\bn{আমি কাজ করি।} & I work. \\
\bottomrule
\end{tabular}
\end{center}

Three verbs, one gender-free engine: the \emph{-i} ``I'' ending, the verb always
last, and \textbf{never} a change for gender. Roots banked: \emph{b\^{o}l\=a}
(native), \emph{th\=ak\=a} (\emph{sth\=a} $\to$ English \emph{stand/stay}),
\emph{k\^{o}r\=a} (\emph{k\d{r}} --- root of \emph{n\^{o}moshkar}; \emph{k\=aj}
``work'' $\leftarrow$ \emph{k\=arya}), and \emph{b\=angl\=a} (\emph{b\^{o}ngo}, the
Ganges delta). From here, Bengali's sentences only grow.
