\chapter{Farewells}
\label{ch:farewells}

Chapter 1 gave you the quick \gu{આવજો} (\emph{\=avjo}, ``come again''). Its
explicit companion:

\begin{center}
\gu{પાછા મળીશું}! \quad(\emph{p\=achh\=a ma\d{l}\=\i shũ} --- ``We'll meet again!'')
\end{center}

\emph{Practice lessons: \texttt{lessons/GU-C04-*.md}.}

\section{\gu{પાછા} \emph{(p\=achh\=a)} --- ``again, back''}
\label{lesson:pachha}

\begin{cousinweb}
\gu{પાછા} (\emph{p\=achh\=a}, ``back, again'') is from Sanskrit \emph{pa\'{s}c\=at}
(``behind, afterward''). It carries the sense of \emph{return}: \emph{p\=achh\=a
\=avvũ} (``to come back''). Add it to ``we'll meet'' for the warm goodbye.
\end{cousinweb}

\section{\gu{મળીશું} \emph{(ma\d{l}\=\i shũ)} --- ``(we) will meet''}
\label{lesson:malishun}

\begin{cousinweb}
\gu{મળીશું} (\emph{ma\d{l}\=\i shũ}, ``we will meet'') is the future of \gu{મળવું}
(\emph{ma\d{l}vũ}, ``to meet''), from Sanskrit \emph{mil-} (``to join'') --- the
same root as Hindi \emph{miln\=a}. Note the retroflex \gu{ળ} (\emph{\d{l}}):
Gujarati keeps this extra letter, like Marathi and the Dravidian south, unlike
Hindi.
\end{cousinweb}

\begin{grammarlens}
\textbf{The future is an ending.} \emph{ma\d{l}-} $+$ \emph{-\=\i shũ} (``we
will'') $\to$ \emph{ma\d{l}\=\i shũ} --- no helper like English ``will.''
\end{grammarlens}

\section{\gu{પાછા મળીશું} --- ``we'll meet again''}
\label{lesson:pachha-malishun}

\gu{પાછા} (again) $+$ \gu{મળીશું} (we will meet) $=$ ``\textbf{we'll meet
again}'' --- the explicit form of Chapter 1's \emph{\=avjo}, the subcontinent's
``promise of return'' (Hindi \emph{phir milenge}, Bengali \emph{\=ashi}).

\section{\gu{કાલે મળીશું} \emph{(k\=ale ma\d{l}\=\i shũ)} --- ``see you tomorrow''}
\label{lesson:kaale}

\begin{cousinweb}
\gu{કાલ} (\emph{k\=al}) means \textbf{both ``tomorrow'' and ``yesterday''} --- the
day on either side of today --- from Sanskrit \emph{k\=ala} (``time''), the root
behind Hindi's \emph{kal} and Punjabi's \emph{ak\=al} (``timeless''). The tense
tells which: \emph{k\=ale ma\d{l}\=\i shũ} (future $\to$ ``tomorrow we'll meet'').
\end{cousinweb}

\section{The farewells}
\label{lesson:GU-C04-practice}

\begin{tabular}{@{}ll@{}}

Roots banked: \emph{\=avjo} (``come again,'' from \emph{\=avvũ}), \emph{ma\d{l}vũ}
(``to meet,'' Sanskrit \emph{mil-}; the retroflex \gu{ળ}), \emph{k\=al}
(\emph{k\=ala} ``time,'' cousin of Hindi \emph{kal}, Punjabi \emph{ak\=al}). Next
chapter: the first real verbs --- \emph{hũ gujar\=at\=\i\ bolũ chhũ}.
