% One vendored font covers all three Japanese writing systems, because all three
% appear inside the same sentence. No `Script=` option is given: kana and kanji
% need no complex shaping, and naming a script fontspec cannot resolve would
% fail the build for no benefit. Every requested shape is mapped to the same
% complete file so a bold or italic teaching context cannot silently fall back
% to a system font.
\newfontfamily\japanesefont[
  Path=../../_fonts/,
  BoldFont=NotoSansJP-Subset.ttf,
  ItalicFont=NotoSansJP-Subset.ttf,
  BoldItalicFont=NotoSansJP-Subset.ttf
]{NotoSansJP-Subset.ttf}

% Inline Japanese snippet within English text. The generated chapters emit one
% \ja{...} per run of hiragana, katakana, or Han characters.
\newcommand{\ja}[1]{{\japanesefont #1}}

\pdfstringdefDisableCommands{%
  \def\ja#1{#1}% Keep the Japanese text while dropping font-only presentation.
  \def\japanesefont{}%
}
