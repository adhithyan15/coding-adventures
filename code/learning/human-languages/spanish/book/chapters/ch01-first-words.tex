\chapter{First Words}
\label{ch:first-words}

Ten words and phrases --- greetings, courtesies, a first question, a first
verb --- each taken fully apart. This is the whole method in miniature: you
will not memorize a list, you will meet ten members of families you already
belong to. \emph{Practice lessons: \texttt{lessons/ES-C01-L01} through
\texttt{L10}.}

\section{\emph{hola} --- hello}
\label{lesson:hola}

\begin{sounds}
The \textbf{h} is silent: \emph{hola} is said \emph{OH-la}, never
\emph{HOH-la}. True of every Spanish \emph{h}.
\end{sounds}

\emph{hola} is the informal ``hi'' --- for friends, children, familiar faces.
It carries no time of day and no deference, which is exactly why formal or
first-meeting situations reach instead for \emph{buenos d\'{i}as} and its kin.

\begin{cousinweb}
\emph{hola}'s origin is genuinely \textbf{unsettled} --- an old hailing call,
and that is about as far as firm evidence goes. In particular, its
resemblance to English \emph{hello} is a \textbf{false friend}: \emph{hello}
is Germanic (\emph{hail}, \emph{holler}), a completely different family. The
two sound alike by pure coincidence. Enjoy \emph{hola}, but don't file it
next to \emph{hello} --- a fake connection is worse than none.
\end{cousinweb}

\section{\emph{buenos d\'{i}as} --- ``good days,'' a little blessing}
\label{lesson:buenos-dias}

\begin{sounds}
\textbf{ue} in \emph{bueno} is a quick \emph{WEH} (\emph{BWEH-no}); the accent
in \emph{d\'{i}as} pulls the stress onto the \emph{i} (\emph{DEE-as}).
\end{sounds}

\begin{cousinweb}
\textbf{bueno} $\leftarrow$ Latin \emph{bonus} (``good''). Its adverb-sibling
\emph{bene} (``well''), the same ancient root, is the piece hiding in
\textbf{bene}volent (``wishing well''), \textbf{bene}fit (``doing well''),
\textbf{bene}diction (``speaking well'' --- a blessing), \textbf{bene}factor,
and worn-down \emph{bonus}, \emph{bonanza}, \emph{bounty}.\\[3pt]
\textbf{d\'{i}as} (sing.\ \emph{d\'{i}a}) $\leftarrow$ Latin \emph{dies}
(``day'') $\to$ \textbf{di}ary, \textbf{di}urnal, \textbf{jour}nal and
\textbf{jour}ney (French \emph{jour}, ``a day''), quoti\textbf{di}an,
meri\textbf{di}an, and \emph{dismal} (\emph{dies mali}, ``evil days'').
\end{cousinweb}

\begin{culture}
Why \emph{d\'{i}as}, plural? Because \emph{buenos d\'{i}as} is a fossil ---
the worn-down stub of \emph{``Buenos d\'{i}as os d\'{e} Dios,''} ``\emph{May
God give you good days}.'' You weren't wishing one good morning; you were
asking heaven for many good days ahead. The plural is the last bone of that
prayer, which is why it feels warm and formal where \emph{hola} feels casual
--- it is, almost literally, a small \emph{bene}diction. (Widely accepted;
exact history debated.)
\end{culture}

\section{\emph{buenas tardes} --- good afternoon}
\label{lesson:buenas-tardes}

\begin{cousinweb}
\textbf{buenas} is \emph{bueno} again (\S\ref{lesson:buenos-dias}), now
feminine to match \emph{tardes} --- gender-agreement in the wild.\\[3pt]
\textbf{tarde} $\leftarrow$ Latin \emph{tardus}, which meant not ``afternoon''
but \textbf{``slow, late''} --- the meaning English kept in \textbf{tard}y,
re\textbf{tard}, and \textbf{tard}igrade (``slow-stepper,'' the water-bear).
The afternoon \emph{is} the late part of the day; Spanish let the word for
``late'' drift to name the lateness itself.
\end{cousinweb}

\begin{culture}
Same amputated blessing as \S\ref{lesson:buenos-dias}. The boundary is
cultural, not clock-bound: \emph{tardes} runs from after lunch until it is
genuinely dark --- Spanish follows the daylight, not 12:00 sharp.
\end{culture}

\section{\emph{buenas noches} --- good evening \emph{and} goodnight}
\label{lesson:buenas-noches}

\begin{cousinweb}
\textbf{noche} $\leftarrow$ Latin \emph{nox}, stem \emph{noct-} $\to$
\textbf{noct}urnal, \textbf{noct}urne, and equi\textbf{nox} (``equal
night''). The same root, smoothed, also gives French \emph{nuit}, Italian
\emph{notte}.
\end{cousinweb}

\begin{culture}
The same blessing --- but \emph{buenas noches} works as both a greeting
\emph{and} a farewell. Arriving at 9pm: ``good evening.'' Leaving at midnight:
``goodnight.'' English splits these into two phrases; Spanish uses one, and
lets context tell you the direction. The blessing doesn't care which way
you're going.
\end{culture}

\section{\emph{gracias} --- thanks, and a family you already own}
\label{lesson:gracias}

\begin{cousinweb}
\textbf{gracias} is the plural of \emph{gracia}, ``grace/favor'' ---
literally ``graces.'' $\leftarrow$ Latin \emph{gratia}, from \emph{gratus}
(``pleasing, grateful''). The family: \textbf{grace}, \textbf{grat}itude,
\textbf{grate}ful, \textbf{grat}uity, \textbf{grat}is, in\textbf{grat}e
(\emph{in-} ``not'' $+$ grat), con\textbf{grat}ulate, and --- through French
\emph{\`{a} gr\'{e}} --- even \textbf{agree}. Not a foreign word to memorize,
but the Spanish member of a family you've known your whole life.
\end{cousinweb}

Note the first prefix in action: \emph{in-} on the front of \emph{-grate}
negates it --- an \emph{ingrate} is someone \emph{not} grateful.

\section{\emph{por favor} --- ``for a favor''}
\label{lesson:por-favor}

\begin{cousinweb}
\textbf{por} $\leftarrow$ Latin \emph{pro} (``for, on behalf of'') --- which
you own directly as English \textbf{pro}, \textbf{pro}ponent, and buried in
\emph{purpose}, \emph{portray}.\\[3pt]
\textbf{favor} $\leftarrow$ Latin \emph{favor}, from \emph{favere} (``to be
kindly disposed'') $\to$ \textbf{favor}able, \textbf{favor}ite,
\textbf{favor}itism.
\end{cousinweb}

\begin{culture}
English ``please'' is a worn-smooth ``if you please'' --- its politeness
hidden. Spanish keeps the gesture \textbf{visible}: \emph{por favor} = ``for
a favor,'' still transparently asking a kindness. The same instinct as the
greeting-blessings --- courtesy kept on the surface.
\end{culture}

\section{\emph{adi\'{o}s} --- ``to God''}
\label{lesson:adios}

\begin{cousinweb}
\textbf{adi\'{o}s} is \emph{a} $+$ \emph{Dios}. \textbf{a} $\leftarrow$ Latin
\emph{ad} (``toward'') --- the workhorse prefix in \textbf{ad}vance,
\textbf{ad}here, \textbf{ad}mire (and disguised in \emph{ac}quire,
\emph{at}tract, where it absorbs the next letter). \textbf{Dios}
$\leftarrow$ Latin \emph{Deus} (``god'') $\to$ \textbf{deit}y,
\textbf{div}ine, \textbf{deif}y --- and French \textbf{adieu}, the exact same
construction (\emph{\`{a}} $+$ \emph{Dieu}, ``to God'').
\end{cousinweb}

\begin{culture}
Shortened from \emph{``a Dios seas/vayas''} --- ``\emph{may you go to God},''
a real blessing for someone leaving on a once-dangerous journey. That's why
\emph{adi\'{o}s} carries a weight of finality; for ``see you later,'' Spanish
prefers \emph{hasta luego}. But every \emph{adi\'{o}s} still quietly means
``go with God.''
\end{culture}

\section{\textquestiondown\emph{c\'{o}mo est\'{a}s}? --- how are you?}
\label{lesson:como-estas}

\begin{cousinweb}
\textbf{c\'{o}mo} $\leftarrow$ Latin \emph{quomodo} (``in what \emph{manner}'')
--- that \emph{modo} is English \textbf{mode}, \textbf{method}, \textbf{model},
\textbf{mod}erate.\\[3pt]
\textbf{est\'{a}s} (from \emph{estar}, ``to be'') $\leftarrow$ Latin
\emph{stare}, which meant \textbf{``to stand''} --- one of English's richest
roots: \textbf{st}and, \textbf{st}ate, \textbf{st}atus, \textbf{st}able,
\textbf{st}ation, con\textbf{st}ant (``standing with''), circum\textbf{st}ance
(``standing around''), sub\textbf{st}ance (``standing under'').
\emph{\textquestiondown C\'{o}mo est\'{a}s?} literally asks ``how are you
\emph{standing}?''
\end{cousinweb}

\begin{grammarlens}
Spanish has \emph{two} verbs for ``to be.'' This is \emph{estar}, for passing
states and conditions (how you are \emph{right now}) --- so it's the one used
for mood and health. (The other, \emph{ser}, is for lasting identity; more
later.) \emph{est\'{a}s} is also the informal ``you''; the formal swaps a
letter: \emph{\textquestiondown c\'{o}mo est\'{a}?}
\end{grammarlens}

\section{\emph{me llamo} --- ``I call myself''}
\label{lesson:me-llamo}

\begin{sounds}
\textbf{ll} sounds like English \emph{y}: \emph{llamo} is \emph{YA-mo}.
\end{sounds}

\begin{cousinweb}
\textbf{llamo} (from \emph{llamar}, ``to call'') $\leftarrow$ Latin
\emph{clamare} (``to shout, cry out'') --- a root that shouts through English
via prefixes: ex\textbf{claim} (\emph{ex-} ``out''), pro\textbf{claim}
(\emph{pro-} ``forth''), ac\textbf{claim} (\emph{ad-} ``toward''),
re\textbf{claim}, dis\textbf{claim}, and \textbf{clam}or itself. Snap a
different prefix on \emph{-claim} and you get a different, precise verb.
\end{cousinweb}

\begin{culture}
Spanish doesn't say ``my name is.'' It says \emph{me llamo} --- ``I call
\emph{myself}'' --- presenting your name not as a fact you \emph{have} but as
what you \emph{do}. The \emph{me} makes it reflexive (the action loops back on
you). To ask it: \emph{\textquestiondown c\'{o}mo te llamas?}, ``how do you
call yourself?'' --- reusing \emph{c\'{o}mo} from \S\ref{lesson:como-estas}.
\end{culture}

\section{\emph{quiero} --- ``I want,'' and a masterclass in prefixes}
\label{lesson:quiero}

\begin{sounds}
\textbf{qu} is a hard \emph{k} with a \textbf{silent u}: \emph{quiero} is
\emph{KYEH-ro}.
\end{sounds}

\begin{cousinweb}
\textbf{quiero} (from \emph{querer}, ``to want'') $\leftarrow$ Latin
\emph{quaerere}, ``\textbf{to seek}.'' Plainly: \textbf{quer}y (a seeking),
\textbf{quest}, \textbf{quest}ion. Then the prefix machine takes over:
\end{cousinweb}

\begin{prefixtable}
\emph{in-} & into & in + quire & \textbf{inquire} & to seek into \\
\emph{re-} & repeatedly & re + quire & \textbf{require} & to seek insistently \\
\emph{ac-} (ad-) & toward & ac + quire & \textbf{acquire} & to seek toward $\to$ get \\
\emph{con-} & fully & con + quer & \textbf{conquer} & to seek fully $\to$ overcome \\
\emph{ex-} & out & ex + quisite & \textbf{exquisite} & sought out $\to$ choice \\
\emph{per-} & through & per + quisite & \textbf{perquisite} & sought through a job $\to$ \emph{perk} \\
\end{prefixtable}

Put \emph{in-} in front of \emph{-quiry} and you get \emph{inquiry}. Every
word above is \emph{quiero}'s cousin --- one root, a handful of prefixes, a
wall of English words. (False friend: \emph{quarrel}/\emph{querulous} come
from a \emph{different} root, \emph{queri} ``to complain'' --- resemblance
isn't proof.)

\begin{culture}
The twist: \emph{querer} means ``to want'' \emph{and}, of a person, ``to
love.'' \emph{Te quiero} = ``I love you'' --- literally ``I want/seek you.''
From the root's view, loving is \emph{seeking}: the same \emph{quaerere} that
gives \emph{quest} and \emph{inquiry}. Where English keeps ``want'' and
``love'' apart, Spanish lets one word of seeking cover both.
\end{culture}

Everything in this chapter now composes: \emph{Hola, me llamo [name].
\textquestiondown C\'{o}mo est\'{a}s? \ldots\ Un caf\'{e}, por favor.
Gracias. Adi\'{o}s.} --- ten words, and not one of them memorized cold.
