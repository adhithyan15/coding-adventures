\chapter*{Pronunciation --- a reference, not a chapter}
\addcontentsline{toc}{chapter}{Pronunciation (reference)}
\markboth{Pronunciation}{Pronunciation}

Not to be read before starting. Each lesson names the sounds its word needs
and points here for more. French spelling hides more sound than Spanish ---
silent letters, nasal vowels, liaison --- so this page is worth dipping into,
but it is never a wall to climb first.

\section*{Three habits to unlearn}

\begin{itemize}
  \item \textbf{Silent final consonants.} Most are silent: \emph{salut} =
    \emph{sa-LU} (no \emph{t}); \emph{les} = \emph{lay} (no \emph{s}).
    Usually kept: final \emph{c, r, f, l}.
  \item \textbf{Nasal vowels.} \emph{on, an, in} send the vowel through the
    nose and silence the \emph{n}: \emph{bon} = \emph{b\~{o}}, \emph{bien} =
    \emph{by\~{e}}.
  \item \textbf{Liaison.} A silent final consonant can reconnect onto a
    following vowel (\emph{vous avez} = \emph{vou-zavez}). Later.
\end{itemize}

\section*{Vowels}

\begin{itemize}
  \item \textbf{ou} = \emph{oo} (\emph{jour} = \emph{zhoor}).
  \item \textbf{u} = a front-rounded \emph{ee} with pursed lips (\emph{salut},
    \emph{tu}) --- no English equivalent, not \emph{oo}.
  \item \textbf{oi} = \emph{wa} (\emph{soir} = \emph{swahr}).
  \item \textbf{\'{e}} = crisp \emph{ay}; \textbf{\`{e}/\^{e}} = open \emph{eh}.
\end{itemize}

\section*{Nasal vowels}

\begin{itemize}
  \item \textbf{on/om} = nasal \emph{\~{o}} (\emph{bon}, \emph{bonjour}).
  \item \textbf{in/ain/ien} = nasal \emph{\~{e}} (\emph{bien}, \emph{bient\^{o}t}).
  \item \textbf{an/en} = nasal \emph{\~{a}} (\emph{comment}, \emph{enchant\'{e}}).
  \item A doubled \emph{nn}/\emph{mm} \emph{cancels} the nasal: \emph{bon}
    (\emph{b\~{o}}) vs.\ \emph{bonne} (\emph{bun}).
\end{itemize}

\section*{Consonants that differ}

\begin{itemize}
  \item \textbf{j} (and soft \textbf{g}) = the \emph{zh} of \emph{measure}
    (\emph{jour}).
  \item \textbf{r} = a soft guttural at the back of the throat --- not the
    Spanish tap nor the English \emph{r}.
  \item \textbf{h} = always silent. \textbf{ch} = English \emph{sh}
    (\emph{enchant\'{e}}). \textbf{gn} = \emph{ny} (= Spanish \~{n}).
\end{itemize}

\section*{Accents}

French accents mark \emph{vowel quality}, not stress (unlike Spanish): \'{e}
(closed), \`{e}/\^{e} (open), \c{c} (soft \emph{s}). French stresses the last
syllable of a group, lightly and evenly.
