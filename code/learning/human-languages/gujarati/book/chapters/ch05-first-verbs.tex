\chapter{The First Verbs}
\label{ch:first-verbs}

Until now, ``to be.'' Now verbs that \emph{act} --- and sentences that move:

\begin{center}
\gu{હું ગુજરાતી બોલું છું}. \quad(\emph{hũ gujar\=at\=\i\ bolũ chhũ} --- ``I speak Gujarati.'')
\end{center}

\emph{Practice lessons: \texttt{lessons/GU-C05-*.md}.}

\section{\gu{બોલવું} \emph{(bolvũ)} --- ``to speak,'' your first full verb}
\label{lesson:bolvun}

\begin{cousinweb}
\gu{બોલવું} (\emph{bolvũ}, ``to speak'') is a native Indo-Aryan verb (the same
\emph{bol-} as Hindi \emph{boln\=a}, Punjabi \emph{bol\d{n}\=a}). Its ending
-\gu{વું} (\emph{-vũ}) is Gujarati's infinitive: \emph{bol-vũ} ``to speak,''
\emph{ma\d{l}-vũ} ``to meet,'' \emph{\=av-vũ} ``to come.'' Strip it for the stem
(\emph{bol-}).
\end{cousinweb}

\begin{grammarlens}
\textbf{Stem $+$ person $+$ copula.} \emph{hũ bolũ chhũ} (I speak), \emph{tũ bole
chhe} (you speak), \emph{tame bolo chho} (you, resp.), \emph{te bole chhe} (he/she
speaks). The plain present is the same for a man or a woman --- Gujarati saves its
\textbf{three genders} for adjectives and the past, not the simple present.
\end{grammarlens}

\section{\gu{હું ગુજરાતી બોલું છું} --- ``I speak Gujarati''}
\label{lesson:hun-gujarati-bolun-chhun}

\gu{હું} (I) $+$ \gu{ગુજરાતી} (\emph{gujar\=at\=\i}, the object) $+$ \gu{બોલું છું}
(speak) --- ``I Gujarati speak,'' verb last. The name \gu{ગુજરાત} (\emph{Gujar\=at})
comes from the \textbf{Gurjar}, a people who entered western India around the
6th--8th centuries and gave their name to the land (\emph{Gurjara-de\'{s}a} $\to$
\emph{Gujar\=at}); \emph{gujar\=at\=\i} is ``[the language] of Gujarat.''

\section{\gu{રહેવું} \emph{(rahevũ)} --- ``to live, to stay''}
\label{lesson:rahevun}

\begin{cousinweb}
\gu{રહેવું} (\emph{rahevũ}, ``to live, remain, stay''), from Sanskrit \emph{rah-}
(``to remain,'' the same as Hindi \emph{rahn\=a}): \emph{hũ Amd\=av\=adm\=a\d{m}
rahũ chhũ} (``I live in Ahmedabad'').
\end{cousinweb}

\begin{grammarlens}
\textbf{``In'' comes after the noun.} \gu{અમદાવાદમાં} (\emph{Amd\=av\=adm\=a\d{m}})
$=$ \emph{Ahmedabad} $+$ \emph{-m\=a\d{m}} (``in''), on the noun's \emph{end}.
Gujarati uses \textbf{post}positions, verb last.
\end{grammarlens}

\section{\gu{કામ કરવું} \emph{(k\=am karvũ)} --- ``to work''}
\label{lesson:kaam-karvun}

\begin{cousinweb}
\gu{કરવું} (\emph{karvũ}, ``to do, make'') is from Sanskrit \emph{k\d{r}} (``to
do''), PIE *k\ensuremath{^w}er- --- the root of Chapter 1's family: \emph{namask\=ar}
(``the \emph{making} of a bow''), \emph{karma} (``a thing \emph{done}''). And
\gu{કામ} (\emph{k\=am}, ``work'') is itself from Sanskrit \emph{karma}. Gujarati
builds compounds as \textbf{noun $+$ \gu{કરવું}}: \gu{કામ કરવું} (\emph{k\=am
karvũ}, ``work-do'' $=$ ``to work''). So \emph{hũ k\=am karũ chhũ} $=$ ``I work''
--- the twin of Hindi's \emph{k\=am karn\=a} and Bengali's \emph{k\=aj kar\=a}.
\end{cousinweb}

\section{Sentences about yourself}
\label{lesson:GU-C05-practice}

\begin{center}
\begin{tabular}{@{}ll@{}}
\toprule
Gujarati & English \\
\midrule
\gu{હું ગુજરાતી બોલું છું}. & I speak Gujarati. \\
\gu{હું અમદાવાદમાં રહું છું}. & I live in Ahmedabad. \\
\gu{હું કામ કરું છું}. & I work. \\
\bottomrule
\end{tabular}
\end{center}

Three verbs, one engine: a stem, a person-ending, the copula forms
\emph{chhũ}, \emph{chhe}, and \emph{chho}
--- the verb always last, ``in'' after its noun. Roots banked: \emph{bolvũ}
(native), \emph{rahevũ} (\emph{rah-} ``remain''), \emph{karvũ} (\emph{k\d{r}} ---
root of \emph{namask\=ar}, \emph{karma}; \emph{k\=am} ``work'' $\leftarrow$
\emph{karma}), and \emph{gujar\=at\=\i} (the \emph{Gurjar} people). From here,
Gujarati --- the tongue of Gandhi --- only opens further.
