\chapter{Greetings}
\label{ch:greetings}

The Sanskrit greetings --- and why they are the other \textbf{taproot} of this
whole curriculum. Every ``thank you'' and ``hello'' in the Hindi, Marathi,
Punjabi, and Bengali tracks points back here; and Sanskrit is at the same time a
\emph{sister} of Latin, Greek, and English, so its words reach west as well as
east. Each lesson introduces the letters and the Sanskrit-specific reading its
word needs. \emph{Practice lessons: \texttt{lessons/SA-C01-*.md}.}

\section{\sk{नमस्ते} \emph{(namaste)} --- ``I bow to you''}
\label{lesson:namaste}

\begin{sounds}
Devanagari runs \textbf{left to right} under a top line; every consonant carries
a built-in ``a.'' \sk{न} \emph{na}, \sk{म} \emph{ma}, \sk{ते} \emph{te} (\sk{त}
$+$ the \emph{e}-sign \sk{े}). A \emph{hasanta} under \sk{स्} strips its vowel so
it joins the next: \sk{स्} $+$ \sk{त} $\to$ \sk{स्त}. Read \sk{नमस्ते} as
\emph{na\,·\,ma\,·\,s\,·\,te}.
\end{sounds}

\begin{cousinweb}
\sk{नमस्ते} $=$ \emph{namas} (``a bow, obeisance'') $+$ \emph{te} (``to you'') ---
literally ``\textbf{a bow to you}.'' This is the \emph{original}: the word every
Indo-Aryan track in this curriculum descends from --- Hindi, Marathi, Punjabi,
Bengali all say a worn form of it. \emph{Namas} comes from the root
\emph{\textsurd{}nam} ``to bend, bow,'' from Proto-Indo-European \emph{*nem-} ``to
allot, take.'' And \emph{te} ``to you'' is a family heirloom: the same
Indo-European pronoun behind Latin \emph{t\=e}, Greek \emph{se}, and archaic
English \textbf{thee}. Say \emph{namas-te} and you are speaking a word Cicero and
Chaucer would each have half-recognised.
\end{cousinweb}

\begin{culture}
Palms pressed together (the gesture is \emph{a\~{n}jali}), a slight bow: the hands
and the word say the same thing. \emph{Namaste} greets the person before you as
worthy of a bow --- hello, and, when parting, goodbye. It is the greeting Sanskrit
gave to half of South Asia.
\end{culture}

\section{\sk{नमस्कारः} \emph{(namask\=ara\d{h})} --- the fuller form}
\label{lesson:namaskara}

\begin{sounds}
New: \sk{का} \emph{k\=a} (\sk{क} $+$ the \=a-sign \sk{ा}), \sk{र} \emph{ra}, and
the \textbf{visarga} \sk{ः} --- a soft breathed ``\d{h}'' at a word's end,
written as two dots. \sk{नमस्कारः} $=$ \emph{namas-k\=a-ra\d{h}}.
\end{sounds}

\begin{cousinweb}
\sk{नमस्कारः} $=$ \emph{namas} (``a bow'') $+$ \emph{k\=ara} (``the making, the
doing,'' from the root \emph{\textsurd{}k\d{r}} ``to do/make''). So \emph{namask\=ara} is
``the \textbf{making} of a bow'' --- an act, where \emph{namaste} is an address.
That root \emph{\textsurd{}k\d{r}} ``to do'' is one of the most productive in Sanskrit
(it gives \emph{karma}, ``a thing done''); it is the Indo-European \emph{*k\d{w}er-}
and a distant cousin of English \emph{-ory}/\emph{-ure} words of making. The final
\textbf{visarga} \emph{-a\d{h}} is the mark of the masculine singular --- your
first case-ending.
\end{cousinweb}

\begin{grammarlens}
\emph{Namaste} and \emph{namask\=ara} share the same first half, \emph{namas}, but
end differently because they \emph{do} different grammatical jobs --- one takes a
pronoun (\emph{te} ``to you''), the other builds a noun (\emph{-k\=ara} ``the
making of''). Sanskrit words are transparently assembled from roots and endings
like this; once you see the seams, long words stop being frightening.
\end{grammarlens}

\section{\sk{धन्यवादः} \emph{(dhanyav\=ada\d{h})} --- ``thank you''}
\label{lesson:dhanyavada}

\begin{sounds}
New: \sk{ध} \emph{dha} (breathy \emph{d}), \sk{व} \emph{va}, \sk{द} \emph{da}, and
the conjunct \sk{न्य} (\sk{न} $+$ \sk{य} \emph{ya}). With the visarga again, read
\sk{धन्यवादः} as \emph{dha\,·\,nya\,·\,v\=a\,·\,da\d{h}}.
\end{sounds}

\begin{cousinweb}
\sk{धन्यवादः} $=$ \emph{dhanya} (``worthy, deserving,'' from \emph{dhana}
``wealth'') $+$ \emph{v\=ada} (``a speaking, a statement,'' from the root
\emph{\textsurd{}vad} ``to speak''). Literally ``a saying of "you are worthy."'' This is
the \emph{source} of the thanks you met to the east --- Hindi \emph{dhanyav\=ad},
Marathi and Punjabi likewise --- here in its full, un-worn Sanskrit shape, visarga
and all.
\end{cousinweb}

\section{\sk{स्वागतम्} \emph{(sv\=agatam)} --- ``welcome''}
\label{lesson:svagatam}

\begin{sounds}
New: the conjunct \sk{स्व} (\sk{स्} $+$ \sk{व} \emph{va}) $=$ \emph{sva}, \sk{ग}
\emph{ga}, and the \emph{halant} \sk{म्} \emph{m} closing the word with no vowel.
Read \sk{स्वागतम्} as \emph{sv\=a\,·\,ga\,·\,tam}.
\end{sounds}

\begin{cousinweb}
\sk{स्वागतम्} $=$ \emph{su} (``well, good'') $+$ \emph{\=agata} (``come,'' from
\emph{\=a} ``toward'' $+$ the root \emph{\textsurd{}gam} ``to go/come'') --- ``\textbf{well
come},'' which is exactly what English \emph{welcome} means too. Two deep
Indo-European threads surface here:
\begin{itemize}
  \item \emph{su-} ``good'' is the twin of \textbf{Greek \emph{eu-}} --- the
    \emph{eu} of \emph{euphoria}, \emph{eulogy}, \emph{euphemism}. Sanskrit
    \emph{su-gata} and Greek \emph{eu-} are the same prefix.
  \item the root \emph{\textsurd{}gam} ``to go/come'' is the Indo-European \emph{*g\d{w}em-}
    --- the very root of English \textbf{come} (and of Latin \emph{venire}, hence
    \emph{venue}, \emph{convene}).
\end{itemize}
So \emph{sv\=agatam} and \emph{welcome} are not just parallel ideas but partly
\emph{cognate} ones.
\end{cousinweb}

\section{\sk{आम्} / \sk{न} \emph{(\=am / na)} --- ``yes'' and ``no''}
\label{lesson:am-na}

\begin{sounds}
New: \sk{आ} --- the \textbf{independent} vowel \=a (the shape for a word-initial
vowel) --- closed by the halant \sk{म्} \emph{m}: \sk{आम्} \emph{\=am}. And \sk{न}
\emph{na}, a single consonant with its built-in ``a.''
\end{sounds}

\begin{cousinweb}
\sk{आम्} (\emph{\=am}, ``yes'') and \sk{न} (\emph{na}, ``no, not''). That
\emph{na} is a word you have already met wearing other clothes: it is
Proto-Indo-European \emph{*ne} ``not,'' \textbf{the same negative} as the Latin
\emph{n\=on} of the last track, as English \textbf{no}, \textbf{not},
\textbf{none}, as German \emph{nein}, and as Hindi \emph{nah\=\i\d{n}}. Of all the
words in this book, \emph{na} may be the one that reaches furthest unchanged: one
tiny syllable spoken, in some form, from Iceland to India for five thousand years.
\end{cousinweb}

\begin{culture}
Classical Sanskrit, like Latin, leans on more than a bare ``yes'': one often
answers by echoing the verb, or with \emph{\=am}, \emph{evam} (``so it is''), or
\emph{b\=a\d{d}ham} (``certainly''). The blunt yes/no habit of English is not
universal --- these old languages preferred to \emph{say what they meant}.
\end{culture}

\section{Recap}
\label{lesson:practice}

\begin{center}
\begin{tabular}{@{}lll@{}}
\toprule
Sanskrit & & Meaning \\
\midrule
\sk{नमस्ते} & \emph{namaste} & hello (``a bow to you'') \\
\sk{नमस्कारः} & \emph{namask\=ara\d{h}} & hello (``the making of a bow'') \\
\sk{धन्यवादः} & \emph{dhanyav\=ada\d{h}} & thank you \\
\sk{स्वागतम्} & \emph{sv\=agatam} & welcome (``well come'') \\
\sk{आम्} / \sk{न} & \emph{\=am / na} & yes / no \\
\bottomrule
\end{tabular}
\end{center}

Sanskrit is the eastern taproot: \emph{namas}, \emph{dhanya-v\=ada}, and the bow
that survives, worn smooth, as every Indo-Aryan hello and thank-you. But look
which way its roots also point --- \emph{te} ``to you'' toward English
\emph{thee}, \emph{su-} ``good'' toward Greek \emph{eu-}, \emph{\textsurd{}gam} toward
\emph{come}, \emph{na} toward \emph{no} and Latin \emph{n\=on}. When Sir William
Jones noticed in 1786 that Sanskrit, Latin, and Greek were kin, \emph{these} are
the kinds of words he meant. Next chapter: introducing yourself --- \emph{mama
n\=ama\ldots} (``my name\ldots'') --- and Sanskrit's astonishing verb, which
marks person, number, \emph{and} how many are speaking.
